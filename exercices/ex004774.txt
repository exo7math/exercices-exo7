\exercice{4774, quercia, 2010/03/16}

\enonce[Parties de \( \Rr^n \)]
Les parties suivantes sont-elles ouvertes ? fermées ? bornées ?

\begin{enumerate}
  \item \( A = \{ (x,y) \in \Rr^2 \text{ tq } xy = 1 \} \).

  \item \( B = \{ (x,y) \in \Rr^2 \text{ tq } x^2+xy+y^2 < 1 \} \).

  \item \( C = \{ z \in \Cc \text{ tq } \Re(z^2) \le 1 \} \).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
