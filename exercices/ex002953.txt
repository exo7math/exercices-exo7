\exercice{2953, quercia, 2010/03/08}

\enonce[Équation trigonométrique]
Soit \( a \in \Rr \). Résoudre :
\[
  \begin{cases}\cos(a) + \cos(a+x) + \cos(a+y) = 0 \\
    \sin(a) + \sin(a+x) + \sin(a+y) = 0.\end{cases}
\]
\finenonce

\noindication

\correction
\( x \equiv -y \equiv \pm \frac{2\pi}3 (\mathrm{mod}\, {2\pi}) \).
\fincorrection

\finexercice
