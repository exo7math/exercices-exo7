\exercice{6220, queffelec, 2011/10/16}

\enonce
Sur l'ensemble \( \Nn \) des entiers naturels, définissons
\begin{eqnarray*}
  d(n,m)&=& 0 \ \ {\rm pour}\ m=n\\
  &=& 1+\frac{1}{{n+m}}\ \ {\rm pour}\ m\neq n
\end{eqnarray*}

\begin{enumerate}
  \item Montrer que \( d \) est une métrique sur \( \Nn \) pour laquelle il est complet.

  \item Construire dans \( (\Nn,d) \) une suite de boules fermées non vides emboîtées dont les rayons ne tendent pas vers \( 0 \), et d'intersection vide.
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
