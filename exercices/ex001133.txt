\exercice{1133, barraud, 2003/09/01}

\enonce
On s'intéresse aux suites réelles \( (u_n)_{n\in\Nn} \) satisfaisant la relation de récurrence
\[
  \forall n\in\Nn\qquad u_{n+2} = \sqrt{2}u_{n+1}-u_n
  \rlap{\hspace{1cm}(\( \star \))}
\]
\begin{enumerate}
  \item Déterminer toutes les suites complexes satisfaisant la relation \( (\star) \).

  \item Déterminer toutes les suites réelles satisfaisant la relation \( (\star) \).

        On considère maintenant le déterminant d'ordre \( n \) suivant :
        \[
          \Delta_n = \left|
          \begin{matrix}
            \sqrt{2} & 1      &        &
            \hspace{-3ex}\raisebox{-2ex}[0pt][0pt]{\text{\LARGE{\( 0 \)}}} \\
            1        & \ddots & \ddots &                                   \\
                     & \ddots & \ddots & 1                                 \\
            \hspace{3ex}\raisebox{1ex}[0pt][0pt]{\text{\LARGE{\( 0 \)}}}
                     &        & 1      & \sqrt{2}
          \end{matrix}\,
          \right|
        \]

  \item Calculer \( \Delta_{n+2} \) en fonction de \( \Delta_{n+1} \) et \( \Delta_n \) pour \( n\in\Nn \) (on pose \( \Delta_0 = 1 \)).

        En déduire la valeur de \( \Delta_n \) en fonction de \( n \).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
