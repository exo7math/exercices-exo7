\exercice{925, legall, 1998/09/01}

\enonce[Caractérisation de la somme directe de trois s.e.v.]
Soient \( U, V, W \) des s.e.v. d'un e.v. \( E \), vérifiant \( (I) : U\cap V = \{0\} = (U+V)\cap W \).
\begin{enumerate}
  \item Démontrer que \( V\cap W = \{0\} = U\cap(V+W) \).\\

  \item Montrer que \( (I) \) équivaut à
        \[
          (II) : (\forall x \in U+V+W) (\exists ! (u,v,w) \in U\times V \times W) (x = u+v+w).
        \]
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
