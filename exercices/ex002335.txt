\exercice{2335, matexo1, 2002/02/01}

\enonce
Trouver les primitives des fonctions suivantes\,:
\[
  { \frac{1}{x \sqrt{x^2-1}}, \qquad {(4x^2+4x+5)^{-1/2}}, \qquad \frac{\sin^4 x}{\cos^2 x}, \qquad \frac{\arctan x}{1+x^2}, \qquad \frac{1}{x\,(1+\ln^2 x)}. }
\]
\finenonce

\noindication

\nocorrection

\finexercice
