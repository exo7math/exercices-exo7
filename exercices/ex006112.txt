\exercice{6112, queffelec, 2011/10/16}

\enonce
\begin{enumerate}
  \item Déterminer l'ensemble des valeurs d'adhérence de la suite de réels \( x_n = (1+\frac{1}{n})\sin(n\frac{\pi}{6}) \); de la suite \( (\frac{1}{m}+\frac{1}{n})_{m\ge1,n\ge1} \).

  \item Montrer que l'ensemble \( \Zz\alpha+\Zz \) est dense dans \( \Rr \) si \( \alpha \) est irrationnel. En déduire l'ensemble des valeurs d'adhérence de la suite \( x_n = \cos(2\pi n\alpha) \).

        (\emph{Indication :} on pourra montrer que tout sous-groupe fermé de \( \Rr \) est soit \( \Rr \), soit discret, de la forme \( a\Zz \).)
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
