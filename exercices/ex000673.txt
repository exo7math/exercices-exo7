\exercice{673, vignal, 2001/09/01}

\enonce
\begin{enumerate}
  \item Soit la fonction réelle définie par \( f(x) = 1 \) si \( x\in\Qq \) et \( f(x) = 0 \) sinon. Montrer que \( f \) n'admet pas de limite en tout point de \( \Rr \).

  \item Soit la fonction réelle définie par \( f(x) = x \) si \( x\in\Qq \) et \( f(x) = 1-x \) sinon. En quels points de \( \Rr \) \( f \) est‑elle continue ?
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
