\exercice{3549, quercia, 2010/03/10}

\enonce[Ensi PC 1999]
Soit \( A\in\mathcal{M}_n(\Cc) \) inversible et diagonalisable et \( B\in \mathcal{M}_n(\Cc) \), \( p\in\Nn^* \) tels que \( B^p = A \).

\begin{enumerate}
  \item Montrer que \( B \) est diagonalisable.

  \item Si \( A \) n'est pas inversible, la conclusion subsiste-t-elle ?
\end{enumerate}
\finenonce

\noindication

\correction
\begin{enumerate}
  \item Polynôme annulateur simple.

  \item Non, ctrex = \( B \) nilpotent.
\end{enumerate}
\fincorrection

\finexercice
