\exercice{1770, gourio, 2001/09/01}

\enonce
Soit \( k\in \Rr^{+*} \),
\[
  \omega_n = \left\{ \left( x,y\right) \in \Rr^2|\left( x-\frac{1}{n}%
  \right)^2+\left( y-\frac{1}{n}\right)^2\le \frac{k^2}{n^2}%
  \right\} ,
\]
et
\[
  \Omega = \bigcup\limits_{n\in \Nn^{*}}\omega_n.
\]
\( \Omega \) est-il ouvert? fermé? ...
\finenonce

\noindication

\nocorrection

\finexercice
