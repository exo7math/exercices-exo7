\exercice{2743, tumpach, 2009/10/25}

\enonce
\begin{enumerate}
  \item Soit \( f \) une application linéaire surjective de \( \Rr^4 \) dans \( \Rr^2 \). Quelle est la dimension du noyau de \( f \)?

  \item Soit \( g \) une application injective de \( \Rr^{26} \) dans \( \Rr^{100} \). Quelle est la dimension de l'image de \( g \)?

  \item Existe-t-il une application linéaire bijective entre \( \Rr^{50} \) et \( \Rr^{72} \)?
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
