\exercice{4672, quercia, 2010/03/16}

\enonce[Série alternée]
On pose \( u_n = \frac{96\times(-1)^n}{(2n-3)(2n-1)(2n+1)(2n+3)(2n+5)} \) et \( v_n = \sum_{k = 0}^n\,u_k \).

\begin{enumerate}
  \item Étudier les suites \( (v_{2n}) \) et \( (v_{2n+1}) \) et montrer que la suite \( (v_n) \) est convergente.

  \item Calculer \( \ell = \lim_{n\to\infty} v_n \) à \( 10^{-5} \) près.

\end{enumerate}
\finenonce

\noindication

\correction
\begin{enumerate}
  \item

  \item \( \ell = \pi \).
\end{enumerate}
\fincorrection

\finexercice
