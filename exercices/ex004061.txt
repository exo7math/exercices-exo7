\exercice{4061, quercia, 2010/03/11}

\enonce[\( y'' + |y| = 1 \)]
Résoudre l'équation \( y'' + |y| = 1 \), avec \( y(0) = 0,\ y'(0) = 1 \).
\finenonce

\noindication

\correction
\[
  y = \begin{cases}e^x-1             & \text{ si } x < 0                  \\
             1-\cos x + \sin x & \text{ si } 0 \le x < \frac{3\pi}2 \\
             e^{3\pi/2-x}-1    & \text{ si } x \ge \frac{3\pi}2.
  \end{cases}
\]
\fincorrection

\finexercice
