\exercice{2291, barraud, 2008/04/24}

\enonce
Soit \( A \) un anneau, \( B \) un sous-anneau de \( A \), \( I \) un idéal de \( A \).
\begin{enumerate}
  \item Montrer que \( B\cap I \) est un idéal de \( B \), \( B+I = \{b+i\,|\, b\in B,\ i\in I\} \) est un sous-anneau de l'anneau \( A \) et \( I \) est un idéal de ce sous-anneau.

  \item Montrer que l'anneau quotient \( B/(B\cap I) \) est isomorphe à l'anneau quotient \( (B+I)/I \). ({\it Indication :} Considérer le composé de l'inclusion \( B\to B+I \) avec la projection canonique \( B+I \to (B+I)/I \).)
\end{enumerate}
\finenonce

\noindication

\correction
\begin{enumerate}
  \item Soit \( J = B\cap I \). Soit \( x,y\in J \), \( a,b\in B \), alors \( ax+by\in B \) puisque \( B \) est un sous-anneau de \( A \). \( ax+by\in I \) puisque \( I \) est un idéal. On en déduit que \( J \) est un idéal.

        \( B+I \) est stable par addition (car \( B \) et \( I \) le sont). Soit \( \alpha = a+x\in B+I \) et \( \beta = b+y\in B+I \). Alors \( \alpha\beta = (ab)+(ay+bx+xy)\in B+I \), donc \( B+I \) est stable par multiplication. \( 1\in B+I \), donc \( B+I \) est un sous-anneau de \( A \). \( I\subset B+I \), et \( I \) est absorbant pour la multiplication dans \( A \), donc aussi dans \( B+I \): \( I \) est un idéal de \( B+I \).

  \item On a le diagramme (de morphismes d'anneaux) suivant :
        \begin{tikzpicture}[>=stealth, xscale=2.5, yscale=1.5] % Définition des nœuds (objets)
          \node (B) at (0, 1) {\( B \)};
          \node (BI) at (2, 1) {\( B+I \)};
          \node (End) at (4, 1) {\( (B+I)/I \)};
          \node (Quot) at (0, 0) {\( B/\ker\phi \)};

          % Définition des flèches (morphismes)
          % Flèches horizontales
          \draw[->] (B) -- node[above] {\( i \)} (BI);
          \draw[->] (BI) -- node[above] {\( \pi \)} (End);

          % Flèche verticale
          \draw[->] (B) -- node[left] {\( \pi_0 \)} (Quot);

          % Flèche diagonale
          \draw[->] (Quot) -- node[below, sloped] {\( \sim \)} (End);

          % Flèche courbe (le bend)
          \draw[->] (B) to[bend left=30] node[above] {\( \phi \)} (End);
        \end{tikzpicture}

        %
        \[
          %          \xymatrix{%
          %            B\ar[r]^i
          %            \ar[d]^{{\pi_0}}
          %            \ar@(ur,ul)[rr]^{\phi}
          %            & B+I \ar[r]^{\pi} & (B+I)/I\\
          %            B/\ker\phi\ar[urr]_{\sim}
          %          }%
          %
        \]
        Or, pour \( x\in B \), on a : \( x\in\ker\phi \Leftrightarrow x = i(x)\in
        \ker\pi = I \). Donc \( \ker\phi = B\cap I \), et par suite :
        \[
          B/(B\cap I)\sim (B+I)/I.
        \]
\end{enumerate}
\fincorrection

\finexercice
