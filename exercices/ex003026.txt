\exercice{3026, quercia, 2010/03/08}

\enonce[Corps \( {\mathbb F}_4 \)]
Chercher les structures de corps à 4 éléments.
\finenonce

\noindication

\correction
\( K = \{0,1,a,b\} \) et \( \{1,a,b\} \) est un groupe multiplicatif \( \Rightarrow b = a^2 \), \( a^3 = 1 \).
\[
  \begin{array}{c|cccc}
    +        & 0   & 1   & a   & a^2 \\
    \hline 0 & 0   & 1   & a   & a^2 \\
    1        & 1   & 0   & a^2 & a   \\
    a        & a   & a^2 & 0   & 1   \\
    a^2      & a^2 & a   & 1   & 0
  \end{array}
  \qquad % Table de multiplication
  \begin{array}{c|ccc}
    \times   & 1   & a   & a^2 \\
    \hline 1 & 1   & a   & a^2 \\
    a        & a   & a^2 & 1   \\
    a^2      & a^2 & 1   & a
  \end{array}
\]
\fincorrection

\finexercice
