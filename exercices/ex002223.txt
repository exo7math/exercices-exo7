\exercice{2223, matos, 2008/04/23}

\enonce[Stratégie de pivotage]
\ \\
\begin{enumerate}
  \item Montrer que pour une matrice quelconque \( A = (a_{ij}) \) de type \( (2\times 2 ) \) on a
        \[
          \mbox{cond}_2(A) = \sigma + (\sigma^2-1)^{1/2} \mbox{ avec } \sigma = \frac{\sum_{i,j = 1}^2 |a_{ij}|^2}{2|\det (A)|}
        \]

  \item Calculer les conditionnements \(\operatorname{cond}_p(\cdot)\) pour \( p = 1,2,\infty \) des matrices exactes obtenues à la première étape de la procédure d'élimination de Gauss pour résoudre le système linéaire
        \[
          \left\{
          \begin{array}{l}
            10^{-4} u_1 + u_2 = 1 \\
            u_1 + u_2 = 2
          \end{array}
          \right.
        \]
        selon que l'on commence, ou non, par échanger les deux équations. Conclusion?
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
