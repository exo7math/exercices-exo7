\exercice{1940, gineste, 2001/11/01}

\enonce[Séries à termes quelconques]
Étudier les séries de termes généraux
\begin{enumerate}
  \item
        \[
          u_n = \frac{(-1)^n}{(\ln n) (n^{1/n})}
        \]

  \item
        \[
          v_n = \frac{(-1)^n}{\sqrt{n^\alpha + (-1)^n}} \mbox{ où } \alpha > 0
        \]

  \item
        \[
          w_n = \ln(1 + \frac{(-1)^n}{n^\alpha}) \mbox{ où } \alpha > 0
        \]
\end{enumerate}
Indication : Des calculs de D.L. peuvent être fructueux ...
\finenonce

\noindication

\nocorrection

\finexercice
