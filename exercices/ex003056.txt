\exercice{3056, quercia, 2010/03/08}

\enonce[Bases de numération]
Soient \( n,p \in \Nn^* \) avec \( n < p! \). Montrer qu'il existe un unique \( p \)-uplet \( (n_1,\dots,n_p) \) d'entiers naturels tel que
\[
  \forall\ k \le p,\ n_k \le k,\quad\text{et}\quad n = \sum_{k = 1}^p n_kk!.
\]
\finenonce

\noindication

\nocorrection

\finexercice
