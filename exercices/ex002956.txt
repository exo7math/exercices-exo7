\exercice{2956, quercia, 2010/03/08}

\enonce[\( \sum C_n^kx^{n-k}\cos(k\alpha) = 0 \)]
Résoudre en \( x \) : \( x^n + C_n^1x^{n-1}\cos\alpha + \dots + C_n^n\cos(n\alpha) = 0 \).
\finenonce

\noindication

\correction
\( \frac 12\Bigl( (x+e^{i\alpha})^n + (x+e^{-i\alpha})^n \Bigr) = 0
\Leftrightarrow x = \cotan\left(\frac{(2k+1)\pi}{2n}\right)\sin\alpha - \cos\alpha \).
\fincorrection

\finexercice
