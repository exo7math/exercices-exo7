\exercice{3007, quercia, 2010/03/08}

\enonce[Idéaux premiers]
Un idéal \( I \) d'un anneau \( A \) est dit premier si : \( \forall\ x,y\in A,\ xy\in I \Rightarrow x\in I \) ou \( y\in I \).
\begin{enumerate}
  \item Quels sont les idéaux premiers de \( \Zz \)?

  \item Montrer que si \( A \) est commutatif non nul et si tous les idéaux de \( A \) sont premiers alors \( A \) est un corps.
\end{enumerate}
\finenonce

\noindication

\correction
\begin{enumerate}
  \item

  \item \( A \) est intègre car \( \{0\} \) est premier et si \( a\in A\setminus\{0\} \) alors \( a\times a\in(a^2) \) qui est premier donc \( a^2 \) divise \( a \) d'où \( a \) est inversible.
\end{enumerate}
\fincorrection

\finexercice
