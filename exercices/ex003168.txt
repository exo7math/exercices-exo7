\exercice{3168, quercia, 2010/03/08}

\enonce[Opérateur différence]
\label{opdiff}
On note \( U_p = \frac{X (X-1) \cdots (X-p+1)}{p!},\quad p \in \Nn \), et \( \Delta : { K[X]} \to { K[X]}, P \mapsto {P(X+1) - P(X)} \)
\begin{enumerate}
  \item Démontrer que la famille \( (U_p)_{p \in \Nn} \) est une base de \( { K[X]} \).

  \item Calculer \( \Delta^n(U_p) \).

  \item En déduire que : \( \forall\ P \in { K_n[X]} \), on a \( P = P(0) + (\Delta P)(0)U_1 + (\Delta^2 P)(0)U_2 + \dots + (\Delta^n P)(0)U_n \).

  \item Soit \( P \in { K[X]} \). Démontrer que :
  \[
    \bigl( \forall\ n \in \Zz, \text{ on a } P(n) \in \Zz \bigr) \Leftrightarrow \bigl(  \text{ les coordonnées de } P \text{ dans la base } (U_p) \text{ sont entières }  \bigr).
  \]

  \item Soit \( f : \Zz \to \Zz \) une fonction quelconque. Démontrer que \( f \) est polynomiale si et seulement si : \( \exists\ n \in \Nn \) tel que \( \Delta^n(f) = 0 \).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
