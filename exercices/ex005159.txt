\exercice{5159, rouget, 2010/06/30}

\enonce[**]
Montrer que \( \forall n\in\Nn^*,\;\forall x\in\Rr,\;E(\frac{E(nx)}{n}) = E(x) \).
\finenonce

\noindication

\correction
Soient \( n\in\Nn^* \) et \( x\in\Rr \).

\begin{align*}
  E(x)\le x<E(x)+1 & \Rightarrow nE(x)\le nx<nE(x)+n\Rightarrow nE(x)\le E(nx)<nE(x)+n
  \Rightarrow E(x)\le\frac{E(nx)}{n}<E(x)+1                                            \\
                   & \Rightarrow E(\frac{E(nx)}{n})=E(x).
\end{align*}
\fincorrection

\finexercice
