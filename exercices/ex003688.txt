\exercice{3688, quercia, 2010/03/11}

\enonce[Forme linéaire sur les polynômes]
On munit \( \Rr_2[X] \) du produit scalaire : \( (P\mid Q) = \int_{t = 0}^1 PQ(t)\,d t \).

\begin{enumerate}
  \item Vérifier que c'est effectivement un produit scalaire.

  \item Soit \( \varphi : {\Rr_2[X]} \to \Rr, P \mapsto {P(0).} \) Trouver le polynôme \( A \) tel que : \( \forall\ P \in \Rr_2[X],\
        \varphi(P) = (A\mid P) \).

\end{enumerate}
\finenonce

\noindication

\correction

  \item

  \item \( 30X^2-36X+9 \).
\end{enumerate}
\fincorrection

\finexercice
