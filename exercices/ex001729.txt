\exercice{1729, ridde, 1999/11/01}

\enonce
Soient \( n \in \Nn^* \) et \( x_1, \ldots, x_n \in ]0, + \infty[ \).
\begin{enumerate}
  \item En utilisant la concavité du \( \log \), montrer que \( (x_1 \ldots x_n)^{\frac 1n} \le \frac{x_1 + \ldots + x_n}n \).

  \item Montrer que \( (x_1 \ldots x_n)^{\frac 1n} \ge \frac 1{\frac 1{x_1} + \ldots +
            \frac 1{x_n}} \).

  \item En déduire que \( n! \le (\frac{n + 1}2)^n \).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
