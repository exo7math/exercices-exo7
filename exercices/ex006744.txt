\exercice{6744, queffelec, 2011/10/16}

\enonce[Intégrales de fractions rationnelles sans pôle sur \( \Rr \)]
\begin{enumerate}
  \item Calculer \( \displaystyle\int_\Rr\frac{dx}{1+x^{2n}} \), \( n\ge2 \), en intégrant \( \displaystyle\frac{1}{1+z^{2n}} \) sur un demi-cercle.

  \item Calculer \( \displaystyle\int_\Rr\frac{dx}{(1+x^2)^n} \).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
