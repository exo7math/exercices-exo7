\exercice{7734, mourougane, 2021/08/11}

\enonce
\item Soit \( E \) un espace muni d'une forme sesquilinéaire non dégénérée réflexive. Soit \( V \) un sous-espace de \( E \). Supposons que le radical \( \text{rad}(V) = V\cap V^\perp \) de \( V \) est de dimension \( 2 \), \( \text{rad}(V) = \langle N_1,N_2\rangle \). Soit \( W \) un supplémentaire de \( \text{rad}(V) \) dans \( V \). Calculer en détail le radical de \( V' = \langle N_1\rangle\oplus W \).
\finenonce

\noindication

\nocorrection

\finexercice
