\exercice{5979, tumpach, 2010/11/11}

\enonce
\textbf{Définition.}
Soit \( h\in \Rr^n \). On définit l'opérateur de translation par \( h \), noté \( \tau_h \), agissant sur une fonction \( f :\Rr^n \rightarrow \Rr \) par \( \tau_hf(x) := f(x-h), \quad \forall x\in\Rr^n \).

\bigskip

\textbf{Théorème.}
%\label{th:t1}
Si \( f\in L^p(\Rr^n) \) avec \( 1\le p < +\infty \), alors \( \lim_{h\rightarrow 0}\|\tau_hf - f\|_p = 0 \), i.e. \( \tau_hf \) tend vers \( f \) dans \( L^p(\Rr^n) \) lorsque \( h \) tend vers \( 0 \).

\bigskip

Le but de cet exercice est de démontrer ce théorème. Soit \( 1 \le p < +\infty \).
\begin{enumerate}
  \item Montrer que si \( f \) est continue à support compact dans la boule \( \mathcal{B}(0, M) \) centrée en \( 0 \) et de rayon \( M \), et si \( |h|\le 1 \), alors
        \[
          |f(x - h) - f(x)|^p \le \mathbf{1}_{\mathcal{B}(0, M+1)}
          2^p \|f\|_{\infty}^p.
        \]
        où \( \mathcal{B}(0, M+1) \) est la boule centrée en \( 0 \) de rayon \( M+1 \).

  \item En déduire que pour \( f \) continue à support compact, on a
        \[
          \lim_{h\rightarrow 0}\|\tau_hf - f\|_p = 0.
        \]

  \item Démontrer le théorème pour une fonction quelconque dans \( L^p(\Rr^n) \), \( 1\le p <+\infty \).

  \item Que se passe-t-il pour \( p = \infty \) ?
\end{enumerate}
\finenonce

\noindication

\correction
Soit \( 1 \le p < +\infty \).
\begin{enumerate}
  \item Si \( f \) est continue à support compact dans la boule \( \mathcal{B}(0, M) \) centrée en \( 0 \) et de rayon \( M \), et si \( |h|\le 1 \), alors
        \[
          \begin{array}{lcl}
            |f(x - h) - f(x)|^p & \le & \left(|f(x - h)| + |f(x)|\right)^p
            \le \left(2\|f\|_{\infty}\mathbf{1}_{\mathcal{B}(0, M+1)}\right)^p = \mathbf{1}_{B(0, M+1)}
            2^p \|f\|_{\infty}^p.
          \end{array}
        \]
        où \( \mathcal{B}(0, M+1) \) est la boule centrée en \( 0 \) de rayon \( M+1 \).

  \item Pour \( f \) continue, on a \( \lim_{h \rightarrow 0} |f(x- h) - f(x)| = 0 \). Puisque la fonction \( g(x) = 2^p
        \|f\|_{\infty}^p\,\mathbf{1}_{B(0, M+1)}(x) \) appartient à \( L^1(\Rr^n) \), le théorème de convergence dominée permet d'intervertir limite et intégrale, et il vient :
        \[
          \lim_{h \rightarrow 0}\|\tau_hf - f\|_p^p = \lim_{h
            \rightarrow 0}\int_{\Rr^n} |f(x - h) - f(x)|^p\,dx =
          \int_{\Rr^n} \lim_{h \rightarrow 0} |f(x - h) - f(x)|^p\,dx = 0.
        \]

  \item Soit \( f \) une fonction quelconque dans \( L^p(\Rr^n) \), \( 1\le p <+\infty \). Par densité des fonctions continues à support compact dans \( L^p(\Rr^n) \), pour tout \( \varepsilon > 0 \), il existe \( f_{\varepsilon} \) continue à support compact telle que \( \|f - f_{\varepsilon}\|_p \le
        \frac{\varepsilon}{3} \). Ainsi :
        \[
          \begin{array}{lcl}
            \|\tau_hf - f\|_p & =   & \|\tau_h(f - f_{\varepsilon}) - (f - f_{\varepsilon}) + \tau_hf_{\varepsilon} - f_{\varepsilon}\|_p           \\
                              & \le & \|\tau_h(f - f_{\varepsilon})\|_p + \|f - f_{\varepsilon}\|_p + \|\tau_hf_{\varepsilon} - f_{\varepsilon}\|_p \\
                              & =   & 2 \|f - f_{\varepsilon}\|_p + \|\tau_hf_{\varepsilon} - f_{\varepsilon}\|_p                                   \\
                              & =   & \frac{2}{3}\varepsilon + \|\tau_hf_{\varepsilon} - f_{\varepsilon}\|_p.
          \end{array}
        \]
        Puisque \( f_{\varepsilon} \) est continue à support compact, d'après la question précédente, il existe \( \delta>0 \) tel que pour \( |h|<\delta \), \( \|\tau_hf_{\varepsilon} - f_{\varepsilon}\|_p < \frac{\varepsilon}{3} \). Ainsi, pour \( |h| <
        \delta \), on a \( \|\tau_hf - f\|_p< \varepsilon \). En d'autre termes \( \lim_{h \rightarrow 0}\|\tau_hf - f\|_p = 0 \).

  \item Pour \( p = \infty \), les fonctions continues à support compact ne sont pas denses dans \( L^{\infty}(\Rr^n) \) ce qui fait que la démonstration précédente ne peut pas s'appliquer dans ce cas. De plus, on vérifie que, pour \( f = \mathbf{1}_{B(0, 1)} \) et \( h \neq 0 \), on a
        \[
          \|\tau_hf - f\|_{\infty} = 1.
        \]
        Alors que pour \( h = 0 \), on a \( \|\tau_hf - f\|_{\infty} = 0 \). On peut également vérifier que \( \lim_{h\rightarrow 0}\|\tau_hf - f \|_{\infty} = 0 \) si et seulement si la fonction \( f \) possède un représentant uniformément continu.
\end{enumerate}
\fincorrection

\finexercice
