\exercice{828, bodin, 1998/09/01}

\enonce
Calculer les primitives suivantes :
\[
  \int \frac{\cos^3 x}{\sin^5 x}dx \quad ; \quad
  \int \frac{\sin^3 x}{1+\cos x}dx \quad ; \quad
  \int \frac{dx}{\cos^4 x+\sin^4 x} \quad ; \quad
  \int \frac{\cos x}{1+\sin 2x}dx \quad ;
\]
\[
  \int \frac{\tan x -\tan a}{\tan x + \tan a}dx \quad ; \quad
  \int \frac{\sh x \ch x}{\sh^4 x+\ch^4 x}dx.
\]
\finenonce

\noindication

\nocorrection

\finexercice
