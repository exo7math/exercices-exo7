\exercice{2925, quercia, 2010/03/08}

\enonce[Équations affines]
\begin{enumerate}
  \item Montrer que toute droite du plan admet pour équation complexe : \( az + \overline{az} = b \) avec \( a \in \Cc^*, b \in \Rr \).

  \item Soient \( a,b,c \in \Cc \), \( a,b \) non tous deux nuls. Discuter la nature de \( E = \{\, z \in \Cc \mid az + b\overline{z} = c \,\} \).
\end{enumerate}
\finenonce

\noindication

\correction
\begin{enumerate}
  \item

  \item si \( |a| \ne |b| \) : une solution unique, \\
        si \( |a| = |b| \) : une droite ou \( \varnothing \).
\end{enumerate}
\fincorrection

\finexercice
