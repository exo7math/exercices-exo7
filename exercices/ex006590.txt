\exercice{6590, hueb, 2011/10/16}

\enonce
Évaluer
\begin{enumerate}
  \item
        \[
          \int_{\gamma}\frac{\sin \pi z^2 +\cos \pi z^2}{(z-1)(z-2)} dz,
        \]

  \item
        \[
          \int_{\gamma} \frac{e^{2z}}{(z+1)^4} dz,
        \]
\end{enumerate}
où \( \gamma \) est le cercle positif \( \{z; |z| = 3\} \).
\finenonce

\noindication

\nocorrection

\finexercice
