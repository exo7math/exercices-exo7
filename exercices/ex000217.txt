\exercice{217, ridde, 1999/11/01}
\video{6f56RN5_li0}

\enonce
Soit \( (E, \le) \) un ensemble ordonné. On définit sur \( \mathcal{P} (E)\setminus\left\{ \emptyset \right\} \) la relation \( \prec \) par
\[
  X \prec Y \quad \text{ ssi } \quad (X = Y \ \text{ ou } \
  \forall x \in X \ \forall y \in Y \ x \le y).
\]
Vérifier que c'est une relation d'ordre.
\finenonce

\noindication

\correction
\begin{itemize}
  \item Réflexivité : pour tout \( X\in\mathcal{P} (E)\setminus\{\emptyset\} \) on a \( X \prec X \) car \( X = X \).

  \item Anti-symétrie : pour \( X,Y\in\mathcal{P} (E) \) tels que \( X \prec Y \) et \( Y \prec X \), alors par définition de \( \prec \) on a
        \[
          \forall x \in X \quad \forall y \in Y \quad x\le y \text{ et } y \le x.
        \]
        Comme la relation \( \le \) est une relation d'ordre alors \( x\le y \) et \( y \le x \) implique \( x = y \). Donc
        \[
          \forall x \in X \quad \forall y \in Y \quad x = y,
        \]
        ce qui implique que \( X = Y \) (dans ce cas en fait \( X \) est vide ou un singleton).

  \item Transitivité : soit \( X,Y, Z \in\mathcal{P} (E) \) tels que \( X \prec Y \) et \( Y \prec Z \). Si \( X = Y \) ou \( Y = Z \) alors il est clair que \( X \prec Z \). Supposons que \( X\neq Y \) et \( Y\neq Z \) alors
        \[
          \forall x \in X \quad \forall y \in Y \quad x\le y \qquad \text{ et } \qquad \forall y \in Y \quad \forall z \in Z \quad y\le z.
        \]
        Donc on a
        \[
          \forall x \in X \quad \forall y \in Y \quad \forall z \in Z \quad x \le y \text{ et } y \le z,
        \]
        alors par transitivité de la relation \( \le \) on obtient :
        \[
          \forall x \in X \quad \forall z \in Z \quad x \le z.
        \]
        Donc \( X \prec Z \).
\end{itemize}
\fincorrection

\finexercice
