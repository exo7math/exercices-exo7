\exercice{3827, quercia, 2010/03/11}

\enonce[Calcul de minimums]
Calculer le minimum sur \( \Rr^2 \) de \( f : {\Rr^2} \to \Rr, {(a,b)} \mapsto { \int_{x = 0}^\pi (\sin x-ax^2-bx)^2\,d x.} \)
\finenonce

\noindication

\correction
\( a = \frac{20}{\pi^3}- \frac{320}{\pi^5} \), \( b = \frac{240}{\pi^4}-\frac{12}{\pi^2} \), \( m = \frac\pi2-\frac8\pi+\frac{160}{\pi^3}-\frac{1280}{\pi^5} \).
\fincorrection

\finexercice
