\exercice{1322, legall, 1998/09/01}

\enonce
\begin{enumerate}
  \item L'ensemble des matrices \( \begin{pmatrix}
          a & c \\
          b & d
        \end{pmatrix} \) avec \( a,b,c,d\in \Rr \) tels que \( ad-bc\not= 0 \) et \( a^2-b^2-c^2-d^2 \le 1 \) est-il un sous-groupe de \( Gl_2(\Rr) \) ?

  \item L'ensemble des matrices \( \begin{pmatrix}
          a & b      \\
          0 & a^{-1}
        \end{pmatrix} \) avec \( a\in \Rr^* \) et \( b\in \Rr \) est-il un sous-groupe de \( Gl_2(\Rr) \) ?

  \item Existe-t-il une valeur \( M\in \Rr \) telle que l'ensemble des matrices \( \begin{pmatrix}
          a & c \\
          b & d
        \end{pmatrix} \) avec \( a,b,c,d\in \Rr \) tels que \( ad-bc\not= 0 \) et \( a\le M \) forme un sous-groupe de \( Gl_2(\Rr) \) ?
\end{enumerate}
\finenonce

\noindication

\correction
\begin{enumerate}
  \item L'ensemble \( G \) des matrices \( \begin{pmatrix}
          a & c \\ b & d
        \end{pmatrix} \) avec \( a,b,c,d\in \Rr \) tels que \( ad-bc\not= 0 \) et \( a^2-b^2-c^2-d^2 \le 1 \) n'est pas un sous-groupe de \( Gl_2(\Rr) \). En effet les deux matrices \( \begin{pmatrix} 1 & 1 \\ 0 & 1/2
        \end{pmatrix} \) et \( \begin{pmatrix} 1 & 0 \\ 1 & 1/2
        \end{pmatrix} \) appartiennent à \( G \) et leur produit \( \begin{pmatrix} 2 & 1/2
                \\ 1/2 & 1/4
        \end{pmatrix} \) n'appartient pas à \( G \).

  \item L'ensemble \( H \) des matrices \( \begin{pmatrix}
          a & b \\ 0 & a^{-1}
        \end{pmatrix} \) avec \( a\in \Rr^* \) et \( b\in \Rr \) est un sous groupe de \( Gl_2(\Rr) \). En effet,

        - \( I_2 \) élément neutre de \( Gl_2(\Rr) \) appartient à \( H \).

        - Soient \( M = \begin{pmatrix} a & b \\ 0 & a^{-1}
        \end{pmatrix} \) et \( M' = \begin{pmatrix} c & d \\ 0 & c^{-1}
        \end{pmatrix} \) deux éléments de \( H \) alors \( MM' = \begin{pmatrix} ac & ad+bc^{-1} \\ 0 & (ac)^{-1}
        \end{pmatrix} \) donc le produit de deux éléments de \( H \) appartient à \( H \).

        - Soit \( M = \begin{pmatrix} a & b \\ 0 & a^{-1}
        \end{pmatrix} \). Alors \( M^{-1} = \begin{pmatrix} a^{-1} & - b \\ 0 & a
        \end{pmatrix} \) appartient à \( H \).

  \item Soit \( K_M \) l'ensemble des matrices \( \begin{pmatrix}
          a & c \\ b & d
        \end{pmatrix} \) avec \( a,b,c,d\in \Rr \) tels que \( ad-bc\not= 0 \) et \( a\le M \). Nous allons montrer, en raisonnant par l'absurde, qu'il n'existe pas de valeur \( M\in \Rr \) telle que \( K_M \) forme un sous-groupe de \( Gl_2(\Rr) \).

        Soit \( M\in \Rr \) telle que \( K_M \) forme un sous-groupe de \( Gl_2(\Rr) \). Alors \( I_2 \) appartient à \( K_M \) donc \( M\ge 1 \). Ainsi, les matrices \( A = \begin{pmatrix} 1 & 1 \\ 0 & 1 \end{pmatrix} \) et, pour tout \( n\in \Nn \), \( A_n = \begin{pmatrix} 1 & 1 \\ n & 1 \end{pmatrix} \) appartiennent à \( K_M \) donc le produit \( AA_n = \begin{pmatrix} 1+n & 0 \\ 0 & 1 \end{pmatrix} \) appartient à \( K_M \). En conséquence, pour tout \( n\in \Nn \), on a : \( 1+n\le M \), ce qui est absurde.
\end{enumerate}
\fincorrection

\finexercice
