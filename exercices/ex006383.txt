\exercice{6383, potyag, 2011/10/16}

\enonce
\begin{enumerate}
  \item Montrer que la quantité suivante (appelée forme de Frobenius) est un produit scalaire sur le groupe matriciel \( M_n(\Rr) \).
        \[
          <X,Y>  = {\rm tr}\ (X^TY).\quad X, Y\in M_n(\Rr)
        \]

  \item Montrer que la forme de Killing reste invariante par rapport à l'action de \( O(n) \) par conjugaison :

        \[
          <gXg^{-1}, gYg^{-1}>\ = \ <X, Y>,\ \ X,Y\in M_n(\Rr),\ g\in O(n).
        \]
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
