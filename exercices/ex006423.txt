\exercice{6423, potyag, 2011/10/16}

\enonce
\begin{enumerate}
  \item Montrer que toute homographie \( \gamma\in PGL_n\Cc \) possède au moins un point fixe

  \item Montrer que toute homographie \( \gamma:\Rr\mathbb{P}^{2n}\to \Rr\mathbb{P}^{2n} \) a toujours au moins un point fixe.

  \item Soit \( \gamma\in PGL(L) \) tel que \( {\rm card}({\rm fix}(\gamma)) < \infty \) et \( {\rm dim} P(L) = n \) alors montrer que \( {\rm card}({\rm fix}(\gamma))\le n+1 \) où \( {\rm fix}(\gamma) = \{x\ \vert \gamma(x) = x\} \).

\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
