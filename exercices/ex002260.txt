\exercice{2260, barraud, 2008/04/24}

\enonce[Idéaux étrangers]
\begin{enumerate}
  \item Montrer que \( I\cdot J\subset I\cap J \) et \( (I+J)\cdot (I\cap J)\subset I\cdot J \)

  \item On dit que deux idéaux \( I \) et \( J \) de \( A \) sont {\it étrangers}
        si \( I+J = A \). Montrer que \( I\cap J = I\cdot J \) si \( I \), \( J \) sont étrangers.
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
