\exercice{6463, drutu, 2011/10/16}

\enonce
\begin{enumerate}
  \item Montrer que \( S_3 \) contient un sous-groupe \( H \) distingué d'ordre \( 3 \) et que \( S_3/H \) est isomorphe à \( \Zz /2\Zz \).

  \item Montrer que les seuls groupes d'ordre \( 6 \) sont, à isomorphisme près, le groupe cyclique et \( S_3 \).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
