\exercice{5962, tumpach, 2010/11/11}

\enonce
Soient \( f, g\in L^1(\mu) \) où \( \mu \) est la mesure de Lebesgue sur \( \Rr^n \). On note \( \hat{f} \) la transformée de Fourier définie par
\[
  \hat{f}(y) = \int_{\Rr^n} f(x)\,e^{-2\pi i (y, x)} \,dx,
\]
où \( (\cdot, \cdot) \) désigne le produit scalaire de \( \Rr^n \). Montrer que
\begin{enumerate}
  \item \( \quad \int_{\Rr^n} f(x)\hat{g}(x)\,dx = \int_{\Rr^n} \hat{f}(x) g(x)\,dx \).

  \item \( \quad
        \widehat{f*g} = \hat{f}\,\hat{g} \).
\end{enumerate}
\finenonce

\noindication

\correction
Soient \( f, g\in L^1(\Rr^n) \). On note \( \hat{f} \) la transformée de Fourier définie par
\[
  \hat{f}(y) = \int_{\Rr^n} f(x)\,e^{-2\pi i (y, x)} \,dx,
\]
où \( (\cdot, \cdot) \) désigne le produit scalaire de \( \Rr^n \).
\begin{enumerate}
  \item On a \( \|\hat{g}\|_{\infty} \le \|g\|_1 \), ce qui implique que \( f\, \hat{g} \) est intégrable. De même \( \hat{f}\,g \) est intégrable. De plus \( F(x, y) = f(x) g(y) e^{-2\pi i (x, y)} \) appartient à \( L^1(\Rr^n\times\Rr^n) \). D'après le théorème de Fubini, \begin{eqnarray*} \int_{\Rr^n}
          f(x)\hat{g}(x)\,dx &=& \int_{\Rr^n}\,dx\, f(x)
          \int_{\Rr^n} g(y) e^{-2\pi i (x, y)} \,dy \\ &=&
          \int_{\Rr^n}\,dy \,g(y) \int_{\Rr^n} f(x)e^{-2\pi i (x, y)} \,dx = \int_{\Rr^n} \hat{f}(y) g(y)\,dx.
        \end{eqnarray*}

  \item On a
        \begin{eqnarray*}
          \widehat{f*g}(x) &=& \int_{\Rr^n} f*g(y)\,e^{-2\pi i (x, y)}\,dy = \int_{\Rr^n}\,dy \,e^{-2\pi i (x, y)}\int_{\Rr^n}f(y-z)\,g(z) \,dz\\
          &=& \int_{\Rr^n} \,dy \int_{\Rr^n}\,e^{-2\pi i (x, y-z)}e^{-2\pi i (x, z)}f(y-z)\,g(z)\,dz \\
          & = & \int_{\Rr^n} \,e^{-2\pi i (x, u)} f(u)\,du
          \int_{\Rr^n}e^{-2\pi i (x, z)}\,g(z) \,dz\\
          & = & \hat{f}(x) \,\hat{g}(x).
        \end{eqnarray*}
\end{enumerate}
\fincorrection

\finexercice
