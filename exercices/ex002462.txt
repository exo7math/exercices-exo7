\exercice{2462, matexo1, 2002/02/01}

\enonce
{\bf Systèmes linéaires.}

\begin{enumerate}
  \item Résoudre le système d'équations linéaires sur \( \Rr \)
        \[
          \left\{
          \begin{array}{rcrcrcc}
            2x & + & y  & + & z  & = & 3  \\
            x  & - & y  & + & 3z & = & 8  \\
            x  & + & 2y & - & z  & = & -3
          \end{array}\right.
        \]

  \item Le système suivant admet-il des solutions sur \( \Rr \)\,:
        \[
          \left\{
          \begin{array}{rcrcrcc}
            2x & + & y  & + & z  & = & 3  \\
            x  & - & y  & + & 3z & = & 8  \\
            x  & + & 2y & - & z  & = & -3 \\
            x  & + & y  & + & 2z & = & -1
          \end{array}\right.
        \]

  \item On considère le système précédent, mais dont les coefficients et les inconnues sont dans le corps \( \Zz/5\Zz \). Résoudre ce système.
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
