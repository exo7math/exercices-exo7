\exercice{4004, quercia, 2010/03/11}

\enonce[Déterminant]
Soit \( f:\Rr \to \Rr \) trois fois dérivable en \( a \). Étudier \( \lim_{h\to0}\frac1{h^4}
\begin{vmatrix} 1 & f(a)    & f(a+h)  \\
                1 & f(a+h)  & f(a+2h) \\
                1 & f(a+2h) & f(a+3h)
\end{vmatrix} \).
\finenonce

\noindication

\correction
\( f'(a)f'''(a) - f''(a)^2 \).
\fincorrection

\finexercice
