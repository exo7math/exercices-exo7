\exercice{50, bodin, 1998/09/01}

\enonce
\begin{enumerate}
  \item Calculer les racines \( n \)-ièmes de \( -i \) et de \( 1+i \).

  \item Résoudre \( z^2-z+1-i = 0 \).

  \item En déduire les racines de \( z^{2n}-z^n+1-i = 0 \).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
