\exercice{6634, queffelec, 2011/10/16}

\enonce
\begin{enumerate}
  \item Montrer que \( L = \limsup u_n\in\Rr \) est caractérisée par la condition suivante :

        Pour tout \( \varepsilon>0 \), tous les \( u_n \) sauf un nombre fini d'entre eux sont \( \le L+\varepsilon \), et une infinité d'entre eux est \( \ge L-\varepsilon \).

  \item Écrire l'analogue pour la \( \liminf \).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
