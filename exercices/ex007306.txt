\exercice{7306, mourougane, 2021/08/10}

\enonce
\begin{enumerate}
  \item Soit \( m \) et \( n \) deux entiers naturels non nuls et premiers entre eux. On a vu comment trouver une relation de Bézout \( um + vn = 1 \). Montrer alors que l'application
        \[
          \begin{array}{ccc}
            (\Zz/m\Zz) \times (\Zz/n\Zz) & \to     & \Zz/(mn)\Zz \\
            (x , y)                      & \mapsto & umy + vnx
          \end{array}
        \]
        est bien définie et est un isomorphisme. Déterminer son isomorphisme réciproque.

  \item Trouver l’entier entre \( 0 \) et \( 100 \) congru à \( 9 \) modulo \( 11 \) et à \( 3 \) modulo \( 13 \).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
