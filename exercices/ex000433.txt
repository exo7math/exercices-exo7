\exercice{433, gourio, 2001/09/01}

\enonce
Soient \( W_n = (X^2-1)^n, \ L_n = \frac{1}{2^nn!}W_n^{(n)} \).
\begin{enumerate}
  \item Donner le degré de \( L_n \), son coefficient dominant, sa parité, calculer \( L_n(1) \). Donner \( L_0,L_1,L_2 \).

  \item Démontrer : \( \forall n\ge 1,(X^2-1)W_n^{\prime} = 2nXW_n \), en déduire :
        \[
          \forall n\in \Nn,(X^2-1)L_n^{\prime\prime}+ 2XL_n^{\prime}-n(n+1)L_n = 0.
        \]

  \item Montrer ensuite : \( \forall n\ge 1,L_n^{\prime } = XL_{n-1}^{\prime}+nL_{n-1} \), puis \( nL_n = XL_n^{\prime }-L_{n-1}^{\prime } \).

  \item Montrer enfin que les polynômes \( L_n \) peuvent être définis par la récurrence :
        \[
          (n+1)L_{n+1} = (2n+1)XL_n-nL_{n-1}.
        \]
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
