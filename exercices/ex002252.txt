\exercice{2252, barraud, 2008/04/24}

\enonce
\begin{enumerate}
  \item Si \( x\cdot y \) est inversible dans un anneau \( A \), alors \( x \) et \( y \) sont inversibles.

  \item Dans un anneau, un élément inversible n'est pas diviseur de zéro et un diviseur de zéro n'est pas inversible.
\end{enumerate}
\finenonce

\noindication

\correction
\begin{enumerate}
  \item Si \( xy\in A^{\times} \), soit \( z\in A, (xy)z = 1 \). Alors \( x(yz) = 1 \) et \( (zx)y = 1 \) donc \( x \) et \( y \) sont inversibles.

  \item Soit \( x\in A^{\times} \) et \( y\in A \) tel que \( xy = 0 \). Alors \( x^{-1}xy = y = 0 \). Donc \( x \) n'est pas diviseur de \( 0 \).
\end{enumerate}
\fincorrection

\finexercice
