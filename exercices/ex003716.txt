\exercice{3716, quercia, 2010/03/11}

\enonce[Signature de \( {}^t\!AA \)]
Soit \( A \in \mathcal{M}_{n,p}(\Rr) \).

\begin{enumerate}
  \item Montrer que \( {}^t\!AA \) est la matrice d'une forme quadratique positive sur \( \Rr^p \).

  \item Déterminer sa signature en fonction de \( \mathrm{rg} A \).

\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
