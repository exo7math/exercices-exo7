\exercice{2966, quercia, 2010/03/08}

\enonce[Essai de tables]
Les opérations suivantes sont-elles des lois de groupe ?

\begin{enumerate}
  \item
        \[
          \begin{array}{c|ccc}
                     & a & b & c \\
            \hline a & a & a & a \\
            b        & a & b & b \\
            c        & a & b & c
          \end{array}
        \]

  \item
        \[
          \begin{array}{c|ccc}
                     & a & b & c \\
            \hline a & b & c & a \\
            b        & c & a & b \\
            c        & a & b & c
          \end{array}
        \]

  \item
        \[
          \begin{array}{c|cccc}
                     & a & b & c & d \\
            \hline a & a & b & c & d \\
            b        & b & a & d & c \\
            c        & d & c & b & a \\
            d        & c & d & a & b
          \end{array}
        \]
\end{enumerate}
\finenonce

\noindication

\correction
\begin{enumerate}
  \item Non, \( a \) n'est pas régulier.

  \item Oui, \( G \approx \Zz/3\Zz \).

  \item Non, il n'y a pas d'élément neutre.
\end{enumerate}
\fincorrection

\finexercice
