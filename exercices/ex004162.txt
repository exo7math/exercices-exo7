\exercice{4162, quercia, 2010/03/11}

\enonce[Changement de variables]
Soit \( f : {\Rr^3} \to {\Rr^3}, {(x,y,z)} \mapsto {(e^{2y}+e^{2z}, e^{2x}-e^{2z}, x-y).} \) Montrer que \( f \) induit un \( \mathcal{C}^1 \)-difféomorphisme de \( \Rr^3 \) sur un ouvert à préciser.
\finenonce

\noindication

\correction
\( f(\Rr^3) = \{ (u,v,w) \text{ tq } u+v > 0,\ ue^{2w}-v > 0\} \).
\fincorrection

\finexercice
