\exercice{288, cousquer, 2003/10/01}

\enonce
\begin{enumerate}
  \item Déterminer \( 2\Zz\cup 3\Zz \). Est-ce un sous-groupe de \( \Zz \)?

  \item Déterminer : \( 7\Zz\cup 49\Zz \); \( 5\Zz\cup 45\Zz \); \( \bigcup_{n = 1}^{28} 2^n \Zz \). Ces ensembles sont-ils des sous-groupes de \( \Zz \)?

  \item Trouver une condition nécessaire et suffisante pour qu'une réunion de deux sous-groupes de \( \Zz \) soit un sous-groupe de \( \Zz \).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
