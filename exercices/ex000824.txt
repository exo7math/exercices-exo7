\exercice{824, cousquer, 2003/10/01}

\enonce
Décomposer les fractions rationnelles suivantes ; en calculer les primitives.
\begin{enumerate}

  \item \( \displaystyle\frac{1}{a^2+x^2} \).

  \item \( \frac{1}{{(1+x^2)}^2} \).

  \item \( \displaystyle\frac{x^3}{x^2-4} \).

  \item \( \displaystyle\frac{4x}{{(x-2)}^2} \).

  \item \( \displaystyle\frac{1}{x^2+x+1} \).

  \item \( \displaystyle\frac{1}{{(t^2+2t-1)}^2} \).

  \item \( \displaystyle\frac{3t+1}{{(t^2-2t+10)}^2} \).

  \item \( \displaystyle\frac{3t+1}{{t^2-2t+10}} \).

  \item \( \displaystyle\frac{1}{{t^3+1}} \).

  \item \( \displaystyle\frac{x^3+2}{{(x+1)}^2} \).

  \item \( \displaystyle\frac{x+1}{{x{(x-2)}^2}} \).

  \item \( \displaystyle\frac{(x^2-1)(x^3+3)}{2x+2x^2} \).

  \item \( \displaystyle\frac{x^2}{{{(x^2+3)}^3 (x+1)}} \).

  \item \( \displaystyle\frac{x^7+x^3-4x-1}{x{(x^2+1)}^2} \).

  \item \( \displaystyle\frac{3x^4-9x^3+12x^2-11x+7}{(x-1)^3(x^2+1)} \).
\end{enumerate}
\finenonce

\noindication

\correction
Résultats valables sur chaque intervalle du domaine de définition.
\begin{enumerate}

  \item \( \frac{1}{x^2+a^2} \) est un élément simple. Primitives : \( \frac{1}{a} \arctan(\frac{x}{a})+k \).

  \item \( \frac{1}{{(1+x^2)}^2} \) est un élément simple. Primitives : \( \frac{1}{2} \arctan x + \frac{x}{2(1+x^2)}+k \).

  \item \( \frac{x^3}{x^2-4} = x+\frac{2}{x-2}+\frac{2}{x+2} \). Primitives : \( \frac{x^2}{2}+ \ln\big( (x^2-4)^2 \big) +k \).

  \item \( \frac{4x}{{(x-2)}^2} = \frac{4}{x-2}+\frac{8}{{(x-2)}^2} \). Primitives : \( 4 \ln\vert x-2\vert-\frac{8}{x-2}+k \).

  \item \( \frac{1}{x^2+x+1} \) est un élément simple. Primitives : \( \frac{2}{\sqrt3} \arctan\frac{(2x+1)}{\sqrt3}+k \).

  \item \( \frac{1}{{(t^2+2t-1)}^2} = \frac{1}{8{(t+1+ \sqrt2)}^2}+ \frac{\sqrt2}{16(t+1+\sqrt2)} + \frac{1}{8{(t+1- \sqrt2)}^2}+ \frac{- \sqrt2}{16(t+1- \sqrt2)} \).\newline Primitives : \( -\frac{t+1}{4(t^2+2t-1)} +\frac{\sqrt2}{16} \ln
        \left\vert\frac{t+1+ \sqrt2}{t+1- \sqrt2} \right\vert+ k \).

  \item \( \frac{3t+1}{{(t^2-2t+10)}^2} \) est un élément simple.\newline Primitives : \( -\frac{3}{2(t^2-2t+10)} +\frac{2(t-1)}{9(t^2-2t+10)}
        +\frac{2}{27} \arctan(\frac{t-1}{3}) +k \).

  \item \( \frac{3t+1}{t^2-2t+10} \) est un élément simple. Primitives : \( \frac{3}{2} \ln(t^2-2t+10) +\frac{4}{3} \arctan(\frac{t-1}{3})+k \).

  \item \( \frac{1}{t^3+1} = \frac{1}{3(t+1)}-\frac{t-2}{3(t^2-t+1)} \). Primitives : \( \frac{1}{3}\ln\vert t+1 \vert-\frac{1}{6} \ln( t^2-t+1) + \frac{1}{\sqrt3} \arctan(\frac{2t-1}{\sqrt3}) +k \).

  \item \( \frac{x^3+2}{{(x+1)}^2} = x-2+\frac{3}{x+1}+\frac{1}{{(x+1)}^2} \). Primitives : \( \frac{x^2}{2}-2x+3 \ln\vert x+1 \vert-\frac{1}{x+1} +k \).

  \item \( \frac{x+1}{x{(x-2)}^2} = \frac{1}{4x}- \frac{1}{4(x-2)}
        +\frac{3}{2{(x-2)}^2} \). Primitives : \( \frac{1}{4} \ln\vert x \vert-\frac{1}{4} \ln\vert x-2 \vert -\frac{3}{2(x-2)} +k \).

  \item \( \frac{(x^2-1)(x^3+3)}{2x+2x^2} = \frac{1}{2} (x^3-x^2+3) -\frac{3}{2x} \). Primitives : \( \frac{x^4}{8}-\frac{x^3}{6}+\frac{3x}{2}-\frac{3}{2}\ln\vert x
        \vert+k \).

  \item \( \frac{x^2}{{(x^2+3)}^3(x+1)} = \frac{1}{4^3(x+1)}+ \frac{1-x}{4^3 (x^2+3)} + \frac{1-x}{4^2 {(x^2+3)}^2}
        -\frac{3(1-x)}{4{(x^2+3)}^3} \).

        Primitives : \( -\frac{x+3}{4^2{(x^2+3)}^2}-\frac{2x-3}{3 \cdot 2^5 \cdot (x^2+3)}
        -\frac{1}{2^7}\ln(x^2+3) - \frac{1}{3 \cdot 2^6 \cdot \sqrt3} \arctan(\frac{x}{\sqrt3}) +\frac{1}{4^3}\ln\vert x+1 \vert+k \).

  \item \( \frac{x^7+x^3-4x-1}{x{(x^2+1)}^2} = x^2-2-\frac{1}{x} +\frac{x+4}{x^2+1}+ \frac{x-6}{{(x^2+1)}^2} \).\newline Primitives : \( \frac{x^3}{3}-2x -\ln\vert x \vert+\frac{1}{2}\ln (1+x^2) +\arctan x -\frac{6x+1}{2(x^2+1)} +k \).

  \item \( \frac{3x^4-9x^3+12x^2-11x+7}{(x-1)^3(x^2+1)} = \frac{1}{(x-1)^3}-\frac{2}{(x-1)^2}+\frac{3}{x-1}-\frac{1}{x^2+1} \).\newline Primitives : \( -\frac{1/2}{(x-1)^2}+\frac{2}{x-1}+3\ln\vert x-1\vert-\arctan x +k \).
\end{enumerate}
\fincorrection

\finexercice
