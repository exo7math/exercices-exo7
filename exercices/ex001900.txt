\exercice{1900, legall, 2003/10/01}

\enonce
\ \par
\begin{center}{\large \bf I Préliminaires}
\end{center}

\begin{enumerate}
  \item Soit \( \mathcal{P} \) l'espace vectoriel des fonctions polynomiales de \( [0,1] \) à valeurs dans \( \Rr \). Montrer que \( \mathcal{P} \) est de dimension infinie.

  \item Soit \( X \) une partie bornée de \( \Rr \). Montrer que \( \sup (X) = \sup \bar{X} \).
\end{enumerate}

\vskip5mm
\begin{center}{\large \bf II}
\end{center}
\vskip5mm

On note \( \mathcal{L} \) l'ensemble des fonctions {\em lipschitziennes} de \( [0,1] \) à valeurs dans \( \Rr \), c'est‑à‑dire telles qu'il existe \( k\in \Rr_+ \) tel que, pour tout \( x,y\in [0,1] \), \( \vert f(x)-f(y)\vert \le k\vert x-y\vert \). On note \( \Cc^1 \) l'ensemble des fonctions de \( [0,1] \) à valeurs dans \( \Rr \) de classe \( C^1 \), c'est‑à‑dire dérivables à dérivée continue.
\begin{enumerate}
  \item Montrer que \( \mathcal{L} \) est un sous‑espace vectoriel de l'espace vectoriel des fonctions de \( [0,1] \) à valeurs dans \( \Rr \), que \( \mathcal{L} \) contient \( \Cc^1 \) et est de dimension infinie.

  \item On pose, pour tout \( f\in \mathcal{L} \) :
        \[
          \displaystyle{ N_1(f) = \vert f(0)\vert +\sup_{(x,y)\in [0,1]^2, x\not= y}\dfrac {\vert f(x)-f(y)\vert }{\vert x-y\vert }}
        \]
        \[
          \displaystyle{ N_2(f) = \vert f(0)\vert +\sup_{x\in ]0,1]}\dfrac {\vert f(x)-f(0)\vert }{\vert x\vert }}
        \]
        \[
          \Vert f \Vert_{\infty} = \sup_{x\in [0,1]}\vert f(x)\vert
        \]
        \[
          \displaystyle{ \lambda (f) = \Vert f\Vert_{\infty} +\sup_{(x,y)\in [0,1]^2, x\not= y}\dfrac {\vert f(x)-f(y)\vert }{\vert x-y\vert }}.
        \]
        \begin{enumerate}
          \item Montrer que \( N_1 \), \( N_2 \), \( \Vert \cdot \Vert_{\infty} \) et \( \lambda \) sont des normes sur \( \mathcal{L} \).

          \item En considérant la suite \( f_n(x) = \sin(2\pi n x) \), montrer que \( N_2 \) n'est pas équivalente à \( \Vert \cdot \Vert_{\infty} \).

          \item Montrer que \( N_1 \) n'est équivalente ni à \( \Vert \cdot \Vert_{\infty} \), ni à \( N_2 \).

          \item Construire une suite \( (g_n)_{n\in \Nn } \) d'éléments de \( \mathcal{L} \) qui converge vers \( 0 \) pour \( \Vert \cdot \Vert_{\infty} \) mais pas pour \( N_2 \). En déduire (de nouveau) que \( N_2 \) n'est pas équivalente à \( \Vert \cdot \Vert_{\infty} \).

          \item Montrer que \( \lambda \) et \( N_1 \) sont équivalentes.
        \end{enumerate}

  \item On pose, pour tout \( f\in \Cc^1 \) : \( \displaystyle{ \nu_1(f) = \vert f(0)\vert +\Vert f'\Vert_{\infty}} \) et \( \displaystyle{ \nu (f) = \Vert f\Vert_{\infty} +\Vert f'\Vert_{\infty}} \).
        \begin{enumerate}
          \item Montrer que \( \nu_1 \) et \( \nu \) sont des normes sur \( \Cc^1 \).

          \item Montrer que \( \nu_1(f) = N_1(f) \), pour tout \( f\in \Cc^1 \).

          \item Les normes \( \nu \) et \( \nu_1 \) sont-elles équivalentes ?
        \end{enumerate}

  \item Soit \( (E, \Vert \cdot \Vert ) \) un espace vectoriel normé. Une suite \( (x_n)_{n\in \Nn } \) d'éléments de \( E \) est dite de {\em Cauchy} si, pour tout \( \epsilon >0 \), il existe \( N \) tel que, si \( m, n\ge N \) alors \( \Vert x_n-x_m\Vert \le \epsilon \). On dit que \( (E, \Vert \cdot \Vert ) \) est {\em complet} si toute suite de Cauchy y est convergente. On rappelle que \( \Rr \) muni de la norme \( x\mapsto \vert x \vert \) est complet.
        \begin{enumerate}
          \item Soit \( C^0 \) l'espace vectoriel des fonctions continues de \( [0,1] \) à valeurs dans \( \Rr \). Montrer que \( (C^0, \Vert \cdot \Vert_{\infty}) \) est complet.

          \item L'espace vectoriel normé \( (\Cc^1, \nu ) \) est-il complet ? Qu'en est-il de \( (\Cc^1, \nu_1) \)?

          \item Soit \( (f_n)_{n\in \Nn } \) une suite de Cauchy dans \( (\mathcal{L} ,
                \lambda ) \). Montrer que \( (f_n)_{n\in \Nn } \) converge uniformément vers une fonction continue \( f \).

          \item Démontrer que pour \( n \) assez grand \( f-f_n \) est lipschitzienne.

          \item En déduire que \( (\mathcal{L} , \lambda ) \) est complet.
        \end{enumerate}
\end{enumerate}

\vskip5mm
\begin{center}{\large \bf III}
\end{center}
\vskip5mm

On munit \( \Cc^1 \) d'une norme \( N \) et \( C^0 \) de la norme \( \Vert \cdot \Vert_{\infty } \). On note \( d \) l'application \( f\mapsto f' \) de \( \Cc^1 \) à valeurs dans \( \Cc^0 \).
\begin{enumerate}
  \item Soit \( \varphi : \Cc^1\rightarrow C^0 \) une application linéaire. On pose \( N(\varphi ) = \displaystyle{ \sup_{f ; N( f )\le 1}\Vert \varphi (f)\Vert_{\infty}} \). Démontrer que \( \varphi \) est continue si et seulement si \( N(\varphi ) \) est fini. Vérifier que \( N \) est une norme sur l'espace vectoriel des applications linéaires continues de \( \Cc^1 \) à valeurs dans \( \Cc^0 \).

  \item Montrer que l'application \( d \) n'est pas continue si \( N = \Vert \cdot \Vert_{\infty} \).

  \item On munit \( \Cc^1 \) de la norme \( \nu \). Montrer que \( d \) est continue et calculer \( N(d) \).
\end{enumerate}
\finenonce

\noindication

\correction
\ \par
\begin{center}{\large \bf I}
\end{center}
\vskip5mm

\begin{enumerate}
  \item Soit \( \mathcal{P} \) l'espace vectoriel des fonctions polynomiales. Supposons \( \mathcal{P} \) de dimension finie \( n \). Notons \( f_k \) la fonction \( x\mapsto x^k \). Alors la famille \( \{ f_0, \cdots , f_n \} \) qui compte \( n+1 \) éléments est liée, donc il existe \( a_0, \cdots , a_n \) des scalaires non tous nuls tels que, pour tout \( x\in \Rr \) on ait \( a_0+a_1x+\cdots a_n x^n = 0 \). Il en résulte que le polynôme non nul à coefficients réels \( a_0+a_1X+\cdots a_nX^n \) a une infinité de racines, ce qui est absurde.

  \item Posons \( M = \sup (\bar{X}) \). On doit vérifier que, {\em i)}
        pour tout \( x\in X , x\le M \) et {\em ii)} pour tout \( \epsilon >0 \) il existe \( x\in X \) tel que \( M-\epsilon \le x \). Comme \( X\subset \bar{X} \) et, pour tout \( x\in \bar{X} , x\le M \) la propriété {\em i)} est vérifiée par \( M \). Soit maintenant \( \epsilon >0 \). Il existe \( x\in \bar{X} \) tel que \( M-\dfrac{\epsilon }{2}<x \). Comme \( x\in\bar{X} \), il existe aussi \( y\in X \) tel que \( \vert x-y\vert <\dfrac{\epsilon }{2} \). Donc \( M-\epsilon< y \) et \( M \) satisfait à {\em ii)}.
        \vskip1mm {\em Remarque : } on note également que \( \sup (X)\in \bar{X} \). En effet, pour tout \( n\in \Nn \), choisissons un élément \( x_n\in X \) tel que \( x_n\ge \sup (X)-\dfrac{1}{n} \). Alors la suite \( (x_n)_{n\in \Nn} \) constituée d'éléments de \( X \) converge dans \( \Rr \) vers \( \sup (X) \) qui appartient donc à \( \bar{X} \). On peut bien sûr en déduire la propriété {\em ii)} de \( M \).

\end{enumerate}

\vskip5mm
\begin{center}{\large \bf II}
\end{center}
\vskip5mm

\begin{enumerate}
  \item Il est clair que \( \mathcal{L} \) est un sous-espace vectoriel de l'espace vectoriel des fonctions de \( [0,1] \) à valeurs dans \( \Rr \). Soit \( f\in \Cc^1 \) et \( x,y \in [0,1] \), avec \( x< y \). Par le théorème des accroissements finis, il existe \( c_x\in ]x,y[ \) tel que \( f(y)-f(x) = f'(c_x)(y-x) \). Or \( f' \) est continue, donc bornée sur \( [0,1] \). Soit \( \displaystyle{M = \sup_{t\in [0,1]}\vert f'(t)\vert .} \) On a l'inégalité \( \vert f(y)-f(x)\vert \le M\vert y-x\vert \) qui montre que \( f\in \mathcal{L} \). Il en résulte que \( \mathcal{L} \) contient \( \mathcal{P} \) donc est de dimension infinie.

  \item
        \begin{enumerate}
          \item Il suffit de vérifier que si \( N_1(f) = 0 \) et \( N_2(g) = 0 \), alors \( f = g = 0 \), les autres propriétés étant claires. Or si \( N_1(f) = 0 \), alors \( f \) est constante et \( f(0) = 0 \), donc \( f = 0 \). Il en va de même pour \( N_2 \).

          \item Pour tout \( n \in \Nn \), \( \Vert f_n \Vert_{\infty} = 1 \). Posons \( \displaystyle X_n = \{ \dfrac {\vert f_n(x)-f_n(0)\vert }{\vert x\vert } , x\not=0 \} \). Comme \( f_n(0) = 0 \), on voit que \( N_1(f_n) = \sup (X_n) \). Or \( \displaystyle \vert f'_n(0)\vert = \lim_{x\rightarrow 0} \dfrac {\vert f_n(x)-f_n(0)\vert }{\vert x\vert } \), appartient à \( \bar{X_n} \) donc, en appliquant I 2) on constate que \( \vert f'_n(0)\vert \le \sup (\bar {X_n}) = \sup (X_n) \). Enfin \( f'_n(0) = 2\pi n \) donc \( N_2(f_n)\ge 2\pi n \). Il n'existe donc pas \( K>0 \) tel que, pour tout \( n\in \Nn \), \( N_2(f_n)<K\Vert f_n \Vert_{\infty} \) soit \( N_2 \) et \( \Vert \ \Vert_{\infty} \) ne sont pas équivalentes.
                \vskip1mm {\em Remarques :} a) on peut obtenir ce résultat (et le préciser) en remarquant que la fonction \( f_n :x\mapsto \dfrac{\sin(2\pi n x)}{x} \) définie sur \( ]0, 1] \) se prolonge en une fonction continue en \( 0 \) en posant \( f_n(0) = 2\pi n \). Puis noter (en fait c'est un cas particulier de I 2)) que \( \displaystyle{ \sup_{]0,1]}\vert f_n\vert = \sup_{[0,1]}\vert f_n\vert } \) et montrer (par une étude classique de fonction) que cette dernière quantité est \( 2\pi n \).
                \vskip1mm b) Ce qui fait l'intérêt pour ce problème des fonctions \( (f_n)_{n\in \Nn } \), c'est qu'elles sont bornées par \( 1 \) mais que leur pente en l'origine peut être rendue arbitrairement grande avec \( n \). On peut donc obtenir le même résultat avec la suite \( (k_n)_{n\in \Nn } \) définie par \( k_n(x) = nx \) si \( x\le \dfrac{1}{n} \) et \( 1 \) sinon, pour laquelle un calcul direct donne \( N_1(k_n) = n \) et \( \Vert k_n\Vert_{\infty} = 1 \).

          \item Comme, pour tout \( f\in \mathcal{L} \), \( N_1(f)\ge N_2(f) \), on déduit de ce qui précéde que \( N_1 \) n'est pas équivalente ni à \( \Vert \cdot \Vert_{\infty} \). Posons \( g_n(x) = x^n \), pour \( n\ge 1 \). Pour tout \( n\ge 1 \), \( N_2(g_n) = 1 \). De plus \( g'_n(1) = n \), donc, par un raisonnement identique à celui qui précéde, \( N_1(g_n)\ge n \) ce qui montre que \( N_1 \) n'est pas équivalente à \( N_2 \).

                \vskip1mm {\em Remarque :} ce qui fait l'intérêt pour ce problème des fonctions \( (g_n)_{n\in \Nn } \), c'est qu'elles sont bornées par \( 1 \) mais que leur pente en \( 1 \) peut être rendue arbitrairement grande avec \( n \). On peut donc obtenir le même résultat avec la suite \( (l_n)_{n\in \Nn } \) définie par \( l_n(x) = 0 \) si \( x\le 1-\dfrac{1}{n} \) et \( nx- (n-1) \) sinon.

          \item On pose \( g_n(x) = x \) si \( x\le \dfrac{1}{n} \) et \( \dfrac{1}{n} \) sinon. Il est clair que \( g_n\in \mathcal{L} \), \( \Vert g_n \Vert_{\infty} = \dfrac{1}{n} \) et \( N_2(g_n) = 1 \). Il n'existe donc pas de constante \( K'\in \Rr_+^* \) telle que, pour tout \( n\in \Nn \), \( \Vert g_n \Vert_{\infty}\ge K'N_2(g_n) \) donc \( N_2 \) n'est pas équivalente à \( \Vert \cdot \Vert_{\infty} \). Enfin \( N_2(g_n) = N_1(g_n) \), ce qui établit le même résultat pour \( N_1 \).

          \item Il est clair que pour tout \( f\in \mathcal{L} \), \( \lambda (f)\ge N_1(f) \). Soit \( x\in ]0,1] \). De l'identité \( f(x) = f(0)+x\dfrac{f(x)-f(0)}{x} \) on déduit que \( \vert f(x)\vert
                \le \vert f(0)\vert +\vert \dfrac{f(x)-f(0)}{x}\vert \) (car \( \vert x\vert \le 1 \)) soit \( \vert f(x)\Vert \le N_1(f) \). L'application \( x\mapsto f(x) \) étant continue sur \( [0,1] \) (ou en appliquant I 2)) on en déduit que, pour tout \( x\in [0,1] \) on a également \( \vert f(x)\Vert \le N_1(f) \). En d'autre termes \( \Vert f \vert_{\infty}\le N_1(f) \) et \( \lambda (f)\le 2N_1(f) \). Les normes \( \lambda \) et \( N_1 \) sont donc équivalentes.
        \end{enumerate}

  \item On pose, pour tout \( f\in \Cc^1 : \) \( \displaystyle{ \nu_1(f) = \vert f(0)\vert +\Vert f'\Vert_{\infty}} \) et \( \displaystyle{ \nu (f) = \Vert f\Vert_{\infty} +\Vert f'\Vert_{\infty}} \).
        \begin{enumerate}
          \item On constate aisément que si \( \nu_1(f) = 0 \), alors \( f \) est constante et, comme de plus \( f(0) = 0 \), elle est nulle. Les autres propriétés sont immédiates, donc \( \nu \) et \( \nu_1 \) sont des normes sur \( \Cc^1 \).

          \item Soit \( f\in \Cc^1 \), notons \( X = \{ \dfrac {\vert f(x)-f(y)\vert }{\vert x-y\vert } ; (x,y)\in [0,1]^2, x\not= y \} \). Pour montrer que \( \nu_1 (f) = N_1(f) \), il suffit de vérifier que \( \sup(X) = \Vert f'\Vert_{\infty} \). Soient \( x, y\in [0,1], x\not= y \). Par le théorème des accroissements finis, il existe \( c \) compris entre \( x \) et \( y \) tel que \( \dfrac {\vert f(x)-f(y)\vert }{\vert x-y\vert } = f'(c)\le \Vert f'\Vert_{\infty} \), donc \( \sup (X)\le \Vert f'\Vert_{\infty} \). Comme \( f' \) est continue, il existe \( x_0\in [0, 1] \) tel que \( f'(x_0) = \Vert f'\Vert_{\infty} \). Alors \( \displaystyle{f'(x_0) = \lim_{x\rightarrow x_0}\dfrac{f(x)-f(x_0)}{x-x_0}} \) appartient à \( \bar{X} \), donc, en appliquant I 2), \( \Vert f'\Vert_{\infty}\le \sup (\bar{X}) = \sup (X) \).
                \vskip1mm {\em Remarque :} on peut formuler ce raisonnement de la manière suivante : soit \( Y = \{ f'(x) ; x\in [0,1] \} \). Par le théorème des accroissements finis, \( X\subset Y \). On a ensuite, par définition de la dérivée, \( Y\subset \bar{X} \). Donc \( \sup (X) \le \sup (Y)\le \sup (\bar{X}) \), puis on applique I 2).

          \item Les normes \( \nu \) et \( \nu_1 \) sont équivalentes. En effet, il est clair que \( \nu_1(f)\le \nu (f) \) pour tout \( f\in \Cc^1 \). Soit \( t_0\in [0,1] \) tel que \( \Vert f\Vert_{\infty} = \vert f(t_0)\vert \). Si \( t_0 = 0 \) alors \( \nu_1(f)\le \nu (f) \). Sinon, par le théorème des accroissements finis, il existe \( c\in ]0, t_0[ \) tel que \( f(t_0) = f(0)+ f'(c)t_0 \) ce dont on déduit que \( \Vert f\Vert_{\infty}\le \nu_1(f) \), puis que \( \nu ( f)\le 2\nu_1(f) \).
        \end{enumerate}

  \item \begin{enumerate}

          \item Soit \( x\in [0,1] \). La suite de nombres réels \( (f_n(x))_{n\in
                    \Nn} \) étant de Cauchy, elle est convergente. On pose \( \displaystyle{f(x) = \lim_{n\rightarrow \infty}f_n(x)} \). Soit \( \epsilon >0 \). La suite \( (f_n) \) étant de Cauchy, il existe \( N \) tel que, si \( m,n\ge N \) alors \( \Vert f_n-f_m \Vert_{\infty}\le
                \epsilon \). Soient \( x\in [0,1] \) et \( m,n\ge N \). On a \( \vert f_n(x)-f_m(x)\vert \le \epsilon \) et, ceci étant vrai pour tout \( m \in \Nn \), on en déduit, par passage à la limite suivant \( m \), que \( \vert f_n(x)-f(x)\vert
                \le \epsilon \), soit \( \Vert f_n-f\Vert_{\infty}\le \epsilon \). Ainsi \( f \) est la limite, pour la convergence uniforme, d'une suite de fonctions continues donc est continue.

          \item Par définition de \( \nu \), une suite \( (f_n)_{n\in \Nn} \) de Cauchy pour \( \nu \) est de Cauchy pour \( \Vert \ \Vert_{\infty} \), donc (uniformément) convergente par la question qui précède. De même \( (f'_n)_{n\in \Nn} \) est de Cauchy pour \( \Vert \cdot \Vert_{\infty} \), donc converge uniformément vers une fonction continue \( g \). Il en résulte que \( f \) est dérivable et a pour dérivée la fonction continue \( g \). Enfin \( (f_n)_{n\in \Nn} \) converge vers \( f \) pour \( \nu \) donc \( (\Cc^1,
                \nu ) \) est complet.

                Soit maintenant \( (g_n)_{n\in \Nn} \) une suite de Cauchy dans \( (\Cc^1,
                \nu_1) \). Comme \( \nu_1 \) est équivalente à \( \nu \), elle est de Cauchy pour \( \nu \) donc convergente. Il existe donc \( h\in \Cc^1 \) telle que \( \displaystyle{\lim_{n\rightarrow \infty}\nu (h-g_n ) = 0} \). Mais puisque \( \nu_1 \) est équivalente à \( \nu \), on a aussi \( \displaystyle{\lim_{n\rightarrow \infty}\nu_1(h-g_n ) = 0} \) donc \( (\Cc^1, \nu_1) \) est complet.

          \item Soit \( (f_n)_{n\in \Nn } \) une suite de Cauchy dans \( (\mathcal{L} ,
                \lambda ) \). Comme \( \lambda (f_n)\ge \Vert f_n \Vert_{\infty} \), la suite \( (f_n)_{n\in \Nn } \) est également de Cauchy dans \( (\Cc^0, \Vert \cdot \Vert_{\infty}) \). Comme \( (\Cc^0, \Vert \cdot \Vert_{\infty}) \) est complet, \( (f_n)_{n\in \Nn } \) converge uniformément vers une fonction continue qu'on notera \( f \).

          \item Soit \( \epsilon >0 \). Comme \( (f_n)_{n\in \Nn } \) est une suite de Cauchy, il existe \( N \) tel que, pour \( m,n\ge N \) on ait, pour tout \( x,y \) et \( z\in [0,1] \), avec \( x\not=y \):
                \[
                  \vert f_n(z)-f_m(z)\vert +\left \vert \dfrac{ (f_n(x)-f_m(x))-(f_n(y)-f_m(y))}{x-y}\right \vert <\epsilon .
                \]
                En faisant tendre \( m \) vers l'infini, on en déduit :
                \[
                  \vert f_n(z)-f(z)\vert +\left \vert \dfrac{ (f_n(x)-f(x))-(f_n(y)-f(y))}{x-y}\right \vert <\epsilon ,
                \]
                donc
                \[
                  \sup_{z\in [0,1]}\vert f_n(z)-f(z)\vert +\sup_{x,y\in [0,1], x\not=y}\left \vert \dfrac{ (f_n(x)-f(x))-(f_n(y)-f(y))}{x-y}\right
                  \vert \le \epsilon .
                \]
                Ainsi la suite \( (f_n)_{n\in \Nn } \) converge vers \( f \) pour la norme \( \lambda \). Par ailleurs, on déduit de la seconde inégalité que, pour tout \( x,y \in [0,1] \), avec \( x\not=y \): \( \vert (f_n-f)(x)-(f_n-f)(y)\vert <\epsilon \vert x-y\vert \), donc que pour \( n \) assez grand \( f-f_n \) appartient à \( \mathcal{L} \). Or \( \mathcal{L} \) est un espace vectoriel et \( f_n\in \mathcal{L} \) donc \( f \) appartient à \( \mathcal{L} \).

          \item Toute suite de Cauchy de \( \mathcal{L} \) admet une limite dans \( \mathcal{L} \) qui est donc complet.
        \end{enumerate}
\end{enumerate}
