\exercice{4881, quercia, 2010/03/17}

\enonce[Usage d'applications affines]
Soient \( A_1,\dots,A_n \) \( n \) points de \( {\cal E} \).

Étudier l'existence de points \( B_1,\dots,B_n \) tels que \( A_i = \text{mil}(B_i,B_{i+1}) \) (\( A_n = \text{mil}(B_n,B_1)) \).
\finenonce

\noindication

\nocorrection

\finexercice
