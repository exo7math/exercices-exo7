\exercice{3060, quercia, 2010/03/08}

\enonce[\( f(f(n)) < f(n+1) \)]
Soit \( f : \Nn \to \Nn \) telle que : \( \forall\ n \in \Nn,\ f(f(n)) < f(n+1) \). On veut montrer que \( f = \text{id}_\Nn \).
\par {\it (Olympiades 1977)}
\begin{enumerate}
  \item Montrer que \( \forall\ n\in \Nn,\ \forall\ x \ge n,\ f(x) \ge n \).

  \item Soit \( n \in \Nn \) et \( a \ge n \) tel que \( f(a) = \min\{f(x)\mid x\ge n\} \). Montrer que \( a = n \).

  \item En déduire que \( f \) est strictement croissante, puis conclure.
\end{enumerate}
\finenonce

\noindication

\correction
\begin{enumerate}
  \item Récurrence sur \( n \).

  \item Si \( a > n \), alors \( a = b+1 \) avec \( b \ge n \), donc \( f(b) \ge n \), donc \( f(f(b)) \ge f(a) \). Contradiction.

  \item
\end{enumerate}
\fincorrection

\finexercice
