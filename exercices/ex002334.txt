\exercice{2334, matexo1, 2002/02/01}

\enonce
On définit
\[
  F(x) = \int_0^x \frac{\sin t}{t}\ dt
\]
et pour tout \( n \in \Nn \), on note
\[
  u_n = F((n+1)\pi) - F(n\pi) = \int_{n\pi}^{(n+1)\pi}
  \frac{\sin t}{t} dt.
\]
\begin{enumerate}
  \item Montrer que \( F(x) \) est bien définie pour tout \( x \in \Rr \).

  \item Montrer que si \( k \ge 1 \), alors
        \[
          \frac{2}{(2k+1)\pi} < u_{2k} < \frac{1}{k\pi}.
        \]
        Trouver une inégalité similaire pour \( u_{2k+1} \), puis pour \( u_{2k}+u_{2k+1} \).

  \item Montrer que la suite de terme général \( \displaystyle{v_n = \sum_{i = 1}^n \frac{1}{i^2}} \) admet une limite finie. En déduire que
        \[
          I = \int_0^\infty \frac{\sin t}{t}\ dt
        \]
        est convergente.
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
