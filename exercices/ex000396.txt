\exercice{396, legall, 2003/10/01}

\enonce
\begin{enumerate}
  \item Montrer que le polynôme \( P(X) = X^5-X^2+1 \) admet une unique racine réelle et que celle-ci est irrationnelle.

  \item Montrer que le polynôme \( Q(X) = 2X^3-X^2-X-3 \) a une racine rationnelle (qu'on calculera). En déduire sa décomposition en produit de facteurs irréductibles dans \( \Cc [X] \).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
