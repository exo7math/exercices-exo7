\exercice{5928, tumpach, 2010/11/11}

\enonce
\emph{Le but de cet exercice est de prouver le Théorème de Carathéodory.}

\textbf{Définition.}
Une mesure extérieure sur un ensemble \( \Omega \) est une application \( m_* :\mathcal{P}(\Omega) \rightarrow [0,+\infty] \) telle que
\begin{enumerate}
  \item[(i)] \( m_*(\emptyset) = 0 ; \) \item[(ii)] (monotonie) \( A
        \subset B \Rightarrow m_*(A) \le m_*(B); \) \item[(iii)] (\( \sigma \)-sous-additivité) Pour toute suite d'ensembles \( \{ A_i \}_{i\in\Nn^*} \subset \mathcal{P}(\Omega) \) on a
        \[
          m_{*}\left(\bigcup_{i = 1}^{\infty} A_i \right) \le
          \sum_{i = 1}^{\infty} m_{*}(A_i).
        \]
\end{enumerate}

\textbf{Théorème de Carathéodory}
Soit \( m_* \) une mesure extérieure sur \( \Omega \). Un ensemble \( A\subset\Omega \) est dit \( m_* \)-mesurable si pour tout \( Q\subset
\Omega \) on a
\[
  m_*(Q) \ge m_*(Q \cap A) + m_*(Q\cap A^c).
\]
Notons \( \mathcal{M}_{m_*} \subset \mathcal{P}(\Omega) \) l'ensemble des parties \( m_* \)-mesurables. Alors
\begin{enumerate}
  \item \( \mathcal{M}_{m_*} \) est une \( \sigma \)-algèbre. \item \( m = m_{*}|_{\mathcal{M}_{m_*}} \) est une mesure sur \( (\Omega,
        \mathcal{M}_{m_*}) \). \item L'espace mesuré \( (\Omega,
        \mathcal{M}_{m_*}, m) \) est complet, i.e. si \( E \in
        \mathcal{M}_{m_*} \) et \( m(E) = 0 \), alors tout sous-ensemble \( A\subset E \) appartient à \( \mathcal{M}_{m_*} \).
\end{enumerate}

Début de l'exercice :
\begin{enumerate}
  \item
        \begin{enumerate}
          \item Rappeler la définition d'une \( \sigma \)-algèbre.

          \item Vérifier que \( \emptyset \) et \( \Omega \in \mathcal{M}_{m_*} \), et \( A\in\mathcal{M}_{m_*} \Rightarrow A^c\in\mathcal{M}_{m_*} \).

          \item Soit \( \{A_i\}_{i\in \Nn^*} \) une suite quelconque d'ensembles \( m_* \)-mesurables. On pose \( B_1 = \emptyset \), \( B_2 = A_1 \) et \( B_j = \cup_{i = 1}^{j-1} A_i \), pour \( j\ge 2 \). Soit \( Q \) un sous-ensemble de \( \Omega \). Montrer par récurrence que l'assertion \( (P_k) \) suivante est vérifiée pour tout \( k\ge 1 \):
                \[
                  (P_k)\quad \quad \quad m_*(Q) = m_*(Q \cap B_{k+1}^c) + \sum_{j = 1}^k m_*(Q \cap B_j^c \cap A_j).
                \]

          \item Soit \( A = \cup_{j = 1}^{\infty} A_j \). Déduire de la question précédente que
                \[
                  m_{*}(Q) \ge m_*(Q\cap A^c) +
                  \sum_{j = 1}^{\infty} m_*(Q \cap B_j^c\cap A_j).
                \]
                \( d) \) En remarquant que \( Q\cap A = \bigcup_{j = 1}^{\infty}(Q\cap B_j^c
                \cap A_j) \), montrer :
                \[
                  m_*(Q\cap A^c) + m_*(Q\cap A) \le m_*(Q),
                \]
                et conclure.
        \end{enumerate}

  \item
        \begin{enumerate}
          \item Rappeler la définition d'une mesure.

          \item En utilisant la question 1.d), montrer la \( \sigma \)-additivité de \( m \).
        \end{enumerate}

  \item Montrer que \( m \) est complète.
\end{enumerate}
\finenonce

\noindication

\correction
\begin{enumerate}
  \item
        \begin{enumerate}
          \item cf cours.

          \item clair.

          \item Soit \( \{A_i\}_{i\in
                    \Nn^*} \) un suite quelconque d'ensembles \( m_* \)-mesurables. On pose \( B_1 = \emptyset \), \( B_2 = A_1 \) et \( B_j = \cup_{i = 1}^{j-1}
                A_i \), pour \( j\ge 2 \). Soit \( Q \) un sous-ensemble de \( \Omega \). Montrons par récurrence que l'assertion \( (P_k) \) suivante est vérifiée pour tout \( k\ge 1 \):
                \[
                  (P_k)\quad \quad \quad m_*(Q) = m_*(Q \cap B_{k+1}^c) + \sum_{j = 1}^k m_*(Q \cap B_j^c \cap A_j).
                \]
                \begin{itemize}
                  \item Pour \( k = 1 \), \( (P_1) \) dit simplement que \( m_*(Q) = m_*(Q\cap A_1^c) + m_*(Q\cap A_1) \). Ceci est une conséquence de la \( m_* \)-mesurabilité de \( A_1 \) et de fait que
                        \[
                          m_*(Q) \le m_*(Q\cap A_1^c) + m_*(Q\cap A_1)
                        \]
                        (on applique la \( \sigma \)-sous-additivité de \( m_* \) à \( C_1 = Q\cap A_1^c \), \( C_2 = Q\cap A_1 \) et \( C_i = \emptyset \) pour \( i\ge 3 \).)

                  \item Montrons que \( (P_k) \Rightarrow (P_{k+1}) \):\\
                        Puisque \( A_{k+1} \) est \( m_* \)-mesurable, on a :
                        \[
                          m_*(Q\cap B_{k+1}^c) = m_*(Q \cap B_{k+1}^c \cap A_{k+1}^c) + m_*(Q\cap B_{k+1}^c \cap A_{k+1}).
                        \]
                        Or \( B_{k+1}^c \cap A_{k+1}^c = (B_{k+1}\cup A_{k+1})^c = B_{k+2}^c \). Ainsi :
                        \begin{equation}\label{la}
                          m_*(Q\cap B_{k+1}^c) = m_*(Q \cap B_{k+2}^c) + m_*(Q\cap B_{k+1}^c \cap A_{k+1}).
                        \end{equation}
                        Supposons que l'assertion \( (P_k) \) soit vérifiée, alors
                        \[
                          m_*(Q) = m_*(Q \cap B_{k+1}^c) + \sum_{j = 1}^k m_*(Q \cap B_j^c \cap A_j),
                        \]
                        et d'après \eqref{la}
                        \begin{eqnarray*} m_*(Q) &=& m_*(Q \cap B_{k+2}^c) + m_*(Q\cap B_{k+1}^c \cap A_{k+1})+ \sum_{j=1}^k m_*(Q \cap B_j^c \cap A_j) \\&=& m_*(Q \cap B_{k+2}^c)+ \sum_{j=1}^{k+1}
                          m_*(Q \cap B_j^c \cap A_j),
                        \end{eqnarray*}
                        qui n'est autre que \( (P_{k+1}) \). \item En conclusion, comme \( (P_1) \) est vrai et \( (P_k)\Rightarrow (P_{k+1}) \), il en découle que l'assertion \( (P_k) \) est vraie pour tout \( k\ge 1 \).
                \end{itemize}

          \item Comme \( B_{k+1} \subset A \), on a \( Q \cap B_{k+1}^c \supset Q
                \cap A^c \) et, par monotonie de \( m_* \),
                \[
                  m_*(Q\cap B_{k+1}^c)
                  \ge m_*(Q \cap A^c).
                \]
                La condition \( (P_k) \) entraîne alors que pour tout \( k \):
                \[
                  m_*(Q) \ge m_*(Q \cap A^c) + \sum_{j = 1}^k m_*(Q\cap B_j^c
                  \cap A_j).
                \]
                Donc, en faisant tendre \( k \) vers \( +\infty \):
                \[
                  m_*(Q) \ge m_*(Q \cap A^c) + \sum_{j = 1}^{\infty} m_*(Q\cap B_j^c \cap A_j).
                \]

          \item On a :\(  Q\cap A = \bigcup_{j = 1}^{\infty}(Q\cap B_j^c \cap A_j)  \) et par \( \sigma \)-sous-additivité de \( m_* \):
                \begin{eqnarray*}
                  m_*(Q\cap A^c) + m_*(Q\cap A) &=& m_*(Q\cap A^c) + m_*\left(\bigcup_{j=1}^{\infty}(Q\cap B_j^c \cap A_j\right)\\
                  & \le & m_*(Q\cap A^c) + \sum_{j=1}^{\infty} m_*(Q\cap B_j^c
                  \cap A_j)\\&\le & m_*(Q).
                \end{eqnarray*}
                On en conclut que \( A = \cup_{j = 1}^{\infty} A_j \) est \( m_* \)-mesurable.
        \end{enumerate}

  \item
        \begin{enumerate}
          \item cf cours.

          \item Soit \( \{A_i\}_{i\in\Nn^*} \) une suite d'éléments \( m_* \)-mesurables, deux à deux disjoints. Choisissons \( Q = A = \cup_{j = 1}^{\infty} A_j \), alors \( Q \cap A^c = \emptyset \) et \( Q \cap B_j^c \cap A_j = A_j \) pour tout \( j \). D'après la question 1.d),
                \[
                  m_{*}(Q) \ge \sum_{j = 1}^{\infty} m_*(A_j).
                \]
                D'après la \( \sigma \)-sous-additivité de \( m_* \), il vient :
                \[
                  m_{*}(Q) = \sum_{j = 1}^{\infty} m_*(A_j).
                \]
        \end{enumerate}

  \item Soit \( E \) un ensemble \( m_* \)-mesurable tel que \( m_*(E) = 0 \) et \( B \) un sous-ensemble de \( E \). Comme \( Q \cap B^c \subset Q \), on a par monotonie de \( m_* \) l'inégalité \( m_*(Q \cap B^c) \le m_*(Q) \). Comme \( Q\cap B \subset E \), on a aussi \( m_*(Q\cap B) = 0 \). On en déduit que
        \[
          m_*(Q) \ge m_*(Q \cap B^c)+ m_*(Q\cap B).
        \]
        Ainsi \( B \) est \( m_* \)-mesurable et \( m \) est complète.
\end{enumerate}
\fincorrection

\finexercice
