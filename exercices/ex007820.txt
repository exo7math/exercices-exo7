\exercice{7820, mourougane, 2021/08/11}

\enonce[Produit de formes linéaires]
\begin{enumerate}
  \item Montrer que le polynôme quadratique à coefficients complexes avec \( a\not=0 \)
        \[
          ax^2+a'y^2+a''z^2+2byz+2b'zx+2b''xy
        \]
        est le produit de deux polynômes linéaires si et seulement si
        \[
          aa'-{b''}^2 = aa''-{b'}^2 = ab-b'b'' = 0.
        \]

  \item L'analogue à coefficients réels est-il vrai ?
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
