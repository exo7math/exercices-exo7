\exercice{3599, quercia, 2010/03/10}

\enonce[Matrice bloc]
Soit \( A \in GL_n(\Cc) \) et \( M = \begin{pmatrix} 0 & A \\ I &0
\end{pmatrix} \in \mathcal{M}_{2n}(\Cc) \). Montrer que \( M \) est diagonalisable si et seulement si \( A \) l'est. (Chercher les sous-espaces propres de \( M \) en fonction de ceux de \( A \))
\finenonce

\noindication

\correction
\( E_\lambda(M) = \left\{ \begin{pmatrix}\lambda Y \\ Y\hfil\end{pmatrix} \text{ tq }
AY = \lambda^2Y \right\} \).
\fincorrection

\finexercice
