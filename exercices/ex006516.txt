\exercice{6516, drutu, 2011/10/16}

\enonce
\begin{enumerate}
  \item Montrer que \( \Zz[X] \) est un anneau factoriel.

  \item Dans \( \Zz [X] \) montrer que l'ensemble des polynômes de terme constant un nombre pair est un idéal mais non principal. En déduire que \( \Zz[X] \) est un anneau non principal.

  \item Montrer que l'idéal \( (X) \) est premier mais non maximal dans \( \Zz[X] \).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
