\exercice{1501, ortiz, 1999/04/01}

\enonce
On munit le \( \Rr \)-espace vectoriel \( \Rr_2[X] \) du produit scalaire défini par
\[
  {\phi:\Rr_2[X]\times\Rr_2[X]\to\Rr, (P,Q)\mapsto\int_{-1}^1P(t)Q(t)\,dt.}
\]
\begin{enumerate}
  \item Déterminer la base orthonormée de Gram-Schmidt de la base canonique de \( \Rr_2[X] \).

  \item Déterminer la distance du polynôme \( P = X^2+X+1 \) au sous-espace vectoriel \( F \) de \( \Rr_2[X] \) formé des polynômes \( f \) tels que \( f'(0) = 0 \).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
