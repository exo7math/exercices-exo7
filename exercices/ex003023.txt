\enonce[Suites stationnaires]

\begin{enumerate}
  \item Montrer que \( A \) est un anneau.

  \item Chercher les morphismes d'anneaux : \( A \to \Zz \).

  \item Soit \( I = \{\text{suites presque nulles}\} \). Montrer que c'est un idéal non principal.
\end{enumerate}
