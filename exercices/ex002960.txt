\exercice{2960, quercia, 2010/03/08}

\enonce[\( x + y - xy \)]
\begin{enumerate}
  \item Sur \( E = [0,1] \), on définit l'opération : \( x * y = x + y - xy \). Vérifier que \( * \) est interne, et étudier ses propriétés (commutativité, associativité, élément neutre, éléments symétrisables, éléments réguliers).

  \item Mêmes questions avec \( E = ]-\infty, 1[ \).

\end{enumerate}
\finenonce

\noindication

\correction
\begin{enumerate}
  \item Commutative, associative, 0 = élt neutre, tout élt \( \ne \) 1 est régulier, seul 0 est symétrisable.

  \item Tout élt est symétrisable et \( x^{-1} = \frac x{x-1} \).
\end{enumerate}
\fincorrection

\finexercice
