\exercice{6412, potyag, 2011/10/16}

\enonce
\begin{enumerate}
  \item Soit \( S = \{z+tj\ \vert\ z\in\Cc,\ t\in\Rr\} \) un sous-ensemble des quaternions \( \mathbb{H} \). Montrer que \( S \) est invariant par l'application :
        \[
          \rho_j : \mathbb{H} \to \mathbb{H},\ q\mapsto jqj^{-1}.
        \]

  \item En identifiant \( S \) avec \( \Rr^3 \) décrire \( \rho_j:S \to S \).

  \item Montrer que l'application \( \rho_k:q\to kqk^{-1} \) laisse \( S \) invariant (i.e. \( \rho_k(S) = S \)). Décrire \( \rho_k \).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
