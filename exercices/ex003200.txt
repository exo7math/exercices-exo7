\exercice{3200, quercia, 2010/03/08}

\enonce[Congruences]
Déterminer les polynômes \( P \in \Qq_3[X] \) divisibles par \( X+1 \) et dont les restes des divisions par \( X+2,X+3,X+4 \) sont égaux.
\finenonce

\noindication

\correction
\( P = \lambda((X+2)(X+3)(X+4)-6) \).
\fincorrection

\finexercice
