\exercice{4782, quercia, 2010/03/16}

\enonce[Adhérence et intérieur dans les fonctions continues]
\begin{enumerate}
  \item Soit \( E = \mathcal{C}([0,1],\Rr) \) muni de la norme de la convergence uniforme. Soit \( P \) l'ensemble des fonctions de \( E \) positives ou nulles. Chercher \( \overline P \) et \( \mathring P \).

  \item Mêmes questions avec la norme : \( \|f\| = \int_{t = 0}^1 |f(t)|\,dt \).

\end{enumerate}
\finenonce

\noindication

\correction
\begin{enumerate}
  \item \( \overline P = P \), \( \mathring P = \{\text{fonctions strictement positives}\} \).

  \item \( \overline P = P \), \( \mathring P = \varnothing \).
\end{enumerate}
\fincorrection

\finexercice
