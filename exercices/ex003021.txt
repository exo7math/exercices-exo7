\exercice{3021, quercia, 2010/03/08}

\enonce[Entiers 2-adiques]
Soit \( A = \{ m/n \in \Qq \text{ tel que } n \text{ est impair} \} \).
\begin{enumerate}
  \item Montrer que \( A \) est un sous-anneau de \( \Qq \).

  \item Chercher les éléments inversibles dans \( A \).

  \item Montrer que les idéaux de \( A \) sont tous principaux engendrés par les nombres de la forme \( 2^k \), \( k\in \Nn \).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
