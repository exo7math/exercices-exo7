\exercice{6348, queffelec, 2011/10/16}

\enonce
Pour quelle(s) valeur(s) de \( k \) l'origine est-elle un puits pour l'équation \( x' = Ax \)?
\begin{enumerate}
  \item \( \begin{pmatrix} a & -k \\ k &2
        \end{pmatrix} \),

  \item \( \begin{pmatrix} 3 & 0 \\ k &-4
        \end{pmatrix} \),

  \item \( \begin{pmatrix} k^2 & 1 \\ 0 &k
        \end{pmatrix} \),

  \item \( \begin{pmatrix} 0 & -1 & 0 \\ 1&0&0 \\ -1 &0 &k
        \end{pmatrix} \).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
