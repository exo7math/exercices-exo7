\exercice{7372, mourougane, 2021/08/10}

\enonce
\begin{enumerate}
  \item Quels sont les éléments inversibles de l'anneau \( (\Zz/12\Zz) \) ?

  \item À quel groupe le groupe \( (\Zz/12\Zz)^\times \) des inversibles de l'anneau \( (\Zz/12\Zz) \) est-il isomorphe ?
\end{enumerate}
\finenonce

\noindication

\correction
\begin{enumerate}
  \item

  \item Comme \( \phi(12) = \phi(3)\phi(2^2) = 2\times 2 = 4 \) il y a quatre éléments inversibles dans l'anneau \( (\Zz/12\Zz) \). Il s'agit de \( 1,-1 = 11,5,-5 = 7 \). Comme \( 5^5 = 25 = 1[12] \), tous les éléments sont d'ordre \( 1 \) ou \( 2 \). Le groupe \( (\Zz/12\Zz)^\times \) est donc isomorphe à \( \Zz/2\Zz\times\Zz/2\Zz \). On peut aussi remarquer que par le théorème chinois, \( (\Zz/12\Zz)^\times = (\Zz/4\Zz)^\times\times(\Zz/3\Zz)^\times = \Zz/2\Zz\times\Zz/2\Zz \).
\end{enumerate}
\fincorrection

\finexercice
