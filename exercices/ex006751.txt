\exercice{6751, queffelec, 2011/10/16}

\enonce
Calculer

\begin{enumerate}
  \item \( \displaystyle \int_0^{2\pi}\frac{dt}{a+\cos t} \) (\( a>1 \)) ;

  \item \( \displaystyle \int_0^{2\pi}\frac{dt}{(a+b\cos t)^2} \) (\( a>b>0 \)) ;

  \item \( \displaystyle \int_0^{2\pi}\frac{\cos^3{3t}}{1-2a\cos t+a^2}\ dt \) (\( a\in\Cc\setminus \{\pm 1\} \)) ;

  \item \( \displaystyle \int_0^{2\pi}\exp{(\cos t)}\cos{(nt-\sin t)}\ dt \) (\( n\in \Zz \)).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
