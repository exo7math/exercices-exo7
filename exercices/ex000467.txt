\exercice{467, bodin, 1998/09/01}

\enonce
Soit
\[
  I = \left\lbrace x \in \Rr \setminus\{0\} \ | -2 < x +\frac{1}{2x} \le 2 \right\rbrace.
\]
\begin{enumerate}
  \item Montrer que \( I \) est la réunion de deux intervalles.

  \item Déterminer (s'ils existent) : les majorants, les minorants, la borne supérieure, la borne inférieure, le plus grand élément, le plus petit élément de \( I \).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
