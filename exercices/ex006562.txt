\exercice{6562, drutu, 2011/10/16}

\enonce
\begin{enumerate}
  \item Soit \( P\in K[X] \) un polynôme irréductible de degré \( n \) et \( E \supset K \) une extension de degré \( m \). Montrer que si \( (m,n) = 1 \) alors \( P \) est irréductible dans \( E[X] \).

  \item Montrer que \( X^3+X+1 \) est irréductible sur \( \Qq [i] \).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
