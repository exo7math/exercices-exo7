\enonce
Les fonctions suivantes sont-elles intégrables au sens de Riemann?
\begin{enumerate}
  \item \( f(x) = [x] \) sur \( [0,2] \)

  \item \( g: [0,1] \rightarrow \Rr,\quad g(x) = \begin{cases}
          \left[\frac 1x\right] & \text{ si } 0 <x \le 1, \\
          1                     & \text{ si } x = 0
        \end{cases} \)

  \item \( h: [0,1] \rightarrow \Rr,\quad h(x) = \begin{cases}
          \frac 1x\sin \left(\frac 1x\right) & \text{ si } 0 < x \le 1, \\
          1                                  & \text{ si } x = 0
        \end{cases} \)

  \item \( k: [0,1] \rightarrow \Rr,\quad k(x) = \begin{cases}
          1 & \text{ si } x\in [0,1]\cap \Qq,      \\
          0 & \text{ si } x\in [0,1]\backslash \Qq
        \end{cases} \)
\end{enumerate}
\finenonce
