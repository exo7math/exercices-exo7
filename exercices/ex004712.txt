\exercice{4712, quercia, 2010/03/16}

\enonce[Système d'ordre 1]
Étudier la convergence des suites \( (x_n) \), \( (y_n) \) définies par : \( 0 < y_0 < x_0 \) et \( \begin{cases}x_{n+1} = \frac{x_n+y_n}2 \\
  y_{n+1} = \frac{2x_ny_n}{x_n+y_n} .
\end{cases} \)
\finenonce

\noindication

\correction
\( x_ny_n = \text{c}^{\text{te}} \Rightarrow x_n,y_n \to \sqrt{x_0y_0} \).
\fincorrection

\finexercice
