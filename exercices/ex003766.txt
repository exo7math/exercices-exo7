\exercice{3766, quercia, 2010/03/11}

\enonce[Diagonalisation de matrices symétriques]
Diagonaliser dans une base orthonormée :

\begin{enumerate}
  \item \( A = \begin{pmatrix} 6 & -2 & 2 \\ -2 &5 &0 \\ 2 &0 &7
        \end{pmatrix} \).

  \item \( A = \frac19\begin{pmatrix} 23 & 2 & -4 \\ 2 &26 &2 \\ -4 &2 &23
        \end{pmatrix} \).

\end{enumerate}
\finenonce

\noindication

\correction
\begin{enumerate}
  \item \( P = \frac13\begin{pmatrix}2 & -1 & 2 \\ 2 &2 &-1 \\ -1 &2 &2
        \end{pmatrix} \), \( D = \text{Diag}(3\ 6\ 9) \).

  \item \( P = \frac13\begin{pmatrix}2 & -1 & 2 \\ 2 &2 &-1 \\ -1 &2 &2
        \end{pmatrix} \), \( D = \text{Diag}(3\ 3\ 2) \).
\end{enumerate}
\fincorrection

\finexercice
