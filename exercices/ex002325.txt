\exercice{2325, matexo1, 2002/02/01}

\enonce
Calculer les primitives des fonctions suivantes :
\begin{align*}
  t & \mapsto \tan t,                                   \\
  t & \mapsto \arg\sinh t,                              \\
  t & \mapsto \frac{\tan^3 t}{\cos^6 t},                \\
  t & \mapsto \frac{1}{\sqrt{t} + \sqrt[3]{t}},         \\
  t & \mapsto \frac{1}{2\sqrt{t} + \sqrt{t+2}},         \\
  t & \mapsto t \sqrt[n]{1+t},                          \\
  t & \mapsto \cosh^3 t,                                \\
  t & \mapsto \frac{t^3}{(a^2-t^2)^{3/2}},              \\
  t & \mapsto \frac{\sqrt[3]{1+\sqrt[4]{t}}}{\sqrt{t}}, \\
  t & \mapsto \sqrt{t \sqrt{t \sqrt{t}}},               \\
  t & \mapsto \frac{1}{t^2 \sqrt[4]{\frac{t+1}{t-1}}}.
\end{align*}
\finenonce

\noindication

\nocorrection

\finexercice
