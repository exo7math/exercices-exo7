\exercice{2501, sarkis, 2009/04/01}

\enonce
Calculez la norme des opérateurs suivants:
\begin{enumerate}
  \item Le shift sur \( l^\infty \) défini par \( S(x)_{n+1} = x_n, S(x)_0 = 0 \) (sur \( l^\infty \) on définit \( ||(x_n)||_\infty = \sup_{n \in \Nn}|x_n|) \).

  \item \( X = {\mathcal C}([0,1]) \) avec la norme sup et l'opérateur \( Tf(x) = f(x)g(x) \) où \( g \in X \).

  \item \( X = {\mathcal C}([0,1]) \) muni de la norme sup et \( u(f) = \int_0^1f(x)g(x)dx \) où \( g \in X \) est une fonction qui ne s'annule qu'en \( x = 1/2 \).

  \item \( X = l^2 \) et \( u(x) = \sum a_nx_n \) où \( (a_n) \) est dans \( X \).

  \item \( X \) l'espace des suites convergentes muni de la norme sup et \( u: X \rightarrow \Rr \) l'application \( u(x) = \lim_{j \rightarrow \infty} x_j \).
\end{enumerate}
\finenonce

\noindication

\correction
\begin{enumerate}
  \item Soit \( x \) une suite, on a
        \[
          ||S(x)||_\infty = Max(\sup_{n \in \Nn}|x_{n-1}|,0) = \sup_{n \in
            \Nn}|x_n| = 1.||x||_\infty.
        \]
        Donc \( ||S|| = 1 \). \item Soit \( f
        \in {\mathcal C}([0,1]) \)
        \[
          ||Tf||_\infty = \sup_{x \in [0,1]}f(x)g(x)\le ||f||_\infty||g||_\infty.
        \]
        Donc
        \[
          ||T||\le ||g||_\infty.
        \]
        Or
        \[
          ||T1||_\infty = ||g||_\infty = ||1||_\infty||g||_\infty.
        \]
        Donc
        \[
          ||T|| \ge ||g||_\infty
        \]
        et finalement on a bien
        \[
          ||T|| = ||g||_\infty.
        \]

  \item Soit \( f \in {\mathcal C}([0,1]) \), on a
        \[
          ||u(f)|| = |\int_0^1f(x)g(x)dx| \le \int_0^1|f(x)||g(x)|dx \le
          \sup_{x \in [0,1]}|g(x)| \int_0^1 |f(x)|dx \le ||g||_\infty.||f||.
        \]
        On a donc
        \[
          ||u|| \le ||g||_\infty.
        \]
        Comme \( g \) ne s'annule qu'au point \( x = 1/2 \), elle ne change de signe qu'une seule fois. Soit
        \[
          f_0 = g/|g|,
        \]
        cette fonction n'est pas continue (ni définie) en \( x = 1/2 \) mais vérifie \( f_0g = |g| \). Prenons \( f_n = g/|g| \) si \( |x-1/2|> 1/n \), pour \( |x-1/2|\le 1/n \), on relie les deux segments du graphe par une ligne. Alors \( 1-1/(2n)\le ||f_n|| \le 1 \) et
        \[
          ||u(f_n)|| = |\int_{|x-1/2|>1/n}f_n(x)g(x)dx+\int_{|x-1/2| \le 1/n}f_n(x)g(x)dx| \ge
        \]
        \[
          |(|\int_{|x-1/2|>1/n}f_n(x)g(x)dx|-|\int_{|x-1/2| \ge 1/n}f_n(x)g(x)dx|)|
        \]
        \[
          \ge ||g||_\infty \int_{|x-1/2|>1/n}|f_n(x)|dx-2/n ||g||_\infty\ge ||g||_\infty(||f_n|| -2/n).
        \]
        Ainsi
        \[
          \lim_{n\rightarrow
            \infty}||u(\frac{f_n}{||f_n||})||\ge ||g||_\infty (1-\frac{1}{2n||f_n||} \ge ||g||_\infty (1-\frac{1}{2n(1-1/2n))}
          \ge ||g||_\infty (1-\frac{1}{2n-1})
        \]
        et donc pour tout \( n \in \Nn^* \):
        \[
          ||u|| \ge ||g||_\infty (1-\frac{1}{2n-1}),
        \]
        en faisant tend \( n \) vers l'infini
        \[
          ||u|| \ge ||g||_\infty
        \]
        ce qui montre la deuxième inégalité et on obtient \( ||u|| = ||g||_\infty \).

  \item si on prend \( (x_n) = (a_n) \) on obtient
        \[
          u((a_n)) = \sum a_n^2 = ||(a_n)||_2^2 = ||(a_n)||_2.||(a_n)||_2
        \]
        et donc
        \[
          ||u|| \ge ||(a_n)||_2.
        \]
        Or D'après Cauchy-Schwartz, on a
        \[
          ||u(a_n)|| = |u(a_n)| = |\sum a_n x_n| \le ||(a_n)||_2||(x_n)||_2
        \]
        et donc \( ||u|| \le ||(a_n)||_2 \) d'où l'égalité
        \[
          ||u|| = ||(a_n)||_2.
        \]

  \item Pour tout \( j \in \Nn \) on a \( |x_j| \le ||(x_n)||_\infty \) et par conséquent
        \[
          |u((x_n))| = |\lim_{j \rightarrow \infty} x_j|
          \le ||(x_n)||_\infty
        \]
        et donc
        \[
          ||u|| \le 1.
        \]
        Prenons la suite \( (x^0) \) définie par \( x_n^0 = 1 \) pour tout \( n \in \Nn \) alors
        \[
          |u(x^0)| = |\lim_{j \rightarrow \infty} 1| = 1 = ||x^0||_\infty
        \]
        et donc
        \[
          ||u|| \ge 1
        \]
        d'où l'égalité \( ||u|| = 1 \).
\end{enumerate}
\fincorrection

\finexercice
