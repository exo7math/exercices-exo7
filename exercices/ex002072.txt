\exercice{2072, maillot, 2001/09/01}

\enonce
On considère le champ de vecteurs \( P:\Rr^2\rightarrow \Rr^2 \) défini par
\[
  P(x,y) = (2x {\mathrm e}^{x^2-2y};-2 {\mathrm e}^{x^2-2y}).
\]
\begin{enumerate}
  \item Vérifier que la forme différentielle associée à \( P \) est fermée.

  \item En déduire que \( P \) est un champ de gradient et en déterminer un potentiel.

  \item Calculer la circulation de \( P \) le long du chemin
        \[
          \gamma:[0,1]\rightarrow
          \Rr^2;\quad t\mapsto (\ln (1+t);{\mathrm e}^t +1).
        \]
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
