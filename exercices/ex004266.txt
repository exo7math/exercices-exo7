\exercice{4266, quercia, 2010/03/12}

\enonce[Diverses primitives]
\[
  \begin{array}{ll}
    x^k\ln x                      & \frac{x^{k+1}}{k+1} \Bigl(\ln x - \frac1{k+1} \Bigr)                                                                       \\
    \ln(1+x^2)                    & x\ln(1+x^2) -2x + 2\arctan x                                                                                               \\
    \frac{x^2+a}{x^2+1} \arctan x & \frac12 \bigl( (2x+(a-1)\arctan x)\arctan x - \ln(1+x^2)\bigr)                                                             \\
    \Bigl(1-\frac1x \Bigr)e^{1/x} & xe^{1/x}                                                                                                                   \\
    \frac x{\cos^2x}              & x\tan x + \ln|\cos x|                                                                                                      \\
    \frac1{\sqrt{e^x-1}}          & 2\arctan\sqrt{e^x-1}                                                                                                       \\
    \arctan\sqrt{\frac{x+1}{x+3}} & (x+2)\arctan\sqrt{\frac{x+1}{x+3}} - \ln\bigl( \sqrt{x+1} + \sqrt{x+3}\, \bigr)                                            \\
    \arcsin\sqrt{\frac x{x+1}}    & x\arcsin\sqrt{\frac x{x+1}} - \sqrt x + \arctan \sqrt x                                                                    \\
    e^{\arcsin x}                 & \frac{x+\sqrt{1-x^2}}2 e^{\arcsin x}                                                                                       \\
    x(\cos^2x)e^{-x}              & \frac{e^{-x}}{50} \bigl( (3-5x)\cos 2x + (4+10x)\sin 2x - 25(x+1) \bigr)                                                   \\
    (x^2+x+1)e^{2x}\cos x         & (\frac{2x^2}5 + \frac{4x}{25} + \frac{39}{125} )e^{2x}\cos x + (\frac{x^2}5 - \frac{3x}{25} + \frac{27}{125} )e^{2x}\sin x
  \end{array}
\]
\finenonce

\noindication

\nocorrection

\finexercice
