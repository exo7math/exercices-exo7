\exercice{3340, quercia, 2010/03/09}

\enonce[Éléments réguliers dans \( \mathcal{L}(E) \)]
Soit \( f \in \mathcal{L}(E,F) \).

\begin{enumerate}
  \item Montrer que : 
  \[
    (f \text{ est injectif}) \iff (\forall g \in \mathcal{L}(E), f \circ g = 0 \Rightarrow g = 0).
  \]
  


  \item Montrer que : 
  \[
    (f \text{ est surjectif}) \iff (\forall g \in \mathcal{L}(F), g\circ f = 0 \Rightarrow g = 0).
  \]

\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
