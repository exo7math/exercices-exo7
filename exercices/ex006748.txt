\exercice{6748, queffelec, 2011/10/16}

\enonce[Calcul d'intégrales semi-convergentes]
\begin{enumerate}
  \item Calculer \( \displaystyle\int_\Rr\frac{\sin x}{x}\ dx \) en intégrant \( f(z) = \displaystyle\frac{e^{iz}}{z} \) sur un contour bien choisi.

  \item Calculer de même \( \displaystyle\int_\Rr\frac{x^2-1}{x^2+1}\frac{\sin x}{x}\ dx \).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
