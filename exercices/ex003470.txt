\exercice{3470, quercia, 2010/03/10}

\enonce[Calcul de rang]
Soient \( x_1, \dots, x_p, y_1, \dots, y_p \in K \). On note \( A,B \) les matrices de termes généraux \( x_i + y_j \) et \( (x_i+y_j)^2 \).

Chercher les rangs de \(A\) et \(B\).
\finenonce

\noindication

\correction
\( A = \begin{pmatrix} 1 & x_1 \\ \vdots &\vdots \\ 1 &x_n
\end{pmatrix}
\begin{pmatrix} y_1 & \dots & y_n \\ 1 &\dots &1
\end{pmatrix} \Rightarrow
\mathrm{rg}(A) = \begin{cases} 2 & \text{ si } (x_i) \text{ et } (y_j) \text{ ne sont pas constantes } \\
              1 & \text{ ou } 0 \text{ sinon.}
\end{cases} \)

\( B = \begin{pmatrix} 1 & x_1 & x_1^2 \\ \vdots &\vdots &\vdots \\ 1 &x_n &x_n^2
\end{pmatrix}
\begin{pmatrix} y_1^2 & \dots & y_n^2 \\ 2y_1 &\dots &2y_n \\ 1 &\dots &1
\end{pmatrix}
\Rightarrow
\mathrm{rg}(B) = \begin{cases} 3 & \text{ si } \mathrm{Card}\,(x_i)\ge 3 \text{ et } \mathrm{Card}\,(y_j) \ge 3 \\
              2 & \text{ si } \min\bigl(\mathrm{Card}\,(x_i),\mathrm{Card}\,(y_j)\bigr) = 2    \\
              1 & \text{ ou } 0 \text{ sinon.}
\end{cases} \)
\fincorrection

\finexercice
