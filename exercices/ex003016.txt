\exercice{3016, quercia, 2010/03/08}

\enonce[Étude de l'anneau \( \Zz^2 \)]
\begin{enumerate}
  \item Soit \( d \in \Nn \). On pose
        \[
          A_d = \{(x,y) \in \Zz^2 \text{ tq } x\equiv y (\mathrm{mod}\, d)\}
        \]
        (\( x = y \) pour \( d = 0 \)). Montrer que \( A_d \) est un sous-anneau de \( \Zz^2 \).

  \item Montrer que l'on obtient ainsi tous les sous-anneaux de \( \Zz^2 \).

  \item Soit \( I \) un idéal de \( \Zz^2 \). On note : \( \begin{cases} I_1 = \{ x \in \Zz \text{ tq } (x,0) \in I \} \\
          I_2 = \{ y \in \Zz \text{ tq } (0,y) \in I \}.\end{cases} \)

        Montrer que \( I_1 \) et \( I_2 \) sont des idéaux de \( \Zz \), et que \( I = I_1 \times I_2 \).

  \item En déduire que \( I \) est un idéal principal.
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
