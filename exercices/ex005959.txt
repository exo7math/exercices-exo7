\exercice{5959, tumpach, 2010/11/11}

\enonce
Soient \( f\in L^1(\Rr^n) \) et \( g\in L^p(\Rr^n) \) avec \( 1\le p\le +\infty \), où \( \Rr^n \) est muni de la mesure de Lebesgue. Montrer que, pour presque tout \( x\in \Rr^n \), la fonction \( y\mapsto f(x-y)\, g(y) \) est intégrable sur \( \Rr^n \) et que le \emph{ produit de convolution} de \( f \) et \( g \) défini par
\[
  f*g(x) = \int_{\Rr^n} f(x-y) \,g(y)\,dy
\]
vérifie \( f*g(x) = g*f(x) \) et
\[
  \|f*g\|_p \le \|f\|_1\,\|g\|_p.
\]
\finenonce

\noindication

\correction
Soient \( f\in L^1(\Rr^n) \) et \( g\in L^p(\Rr^n) \) avec \( 1\le p\le +\infty \), où \( \Rr^n \) est muni de la mesure de Lebesgue. L'identité \( f*g(x) = g*f(x) \) s'obtient par changement de variable. En ce qui converne l'inégalité \( \|f*g\|_p \le \|f\|_1\,\|g\|_p \), on distingue les cas en fonction de la valeur de \( p \).
\begin{enumerate}
  \item Pour \( p = + \infty \), c'est clair.

  \item Supposons que \( p = 1 \) et posons \( F(x, y) = f(x-y)\,g(y) \). Pour presque tout \( y\in\Rr^n \), on a :
        \[
          \int_{\Rr^n} |F(x, y)|\,dx = |g(y)|
          \int_{\Rr^n}|f(x- y)|\,dx = |g(y)| \cdot\|f\|_1,
        \]
        et
        \[
          \int_{\Rr^n} dy \int_{\Rr^n}|F(x, y)|\,dx =
          \|f\|_1\,\|g\|_1.
        \]
        D'après le théorème de Tonelli, \( F\in L^1\left(\Rr^n\times\Rr^n\right) \). D'après le théorème de Fubini, on a
        \[
          \int_{\Rr^n} |F(x, y)|\,dy <+\infty \quad\quad\text{pour presque tout } x\in\Rr^n,
        \]
        et
        \[
          \int_{\Rr^n} dx \int_{\Rr^n}|F(x, y)|\,dy
          \le\|f\|_1\,\|g\|_1.
        \]
        Ainsi,
        \[
          \|f*g\|_1 = \int_{\Rr^n} \,dx |f*g(x)| = \int_{\Rr^n} \,dx \left|\int_{\Rr^n} F(x, y)\,dy\right| \le\|f\|_1\,\|g\|_1.
        \]

  \item Supposons que \( 1< p<+\infty \). Utilisons le cas précédent, en faisant jouer ici à \( g^p \) le rôle alors joué par \( g \). Alors pour presque tout \( x\in\Rr^n \) fixé, la fonction \( y \mapsto |f(x-y)| |g(y)|^p \) est intégrable sur \( \Rr^n \), i.e. la fonction \( y \mapsto |f(x-y)|^{\frac{1}{p}}
        |g(y)| \) appartient à \( L^p(\Rr^n) \). Soit \( p' \) tel que \( \frac{1}{p} + \frac{1}{p'} = 1 \). La fonction \( y\mapsto |f(x-y)|^{\frac{1}{p'}} \) appartient à \( L^{p'}(\Rr^n) \) car \( f\in L^1(\Rr^n) \) et la mesure de Lebesgue est invariante par translation. D'après l'inégalité de Hölder,
        \[
          |f(x-y)| \,|g(y)| = |f(x-y)|^{\frac{1}{p}}
          \,|g(y)|\cdot|f(x-y)|^{\frac{1}{p'}}\in L^1(\Rr^n)
        \]
        et
        \[
          \int_{\Rr^n} |f(x-y)| \,|g(y)| \le \left(
          \int_{\Rr^n} |f(x-y)|
          \,|g(y)|^p\right)^{\frac{1}{p}}\cdot\|f\|_1^{\frac{1}{p'}},
        \]
        ainsi
        \[
          |(f*g)(x)|^p \le (|f|*|g|^p)(x)\cdot\|f\|_1^{\frac{p}{p'}}.
        \]
        D'après le cas précédent, on voit que
        \[
          f*g\in L^p(\Rr^n) \quad \text{et}\quad \|f*g\|_p^p \le
          \|f\|_1\|g\|_p^p\cdot\|f\|_1^{\frac{p}{p'}},
        \]
        c'est-à-dire
        \[
          \|f*g\|_p \le \|f\|_1\cdot\|g\|_p.
        \]
\end{enumerate}
\fincorrection

\finexercice
