\exercice{4535, quercia, 2010/03/14}

\enonce[Fonction définie par une série]
\begin{enumerate}
  \itemÉtablir la convergence simple sur \( \Rr \) de la série de fonctions : \( f(x) = \sum_{n = 0}^\infty \frac{(\sin x)^2}{\ch nx} \).

  \item Montrer que la convergence est uniforme sur toute partie de la forme \( \Rr\setminus[-\alpha,\alpha] \), \( \alpha > 0 \). Que pouvez-vous en déduire pour \( f \) ?
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
