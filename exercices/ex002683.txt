\enonce
A

\begin{enumerate}
  \item Soient \( a \) et \( z \) deux réels. Soit \( f \) une fonction de classe \( C^{n+1} \) sur le segment d'extrémités \( a \) et \( z \) et \( \phi \) un polynôme de degré \( n \). Prouver que pour tout \( t \) compris dans l'intervalle \( [0,1] \),
        \[
          \frac{d}{dt}\sum_{m = 1}^n (-1)^m (z-a)^m
          \phi^{(n-m)}(t) f^{(m)}\left(a+t(z-a)\right)
        \]
        \[
          = -(z-a) \phi^{(n)}(t) f'\left(a+t(z-a)\right) + (-1)^n (z-a)^{n+1} \phi(t) f^{(n+1)}\left(a+t(z-a)\right)
        \]

  \item

        \begin{enumerate}
          \item Montrer que la fonction \( t\mapsto \frac{t}{e^t-1} \) est prolongeable par continuité en zéro, que son prolongement est indéfiniment dérivable et admet des développements limités en zéro de la forme:
                \[
                  1 - \frac{t}{2} + \frac{b_1 t^2}{2!}
                  + \frac{b_2 t^4}{4!} + \dots + \frac{b_n t^{2n}}{(2n)!}
                  +o(t^{2n+1}),
                \]
                où les \( b_i \) sont des réels qu'on ne cherchera pas à déterminer.

                Montrer que la dérivée \( n \)-ème en zéro, notée \( \phi_n(z) \), de la fonction \( t\mapsto t \frac{e^{zt}-1}{e^t-1} \) est un polynôme en \( z \) de degré \( n \) et que
                \[
                  \phi_n(z) = z^n - \frac{1}{2}nz^{n-1} + C_n^2 b_1 z^{n-2}
                  + C_n^4 b_2 z^{n-4} + \dots + C_n^{2N} b_N z^{n-2N}
                \]
                où \( N = E(\frac{n-1}{2}) \), \( E \) désignant la fonction partie entière.

          \item Prouver que \( nz^{n-1} = \phi_n(z+1) - \phi_n(z) \)
        \end{enumerate}

  \item Prouver que
        \[
          \begin{matrix}
            (i)   & \phi_n^{(n-k)}(1) = \phi_n^{(n-k)}(0)\ \ (2\le k\le n)      & \quad & (ii) & \phi_n^{(n-2k-1)}(0) = 0 \ \ (1\le k\le N) \\
            (iii) & \phi_n^{(n-2k)}(0) = \frac{n! b_k}{(2k)!} \ \ (1\le k\le N) & \quad & (iv) & \phi_n^{(n-1)}(0) = -\frac{1}{2} n!        \\
            (v)   & \phi_n^{(n-1)}(1) = +\frac{1}{2} n!                         & \quad & (vi) & \phi_n^{(n)} = n!
          \end{matrix}
        \]

  \item

        \begin{enumerate}
          \item On suppose \( f \) de classe \( C^{2n+1} \). Prouver que
                \begin{eqnarray*}
                  0 &=& f(z) - f(a) - \frac{z-a}{2} \left[ f'(z) + f'(a) \right]
                  \\
                  &&+ \sum_{m=1}^{n-1} b_m \frac{(z-a)^{2m}}{(2m)!} \left[ f^{(2m)}(z) - f^{(2m)}(a) \right]
                  \\
                  && - \frac{(z-a)^{2n+1}}{(2n)!} \int_0^1 \phi_{2n}(t) f^{(2n+1)}\Big( a+(z-a)t \Big) dt
                \end{eqnarray*}

          \item En déduire que si \( F \) est de classe \( C^{2n} \) sur \( [a,a+r\omega] \) où \( r\in \Nn \) et \( \omega>0 \), alors
                \begin{eqnarray*}
                  \int_a^{a+r\omega} F(x)dx &=&
                  \omega \Big[ \frac{1}{2} F(a) + F(a+\omega) + \dots + F(a+{\scriptstyle (r-1)}\omega) + \frac{1}{2} F(a+r\omega) \Big]
                  \\
                  && - \sum_{m=1}^{n-1} b_m \frac{\omega^{2m}}{(2m)!} \left[ F^{(2m-1)}(a+r\omega) - F^{(2m-1)}(a) \right] + R_n
                \end{eqnarray*}
                où
                \[
                  R_n = \frac{\omega^{2n+1}}{(2n)!} \int_0^1 \phi_{2n}(t)
                  \sum_{m = 0}^{r-1} F^{(2n)}(a+m\omega+\omega t) dt \; .
                \]
        \end{enumerate}
\end{enumerate}

B

\begin{enumerate}
  \item Soit \( u_k: x>0 \mapsto \ln(x+k)-\ln(k)+x\ln\left(\frac{k}{k+1}\right)
        \qquad(k\in \Nn^*) \)\par Montrer que pour tout \( x \) strictement positif, la série \( \sum_{k\ge 1}u_k(x) \) est convergente. On pose pour la suite \( G(x) = \ln(x)+\sum_{k = 1}^{\infty}u_k(x) \)

  \item Prouver que \( G \) vérifie l'équation fonctionnelle
        \[
          \forall x>0 \quad G(x+1) = G(x)-\ln(x).
        \]

  \item En déduire que \( \forall m\in \Nn \quad \exp(-G(m+1)) = m! \)

  \item Soit \( x \) et \( y \) deux réels strictement positifs. Montrer que la série\par
        \centerline{\( \sum_{k\ge1}\bigl[ \ln(y+k) - \ln(x+k) + (y-x)\ln\left(\frac{k}{k+1}\right) \bigr] \)}
        est convergente et que sa somme est \( G(y) - G(x) - \ln y + \ln x \).

  \item Prouver à l'aide de A que pour tous entiers strictement positifs \( n \) et \( p \)
        \begin{eqnarray*}
          \sum_{k=0}^n \ln(y+k) - \ln(x+k) &=&
          \int_0^n f(t)dt + \frac{1}{2} \big( f(0)+f(n) \big)
          \\
          &&+\sum_{h=1}^{p-1} \frac{b_h}{(2h)!} \big( f^{(2h-1)}(n) - f^{(2h-1)}(0) \big) + T_{p,n}(x,y)
        \end{eqnarray*}
        où \( f: t\mapsto \ln(y+t) -\ln(x+t) \) et \( T_{p,n}(x,y) \) est une expression que l'on précisera.

  \item Prouver que \( R_p(x,y) = \lim_{n \to +\infty}T_{p,n}(x,y) \) existe.

  \item On pose \( g(z) = z\ln z - z - \frac{1}{2} \ln z + \sum_{h = 1}^{p-1}
        \frac{b_h}{(2h)(2h-1)} \frac{1}{z^{2h-1}} \). Montrer que \( G(y)+g(y) = G(x)+g(x)+R_p(x,y) \)

  \item Montrer que \( R_p(x,y) = O\left( \frac{1}{\bigl[ \inf(x,y) \bigr]^{2p-1}} \right) \) quand \( \inf(x,y)\to +\infty \).

  \item Prouver à l'aide de la formule de Stirling que \( G(m)+g(m)\to \frac{1}{2}\ln2\pi \) quand \( m \to +\infty \).

  \item Montrer que
        \[
          G(y) = -y\ln y + y + \frac{1}{2}\ln y - \frac{1}{2}\ln2\pi - \sum_{h = 1}^{p-1} \frac{b_h}{(2h)(2h-1)} \frac{1}{y^{2h-1}}
          + O(\frac{1}{y^{2p-1}})
        \]

  \item Donner un développement asymptotique de \( \ln(m!) \) quand \( m \) tend vers \( +\infty \) à un \( O(\frac{1}{m^7}) \) près.
\end{enumerate}
\finenonce

Effectuons le changement d'indice de sommation \( m = m'+1 \) dans la deuxième somme; tous les termes s'éliminent deux à deux, à l'exception du premier terme de la première somme et du dernier terme de la deuxième somme, d'où le résultat demandé.

\medskip 2.a. Plus généralement, on a le résultat suivant: si une fonction \( f \) est nulle en zéro et de classe \( C^{n+1} \) sur un intervalle \( I \) contenant zéro, \( g(t) = f(t)/t \) est prolongeable par continuité en zéro et son prolongement est de classe \( C^n \). Gardons nous de déduire fallacieusement ce résultat de l'existence d'un développement limité d'ordre \( n \) de \( g \). On montre à l'aide d'un développement limité d'ordre 1 de \( f \) que \( g \) est prolongeable par continuité en \( 0 \) en posant \( \tilde g(0) = f'(0) \). Par ailleurs, \( g \) est de classe \( C^{n+1} \) sur \( I\backslash\{0\} \). Faisons l'hypothèse de récurrence \( g^{(k-1)}(t) = \frac{f^{(k)}(0)}{k} + \frac{f^{(k+1)}(0)}{k+1} \frac{t}{1!}
+ \dots + \frac{f^{(n+1)}(0)}{n+1} \frac{t^{n-k+1}}{(n-k+1)!}+o(t^{n-k+1}) \) \( (1\le k \le n,\ t\neq0) \). En dérivant \( k \) fois l'identité \( f(t) = tg(t) \), on obtient que \( \forall t\neq0 \ \ g^{(k)}(t) = \frac{f^{(k)}(t) - k g^{(k-1)}(t)}{t} \). Or \( f^{(k)}(t) = f^{(k)}(0)+f^{(k+1)}(0)\frac{t}{1!}+\dots +f^{(n+1)}(0)\frac{t^{n+1-k}}{(n-k)!}+o(t^{n+1-k}) \). En substituant ces développements limités dans l'identité précédente, on montre que l'hypothèse de récurrence est vérifiée au rang suivant, donc pour tout entier \( k \) de 1 à \( n+1 \). Faisons l'hypothèse de récurrence que \( g^{(k)}(0) \) existe et est égale à \( f^{(k+1)}(0)/(k+1) \) (\( 0\le k\le n \)). Du développement limité de \( g^{(k)} \) (tronqué à l'ordre 1) et de l'hypothèse de récurrence, il résulte que \( g^{(k)} \) est continue et dérivable en zéro et que, si \( k<n \), \( g^{(k+1)}(0) = f^{(k+2)}(0)/(k+2) \), ce qui prouve par récurrence que \( g \) est \( n \) fois continument dérivable sur \( I \).

Par conséquent, \( t\mapsto\frac{e^t-1}{t} \) est prolongeable par continuité en une fonction indéfiniment dérivable sur \( \Rr \). Cette fonction ne s'annulant jamais, son inverse est également définie et indéfiniment dérivable sur \( \Rr \).

La fonction \( t \mapsto \frac{t}{e^t-1} + \frac{t}{2} \) est paire car
\[
  \begin{aligned}
    \frac{-t}{e^{-t}-1} + \frac{-t}{2}
     & = \frac{-t e^t}{1-e^t} - \frac{t}{2}        \\
     & = \frac{t e^t - t + t}{e^t-1} - \frac{t}{2} \\
     & = \frac{t}{e^t-1} + \frac{t}{2}             \\
  \end{aligned}
\]
Donc le développement limité de \( \frac{t}{e^t-1} \), dont l'existence est garantie par le fait que la fonction est indéfiniment dérivable, est de la forme demandée par l'énoncé.

Par conséquent
\begin{eqnarray*}
  \frac{t}{e^t-1} (e^{zt}-1) &=&
  \Big( 1-\frac{t}{2}+ \frac{b_1 t^2}{2!} + \frac{b_2 t^4}{4!} + \dots + \frac{b_N t^{2N}}{(2N)!} + o(t^n) \Big)
  \\
  \times \Big( zt + \frac{z^2 t^2}{2!} + \dots + \frac{z^n t^n}{n!} + o(t^n) \Big)
\end{eqnarray*}
et \( \phi_n(z)/n! \) est le coefficient de \( t^n \) dans ce développement:
\[
  \phi_n(z)/n!= \frac{z^n}{n!} - \frac{1}{2} \frac{z^{n-1}}{(n-1)!}
  + \frac{b_1}{2!}\frac{z^{n-2}}{(n-2)!} + \frac{b_2}{4!}\frac{z^{n-4}}{(n-4)!}
  + \dots + \frac{b_N}{(2N)!}\frac{z^{n-2N}}{(n-2N)!},
\]
d'où l'expression de \( \phi_n \) demandée.
