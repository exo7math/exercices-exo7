\exercice{2926, quercia, 2010/03/08}

\enonce[Transformation homographique]
Soit \( f : {\Cc\setminus\{i\}} \to {\Cc\setminus\{1\}}, z \mapsto {\frac{z+i}{z-i}} \)
\begin{enumerate}
  \item Montrer que \( f \) est bijective.

  \item Déterminer \( f(\Rr) \), \( f(\mathbb{U}\setminus\{i\}) \), \( f(i\Rr\setminus\{i\}) \).
\end{enumerate}
\finenonce

\noindication

\correction
\begin{enumerate}
  \item

  \item \( \mathbb{U}\setminus\{1\} \), \( i\Rr \), \( \Rr\setminus\{1\} \).
\end{enumerate}
\fincorrection

\finexercice
