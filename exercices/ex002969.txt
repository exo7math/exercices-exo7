\exercice{2969, quercia, 2010/03/08}

\enonce[Transport de structure]
Pour tout \( x,y \in \Rr \), on pose \( x
\star y = x \sqrt{1+y^2} + y \sqrt{1+x^2} \).
\begin{enumerate}
  \item Vérifier que \( \sqrt{1+(x\star y)^2} = \sqrt{1+x^2}\sqrt{1+y^2} + xy \).

  \item Montrer que \( (\Rr, \star) \) est un groupe.

  \item Montrer que l'application \( \sh \) est un isomorphisme entre \( (\Rr,+) \) et \( (\Rr, \star) \).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
