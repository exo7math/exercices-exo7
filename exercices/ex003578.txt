\exercice{3578, quercia, 2010/03/10}

\enonce[Trigonalisation de matrices]
Soit \( A = \begin{pmatrix} -1 & \phantom-2 & 0 \\ 2 &2 &-3 \\ -2 &2 &1
\end{pmatrix} \) et \( \varphi \) l'endomorphisme de \( \Rr^3 \) canoniquement associé à \( A \).

\begin{enumerate}
  \item Vérifier que \( A \) n'est pas diagonalisable.

  \item Chercher deux vecteurs propres de \( A \) linéairement indépendants.

  \item Compléter ces vecteurs en une base de \( \Rr^3 \).

  \item Écrire la matrice de \( \varphi \) dans cette base.

  \item Résoudre le système différentiel : \( X' = AX \).

\end{enumerate}
\finenonce

\noindication

\correction
\begin{enumerate}
  \item 1 est valeur propre double, \( d_1 = 1 \).

  \item \( \begin{pmatrix} 1 \\ 1 \\ 1\end{pmatrix} \), \( \begin{pmatrix} 2 \\ 1 \\ 2\end{pmatrix} \).

  \item \( \begin{pmatrix} 1 \\ 0 \\ 0\end{pmatrix} \).

  \item \( \begin{pmatrix} 1 & 0 & 6 \\ 0 &0 &-4 \\ 0 &0 &1
        \end{pmatrix} \).

  \item \( X = \begin{pmatrix} (6\alpha t + \gamma)e^t + 2\beta          \\
          (6\alpha t + \gamma + 3\alpha)e^t + \beta \\
          (6\alpha t + \gamma - \alpha)e^t + 2\beta
        \end{pmatrix} \).
\end{enumerate}
\fincorrection

\finexercice
