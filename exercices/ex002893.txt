\exercice{2893, quercia, 2010/03/08}

\enonce[Propriétés des applications \( A \mapsto f(A) \) et \( B \mapsto f^{-1}(B) \)]
Soit \( f: E\to F \). On considère les applications
\[
  \Phi: {{\cal P}(E)}\to {{\cal P}(F)}, A \mapsto{f(A)}
  \quad \text{ et } \quad
  \Psi :{{\cal P}(F)}\to {{\cal P}(E)}, B \mapsto {f^{-1}(B)}.
\]
Montrer que :
\begin{tabular}{rlll}
  \textbf{1.} & \( f \) est injective  & \( \iff \Phi \) est injective  & \( \iff \Psi \) est surjective. \\
  \textbf{2.} & \( f \) est surjective & \( \iff \Phi \) est surjective & \( \iff \Psi \) est injective.  \\
\end{tabular}
\finenonce

\noindication

\nocorrection

\finexercice
