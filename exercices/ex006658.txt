\exercice{6658, queffelec, 2011/10/16}

\enonce
\begin{enumerate}
  \item Montrer que \( \Re(\cos z)>0 \) si \( |\Re z|<\frac{\pi}{2} \). En déduire une détermination holomorphe du logarithme de \( \cos z \) dans \( \{|\Re z|<\frac{\pi}{2}\} \).

  \item Montrer que l'on peut définir une fonction holomorphe \( f(z) = \hbox{Log}
        \frac{1+iz}{1-iz} \) sur l'ouvert \( U = \Cc\backslash S \) où \( S = \{ix\ ;\ |x|\ge1\} \).

  \item Montrer que l'on peut définir une fonction holomorphe \( f(z) = \hbox{Log}
        \sqrt{z^3-1} \) sur un ouvert \( U \) à déterminer (où \( \sqrt{} \) désigne la détermination principale de la racine).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
