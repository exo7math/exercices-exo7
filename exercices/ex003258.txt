\exercice{3258, quercia, 2010/03/08}

\enonce[\( a+b+c \), \( a^2+b^2+c^2 \), \( 1/a+1/b+1/c \) donnés]
Résoudre
\[
  \left\{\begin{aligned} a        & + & b        & + & c        & = & 0   \\
                a^2      & + & b^2      & + & c^2      & = & 1   \\
                \frac 1a & + & \frac 1b & + & \frac 1c & = & -1. \\\end{aligned}\right.
\]
\finenonce

\noindication

\correction
\( \{ a,b,c \} = \left\{ 1, -\frac{1+i}2, -\frac{1-i}2 \right\} \).
\fincorrection

\finexercice
