\exercice{3585, quercia, 2010/03/10}

\enonce[Ensi PC 1999]
Soit \( A = \begin{pmatrix}-1 & 2 & 1 \\2&-1&-1 \\-4&4&3
\end{pmatrix} \).
\begin{enumerate}
  \item Calculer \( A^n \).

  \item Soit \( U_0 = \begin{pmatrix}-2 \\4 \\1\end{pmatrix} \) et \( (U_n) \) définie par la relation : \( U_{n+1} = AU_n \). Calculer \( U_n \) en fonction de \( n \).

  \item Soit \( X(t) = \begin{pmatrix}x(t) \\ y(t) \\ z(t)
        \end{pmatrix} \). Résoudre \( \frac{d X}{d t} = AX \).
\end{enumerate}
\finenonce

\noindication

\correction
\begin{enumerate}
  \item \( A^{2k} = I \), \( A^{2k+1} = A \).
\end{enumerate}
\fincorrection

\finexercice
