\exercice{4284, quercia, 2010/03/12}

\enonce[Polynôme\( \times e^{-t} \)]
Soit \( \varphi : {\Rr_n[X]} \to {\Rr^{n+1}}, P \mapsto {(a_0,\dots,a_n)} \) avec \( a_k = \int_{t = 0}^{+\infty} e^{-t}t^kP(t)\,d t \).

\begin{enumerate}
  \item Justifier l'existence de \( \varphi \).

  \item Montrer que \( \varphi \) est un isomorphisme d'espaces vectoriels.
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
