\exercice{3251, quercia, 2010/03/08}

\enonce[Polynômes irréductibles sur \( \Zz \)]
Soient \( a_1, \dots, a_n \in \Zz \) distincts.
\begin{enumerate}
  \item Montrer que \( (X-a_1) \dots (X-a_n) - 1 \) est irréductible dans \( {\Zz[X]} \).

  \item Même question avec \( (X-a_1) \dots (X-a_n) + 1 \), \( n \) impair.
\end{enumerate}
\finenonce

\noindication

\correction
\begin{enumerate}
  \item Si \( P = QR \) alors \( Q(a_i)R(a_i) = -1 \Rightarrow Q(a_i) = -R(a_i) = \pm1 \), donc \( Q+R \) a \( n \) racines, donc est nul, et \( P = -Q^2 \): contradiction pour \( x \to \infty \).

  \item Même raisonnement : \( P = Q^2 \), donc \( Q^2-1 = (Q-1)(Q+1) = (X-a_1) \dots (X-a_n) \).
        \par On répartit les facteurs entre \( Q-1 \) et \( Q+1 \) : \( n = 2p \), contradiction.
\end{enumerate}
\fincorrection

\finexercice
