\exercice{6470, drutu, 2011/10/16}

\enonce
Démontrer les propriétés suivantes

\begin{enumerate}
  \item \( (\Cc / \Rr ,+) \simeq (\Rr,+) \) ;

  \item \( \Cc^*/\Rr_+^*\simeq \mathbb{S}^1 \) ;

  \item \( \Cc^*/\mathbb{S}^1\simeq \Rr_+^*\simeq \Rr^*/\{ \pm 1 \} \) ;

  \item \( \Rr^*/ \Rr_+^*\simeq \{ \pm 1 \} \) ;

  \item \( (\Cc / \Zz ,+)\simeq \Cc^* \) ;

  \item \( \Cc^* /\Rr^* \simeq \mathbb{S}^1/\{ \pm 1 \} \).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
