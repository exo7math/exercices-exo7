\exercice{3828, quercia, 2010/03/11}

\enonce[Calcul de minimums]
\begin{enumerate}
  \item Soit \( \varphi : \Rr^n \to \Rr \) définie par \( \varphi(x_1,\dots,x_n) = \int_{t = 0}^1 (1+t x_1+\dots+t^n x_n)^2\,d t \). 
  Montrer que \( \varphi \) admet un minimum absolu et le calculer lorsque \( n = 3 \).

  \item Même question avec \( \psi(x_1,\dots,x_n) = \int_{t = 0}^{+\infty} e^{-t}(1+t x_1+\dots+t^n x_n)^2\,d t \).

\end{enumerate}
\finenonce

\noindication

\correction
\begin{enumerate}
  \item \( \frac1{16} \).

  \item \( \frac14 \).
\end{enumerate}
\fincorrection

\finexercice
