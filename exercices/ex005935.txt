\exercice{5935, tumpach, 2010/11/11}

\enonce
Soit \( \Omega = \Nn \), \( \Sigma = \mathcal{P}(\Nn) \) et \( \mu \) la mesure de comptage sur \( \Nn \) définie par :
\[
  \mu(E) = \sharp E = \sum_{k \in E} 1,
\]
où \( E\in \Sigma \). Soit \( f :\Nn \rightarrow \Rr \) une fonction positive ou nulle. Montrer que \( f \) est (\( \Sigma \)-\( \mathcal{B}(\Rr) \))-mesurable et que :
\[
  \int_{\Omega} f d\mu = \sum_{n = 1}^{\infty} f(n).
\]
\finenonce

\noindication

\correction
Soit \( \Omega = \Nn \), \( \Sigma = \mathcal{P}(\Nn) \) et \( \mu \) la mesure de comptage sur \( \Nn \) définie par :
\[
  \mu(E) = \sharp E = \sum_{k \in E} 1,
\]
où \( E\in \Sigma \). Soit \( f :\Nn \rightarrow \Rr \) une fonction positive ou nulle. Pour tout borélien \( E \), \( f^{-1}(E) \) appartient à \( \mathcal{P}(\Nn) \), donc \( f \) est (\( \Sigma \)-\( \mathcal{B}(\Rr) \))-mesurable. Par définition de l'intégrale,
\[
  \int_{\Omega} f d\mu = \int_0^{\infty}
  \mu\left(S_f(t)\right)\,dt,
\]
où \( S_f(t) = \{n\in\Sigma, \,f(n)>t\} \). Pour tout \( y\in[0,+\infty[ \), posons \( A_y := \{n\in\Nn,\, f(n) = y\} \). Alors
\[
  S_f(t) = \cup_{y > t}A_y
\]
où l'union est disjointe et où \( A_y \) est vide sauf pour un ensemble dénombrable \( \{y_i\}_{i\in\Nn} \) de valeurs de \( y \). Par \( \sigma \)-additivité de la mesure \( \mu \),
\[
  \mu\left(S_f(t)\right) = \mu\left(\cup_{y_i > t} A_{y_i}
  \right) = \sum_{y_i > t} \mu\left(A_{y_i}\right) = \sum_{y_i > t}\mu\left(\{f = y_i
  \}\right).
\]
Ainsi :
\begin{eqnarray*} \int_{\Omega} f d\mu & =&
  \int_0^{\infty}\sum_{y_i > t} \mu\left(\{f = y_i \}\right)\,dt = \sum_{i=0}^{\infty}
  \int_{0\le t<y_i}\mu\left(\{f = y_i \}\right)\,dt \\&=&
  \sum_{i=0}^{\infty} y_i \cdot \mu\left(\{f = y_i \}\right) =
  \sum_{i=0}^{\infty} y_i\cdot\sharp\{n\in\Nn, f(n) = y_i\} = \sum_{n=0}^{\infty} f(n).
\end{eqnarray*}
\fincorrection

\finexercice
