\exercice{6745, queffelec, 2011/10/16}

\enonce[Utilisation du logarithme]
\begin{enumerate}
  \item Calculer \( \displaystyle\int_0^{+\infty}\frac{dx}{1+x^3} \), en intégrant \( \displaystyle\frac{\log z}{1+z^3} \) sur un cercle privé de \( \Rr^+ \), \( \hbox{log} \) désignant ici la détermination du logarithme avec coupure sur \( \Rr^+ \).

  \item En choisissant la même détermination du logarithme et le même contour, calculer simultanément les intégrales \( I = \displaystyle\int_\Rr \frac{dx}{1+x^4} \) et \( J = \displaystyle\int_\Rr\frac{\hbox{Ln} x}{1+x^4}\ dx \). (Intégrer cette fois \( \displaystyle\frac{(\log z)^2}{1+z^4} \).)

  \item Calculer \( \displaystyle\int_0^{+\infty}\frac{dx}{1+x^n} \) et \( \displaystyle\int_0^{+\infty}\frac{\hbox{Ln} x}{1+x^n}\ dx \), \( n\ge2 \), en intégrant \( \displaystyle\frac{\log z}{1+z^n} \) sur un secteur épointé bien choisi.
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
