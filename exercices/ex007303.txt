\exercice{7303, mourougane, 2021/08/10}

\enonce
\begin{enumerate}
  \item Montrer que \( \Zz \times \Zz \) peut être muni d'une structure naturelle d'anneau.

  \item Montrer que \( \Rr[X] \) est un anneau, et que pour tout \( a \in \Rr \), l'ensemble
        \[
          I_a = \{P \in \Rr[X]  /  P(a) = 0\}
        \]
        est un idéal de \( \Rr[X] \).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
