\exercice{207, bodin, 1998/09/01}

\enonce
\begin{enumerate}
  \item Soit \( E = \Nn\times\Nn \), on définit \( \mathcal{R} \) par : \( (a,b)\mathcal{R}(a',b')\Leftrightarrow a+b' = b+a' \). Montrer que \( \mathcal{R} \) est une relation d'équivalence. Identifier \( E/\mathcal{R} \).

  \item Mêmes questions avec \( E = \Zz\times\Nn^* \) et \( (p,q)\mathcal{R}(p',q') \Leftrightarrow pq' = p'q \).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
