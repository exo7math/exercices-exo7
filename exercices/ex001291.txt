\exercice{1291, gourio, 2001/09/01}

\enonce
Convergence et calcul de :
\[
  \int_0^1\frac{\ln (1+t^2)dt}{t^2}, \qquad
  \int_0^{\infty }\ln \left( 1+\frac{1}{t^2}\right) dt, \qquad
  \int_1^{\infty }\frac{\ln t}{t^n}dt.
\]
\finenonce

\noindication

\correction
Réponses : \( \frac{\pi}{2}-\ln 2,\pi ,\frac{1}{\left( n-1\right)^2} \).
\fincorrection

\finexercice
