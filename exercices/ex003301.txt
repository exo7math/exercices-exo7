\exercice{3301, quercia, 2010/03/09}

\enonce[Étude de liberté]
Étudier la liberté des familles suivantes :
\begin{enumerate}
  \item \( E = \{ {\text{fcts}} : \Rr \to \Rr \} \), \( {\cal F} = (\sin, \cos) \).

  \item \( E = \{ {\text{fcts}} : {\Rr^{+*}} \to \Rr \} \), \( {\cal F} = (f_a : x \longmapsto x^a),\ a\in\Rr \).

  \item \( E = \{ {\text{fcts}} : \Rr \to \Rr \} \), \( {\cal F} = (f_a : x \longmapsto |x-a|),\ a\in\Rr \).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
