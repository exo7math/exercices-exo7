\exercice{3503, quercia, 2010/03/10}

\enonce[Diagonalisation]
Diagonaliser \( M = \begin{pmatrix} (0) &  & 1 \\ &\cdots \\ 1& &(0)
\end{pmatrix}
\in \mathcal{M}_n(K) \).
\finenonce

\noindication

\correction
\( n \) pair : \( P = \begin{pmatrix}1 &        &   &    &        & 1  \\
                 & \ddots &   &    & \cdots      \\
                 &        & 1 & 1                \\
                 &        & 1 & -1               \\
                 & \cdots &   &    & \ddots      \\
               1 &        &   &    &        & -1
\end{pmatrix} \), \( D = \text{diag}(1,\dots,1,-1,\dots,-1) \).\par \( n \) impair : \( P = \begin{pmatrix}1 &        &   &   &    &        & 1  \\
                 & \ddots &   &   &    & \cdots      \\
                 &        & 1 &   & 1                \\
                 &        &   & 1 &                  \\
                 &        & 1 &   & -1               \\
                 & \cdots &   &   &    & \ddots      \\
               1 &        &   &   &    &        & -1
\end{pmatrix} \), \( D = \text{diag}(1,\dots,1,-1,\dots,-1) \).
\fincorrection

\finexercice
