\exercice{243, bodin, 1998/09/01}

\enonce
\begin{enumerate}
  \item ({\it principe des bergers}) Soient \( E, F \) deux ensembles avec \( F \) un ensemble fini, et \( f \) une surjection de \( E \) sur \( F \) vérifiant :
        \[
          \forall y\in F,\ \mathrm{Card}(f^{-1}({y})) = p
        \]
        Montrer que \(E\) est alors un ensemble fini et \( \mathrm{Card}(E) = p\mathrm{Card}(F) \).

  \item ({\it principe des tiroirs}) Soient \( \alpha_1,\alpha_2,\ldots,\alpha_p \), \( p \) éléments distincts d'un ensemble \( E \), répartis entre une famille de \( n \) sous-ensembles de \( E \). Si \( n<p \) montrer qu'il existe au moins un ensemble de la famille contenant au moins deux éléments parmi les \( \alpha_i \) (on pourra raisonner par l'absurde).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
