\exercice{6532, drutu, 2011/10/16}

\enonce
\begin{enumerate}
  \item Soit \( a\in \Zz \) un nombre sans facteurs carrés, \( a\not\in \{ -1,\; 0,\; 1\} \). Montrer que \( X^n-a \) est irréductible sur \( \Qq \) pour tout \( n\in \Nn^* \).

  \item Montrer que \( 2X^{10}-21 \), \( 3X^5-35 \), \( X^5+1000X^2+6 \) sont irréductibles sur \( \Qq \).

  \item Démontrer l'irréducibilité dans \( \Qq[X] \) des polynômes \( X^4-8X^3+12X^2-6X+2 \) et \( X^5-12X^3+36X-12 \).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
