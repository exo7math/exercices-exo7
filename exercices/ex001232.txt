\exercice{1232, legall, 1998/09/01}

\enonce
\begin{enumerate}
  \item Montrer que pour tout \( x>0 \) on a : \( \displaystyle{\frac{1}{x+1}<\hbox{log}(x+1) -\hbox{log}(x)<\frac{1}{x}} \).

  \item En déduire que pour tout entier \( n\ge 1 : \) \( \displaystyle {
          \hbox{log}(n+1)< 1+ \frac{1}{2}+\cdots +\frac{1}{n}<1+\hbox{log}(n)} \).

  \item Posons \( \displaystyle{u_n = 1+ \frac{1}{2}+\cdots +\frac{1}{n}-\hbox{log}(n)} \) Montrer que la suite \( (u_n)_{n \in \Nn} \) est décroissante et convergente.
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
