\exercice{3209, quercia, 2010/03/08}

\enonce[pgcd\( (X^n-1,X^m-1) \)]
Soient \( m,n \in \Nn^* \). Chercher \( \pgcd(X^n-1,X^m-1) \).
\finenonce

\noindication

\correction
\( n = qm+r \Rightarrow X^n-1 \equiv X^r-1 (\mathrm{mod}\,{X^m-1}) \). On applique la méthode des divisions euclidiennes entre \( n \) et \( m \) \( \Rightarrow \) \hbox{pgcd\( {} = X^{n \wedge m} - 1 \).}
\fincorrection

\finexercice
