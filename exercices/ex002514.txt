\exercice{2514, mayer, 2009/04/01}

\enonce
\begin{enumerate}
  \item Que peut-on dire de la différentiabilité de l'application \( f:\Rr^2 \to \Rr \) définie par \( f(x_1,x_2) = \|x\|_\infty = \max{(|x_1|, |x_2|)} \)? \item Généraliser ceci à \( f:{\cal F} \to \Rr \), \( f(x) = \|x\|_\infty \), avec \( {\cal F} = \Rr^n \) ou \( {\cal F} \) l'ensemble des suites convergentes vers zéro.
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
