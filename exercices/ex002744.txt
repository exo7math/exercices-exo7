\exercice{2744, tumpach, 2009/10/25}

\enonce
Soit la matrice
\[
  A = \left( \begin{array}{ccc}2 & 7 & 1\\-1 & 2 & 0\\ 3 & 5 & 1 \end{array}\right).
\]

  \item Déterminer une base du noyau de \( A \).

  \item Déterminer une base de l'image de \( A \).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
