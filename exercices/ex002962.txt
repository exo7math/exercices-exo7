\exercice{2962, quercia, 2010/03/08}

\enonce[Opération induite sur les parties]
Soit \( * \) une opération sur \( E \). Pour \( A,B \subset E \), on pose \( A*B = \{a*b \) tq \( a \in A,\ b \in B \} \).
\begin{enumerate}
  \item Étudier les propriétés de \( * \) sur \( {\cal P}(E) \) en fonction de celles de \( * \) sur \( E \) (commutativité, associativité, élément neutre, éléments symétrisables).

  \item Est-ce que \( * \) est distributive par rapport à \( \cup \) ?
\end{enumerate}
\finenonce

\noindication

\correction
\begin{enumerate}
  \item Associative, commutative, \( \{e\} \) = élement neutre, \( A \) est symétrisable \( \iff A = \{a\} \) avec \( a \) symétrisable.

  \item Oui.
\end{enumerate}
\fincorrection

\finexercice
