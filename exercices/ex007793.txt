\exercice{7793, mourougane, 2021/08/11}

\enonce
\begin{enumerate}
  \item Quels sont les sous-espaces propres de la matrice \( J \) de taille \( n\times n \) dont tous les coefficients sont égaux à \( 1 \) ?

  \item Diagonaliser dans une base orthonormée pour le produit scalaire standard, la forme bilinéaire symétrique sur \( \Rr^n \). Discuter son rang et sa signature.
        \[
          f(x,y) = \sum_{{1\le i,j\le n\atop i\not= j}}x_iy_j+x_jy_i.
        \]
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
