\exercice{1865, gineste, 2001/11/01}

\enonce[Normes sur \( \Rr^2 \)]
Pour tout \( (x,y) \in
\Rr^2 \), on pose \( N_1(x,y) = \text{Max} (\sqrt{x^2+y^2},|x-y|) \) et \( N_2(x,y) = \sqrt{ x^2/9 + y^2/4 } \).
\begin{enumerate}
  \item Montrer que \( N_1 \) et \( N_2 \) sont des normes sur \( \Rr^2 \) et représenter les boules unités fermées associées à ces normes.

  \item Montrer que \( N_2 \le \| . \|_\infty \le \| . \|_2 \le N_1 \le \|.\|_1 \le 4N_2 \).

  \item Déterminer le plus petit réel \( k>0 \), tel que \( \|.\|_1 \le kN_2 \). (utiliser Cauchy-Schwarz)
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
