\exercice{343, cousquer, 2003/10/01}

\enonce
\begin{enumerate}
  \item Soit \( p \in \Zz \) un nombre premier. Montrer que si \( a\in \Zz \) n'est pas congru à \( 0 \) modulo \( p \) alors \( p \) ne divise pas \( a \) et donc \( \mbox{pgcd}(a,p) = 1 \).

  \item Soit \( a\in \Zz \) non congru à \( 0 \) modulo \( p \) avec \( p \) premier. Montrer en utilisant le a) qu'il existe \( u \in \Zz \) non congru à \( 0 \) modulo \( p \) vérifiant \( au \equiv 1[p] \). (Remarquer que cela donne un inverse de \( a \) modulo \( p \)).

  \item Montrer que si \( p \) n'est pas premier, il existe des éléments \( a,u\in \Zz \) non nuls modulo \( p \) tels que \( au\equiv 0[p] \).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
