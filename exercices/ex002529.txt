\exercice{2529, tahani, 2009/04/01}

\enonce
Soit \( \varphi \) l'application de \( \Rr^2 \) dans \( \Rr^2 \) définie par
\[
  \varphi(x,y) = (\sin(y/2)-x,\sin(x/2)-y).
\]
\begin{enumerate}
  \item Justifier que \( \varphi \) est de classe \( C^1 \), calculer sa différentielle et voir que \( D\varphi(x,y) \) est inversible pour tout \( (x,y) \in \Rr^2 \). \item Montrer que \( \varphi \) est un \( C^1 \)-difféomorphisme de \( \Rr^2 \) sur \( \varphi(\Rr^2) \) et justifier que \( \varphi(\Rr^2) \) est un ouvert.

  \item Montrer que \( \varphi^{-1} \) est lipschitzienne (on prendra comme norme sur \( \Rr^2 \): \( \|(x,y)\| = |x|+|y| \)).

  \item En déduire que \( \varphi \) est un difféomorphisme de \( \Rr^2 \) sur \( \Rr^2 \)

  \item Calculer \( D\varphi^{-1}(p) \) où \( p = (1-\pi/2, \sqrt{2}/2-\pi) \).
\end{enumerate}
\finenonce

\noindication

\correction
\begin{enumerate}
  \item \( \varphi \) a des coordonnées de classe \( C^1 \), elle l'est donc aussi. On a
        \[
          Jac(\varphi)(x,y) = \left (\begin{array}{cc}
              -1            & 1/2 \cos(y/2) \\
              1/2 \cos(x/2) & -1
            \end{array}\right )
        \]
        On a \( det(Jac(\varphi)(x,y)) = 1-1/4\cos(x/2)\cos(y/2) \ge 3/4 > 0 \). Par conséquent la jacobienne est inversible et \( D\varphi(x,y) \in Isom(\Rr^2,
        \Rr^2) = GL(\Rr^2 \). \item D'après le théorème de l'inverse local, Il suffit de montrer que \( \varphi \) est injective. Supposons \( \varphi(x_1,y_1) = \varphi(x_2,y_2) \), alors \( \sin(y_1/2)-x_1 = \sin(y_2/2)-x_2 \) et \( \sin(x_1/2)-y_1 = \sin(x_2/2)-y_2 \). D'où \( \sin(y_1/2)-\sin(y_2/2) = x_1-x_2 \) et \( \sin(x_1/2)-\sin(x_2/2) = y_1-y_2 \). Or, \( \forall a,b \in
        \Rr, |\sin a -\sin b| \le |a-b| \) (conséquence des accroissements finis appliqué à \( \sin x \)). Donc \( |x_1-x_2|\le |y_1/2-y_2/2| \) et \( |y_1-y_2| \le |x_1/2-x_2/2| \) d'où \( |x_1-x_2|\le 1/4 |x_1-x_2| \Rightarrow x_1 = x_2 \) et \( y_1 = y_2 \). \( \varphi: U \rightarrow F \) est injective. L'ensemble \( f(U) \) est ouvert car il est réunion d'ouverts (d'après thm inverse local). C'est un difféomorphisme en \( U \) et \( \varphi(U) \). \item Soient \( (X_1,y_1), (X_2,Y_2) \in \varphi(\Rr^2) \) avec \( \varphi(x_1,y_1) = (X_1,Y_1) \) et \( \varphi(x_2,y_2) = (X_2,Y_2) \) ou encore \( \varphi^{-1}(X_1,Y_1) = (x_1,y_1) \) et \( \varphi^{-1}(X_2,Y_2) = (x_2,y_2) \). On a
        \[
          ||\varphi^{-1}(X_1,Y_1)-\varphi^{-1}(X_2,Y_2)|| = ||(x_1,y_1)-(x_2,y_2)|| = ||(x_1-x_2,y_1,y_2)|| = |x_1-x_2|+|y_1-y_2|
        \]
        Or \( \sin(y_1/2)-x_1 = X_1 \), \( \sin(y_2/2)-x_2 = X_2 \), \( \sin(x_1/2)-y_1 = Y_1 \) et \( \sin(x_2/2)-y_2 = Y_2 \). Par conséquent
        \[
          x_1-x_2 = \sin(y_1/2)-X_1-\sin(y_2/2)+X_2
        \]
        \[
          y_1-y_2 = \sin(x_1/2)-Y_1-\sin(x_2/2)+Y_2.
        \]
        D'où
        \[
          |x_1-x_2|+|y_1-y_2| \le |X_2-X_1|+|\sin(y_1/2)-\sin(y_2/2)|+ |Y_2-Y_1|+ |\sin(x_1/2)-\sin(x_2/2)|
        \]
        \[
          \le |X_2-X_1|+1/2|y_1-y_2|+|Y_2-Y_1|+1/2|x_1-x_2|
        \]
        d'où
        \[
          |x_1-x_2|+|y_1-y_2| \le 2(|X_2-X_1|+|Y_2-Y_1|) \le 2 ||(X_1,Y_1)-(X_2,Y_2)||.
        \]
        Donc \( \varphi^{-1} \) est lipschitzienne.

  \item Soit \( (X_n,Y_n) \) une suite de cauchy dans \( \varphi(\Rr^2) \), (\( (X_n,Y_n) = \varphi(x_n,y_n) \); \( (x_n,y_n) = \varphi^{-1}(X_n,Y_n) \)). Pour tout \( \epsilon > 0,
        \exists n \in \Nn, p,q \ge n \Rightarrow ||(X_p,Y_p)-(X_q,Y_q)|| < \epsilon \). Par conséquent, \( \forall
        \epsilon > 0 \), \( \exists n \in \Nn; p,q \ge n \Rightarrow ||(x_p,y_p)-(x_q,y_q)||< 2 \epsilon \). La suite \( (x_n,y_n) \) est alors de cauchy dans \( \Rr^2 \), qui est complet. Par conséquent elle converge. Soit \( (x,y) \) sa limite. Comme \( \varphi \) est continue et que \( \lim_n(x_n,y_n) = (x,y) \) alors \( \lim_n \varphi(x_n,y_n) = \varphi(x,y) \). La suite \( (X_n,Y_n) \) est une suite de Cauchy de \( \Rr^2 \). Elle converge. Soit \( (X,Y) \) sa limite, alors \( (X,Y) = \varphi(x,y) \) car \( (X_n,Y_n) \rightarrow (X,Y) \) et \( \varphi(x_n,y_n) \rightarrow \varphi(x,y) \). Donc \( (X,Y)
        \in \varphi(\Rr^2) \). \( \varphi(\Rr^2) \) est alors complet et donc fermé. Comme \( \varphi(\Rr^2 \) est un ouvert fermé et non vide (il contient \( (0,0) = \varphi(0,0) \)) dans le connexe \( \Rr^2 \), on a \( \varphi(\Rr^2) = \Rr^2 \). \item \( p = (1-\pi/2,
        \sqrt{2}/2 - \pi) = \varphi(\pi/2,\pi) = \varphi(q) \) où \( q = (\pi/2,\pi) \). \( \varphi: E \rightarrow F \) est un \( C^1 \)-difféormorphisme donc \( \varphi^{-1} \circ \varphi = Id \) et donc
        \[
          Id = D(\varphi^{-1} \circ
          \varphi)(q) = D\varphi^{-1}(\varphi(q))\circ D\varphi(q).
        \]
        Or \( D\varphi^{-1}(\varphi(q)) = (D\varphi(q))^{-1} \) et donc \( Jac
        \varphi^{-1}(p) = (Jac \varphi (\pi/2,\pi))^{-1} \). Or
        \[
          Jac \varphi(x,y) = \left (
          \begin{array}{cc}
              -1            & 1/2 \cos(y/2) \\
              1/2 \cos(x/2) & -1
            \end{array} \right ).
        \]
        D`où
        \[
          Jac \varphi(\pi/2,\pi) = \left (
          \begin{array}{cc}
              -1         & 0  \\
              \sqrt{2}/4 & -1
            \end{array} \right )
        \]
        et donc
        \[
          Jac \varphi^{-1}(p) =
          \left (
          \begin{array}{cc}
              -1          & 0  \\
              -\sqrt{2}/4 & -1
            \end{array} \right ).
        \]
\end{enumerate}
\fincorrection

\finexercice
