\exercice{7304, mourougane, 2021/08/10}

\enonce
\begin{enumerate}
  \item Déterminer l'ordre de \( \overline{2} \) dans \( (\Zz/10\Zz,+) \).

  \item Montrer que \( \{\overline{0},\overline{3},\overline{6},\overline{9}\} \) est un sous-groupe de \( (\Zz/12\Zz,+) \).

  \item Expliciter un sous-groupe d'ordre \( 6 \) de \( (\Zz/12\Zz,+) \).

  \item Quels sont les ordres possibles des sous-groupes de \( (\Zz/6\Zz,+) \) ?

  \item Soient \( n \) un entier naturel non nul. Le but de cette question est de déterminer tous les sous-groupes de \( \Zz/n\Zz \). Soit \( d \) un diviseur de \( n \).
        \begin{enumerate}
          \item Expliciter un sous-groupe \( G_d \) de \( \Zz/n\Zz \) d'ordre \( d \).

          \item Soit \( H \) un sous-groupe de \( \Zz/n\Zz \) d'ordre \( d \). Montrer que pour tout \( \overline x\in H \), \( d\cdot \overline x = 0 \). Combien d'éléments de \( \Zz/n\Zz \) vérifient cette équation ? En déduire que \( H = G_d \).

          \item Conclure.
        \end{enumerate}

  \item Donner la liste de tous les sous-groupes de \( (\Zz/6\Zz,+) \).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
