\exercice{3462, quercia, 2010/03/10}

\enonce[Calcul de rang]
Chercher \( \mathrm{rg}\begin{pmatrix}1 & -1 & 0  & 1  \\
               m & 1  & -1 & -1 \\
               1 & -m & 1  & 0  \\
               1 & -1 & m  & 2
\end{pmatrix} \) en fonction de \( m\in\Cc \).
\finenonce

\noindication

\nocorrection

\finexercice
