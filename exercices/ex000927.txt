\exercice{927, cousquer, 2003/10/01}

\enonce
\emph{Notations :\\
  \( \mathcal{C} \): ensemble des fonctions numériques continues sur \( [0,1] \).\\
  \( \mathcal{C}_d \): ensemble des fonctions numériques ayant une dérivée continue sur \( [0,1] \).\\
  \( \mathcal{C}(\Rr) \) et \( \mathcal{C}^1(\Rr) \): définis de façon analogue pour les fonctions définies sur \( \Rr \).\\
  \( \mathcal{P} \): ensemble des polynômes sur \( \Rr \).\\
  \( \mathcal{P}_n \): ensemble des polynômes sur \( \Rr \), de degré \( \le n \).}

\medskip

Dire si les applications suivantes sont des applications linéaires :

\begin{enumerate}

  \item \( \Rr \rightarrow \Rr : x\mapsto 2x^2 \).

  \item \( \Rr \rightarrow \Rr : x\mapsto 4x-3 \).

  \item \( \Rr \rightarrow \Rr : x\mapsto \sqrt{x^2} \).

  \item \( \Rr^2 \rightarrow \Rr^2 :(x,y)\mapsto (y,x) \).

  \item \( \mathcal{C}\rightarrow \mathcal{C} : f \mapsto \{t\mapsto \frac{f(t)}{1+t^2}\} \).

  \item \( \mathcal{C} \rightarrow \Rr : f\mapsto f(3/4) \).

  \item \( \mathcal{C} \rightarrow \Rr : f\mapsto f(1/4)-\int_{1/2}^1f(t)\,dt \).

  \item \( \Rr^2 \rightarrow \Rr :(x,y)\mapsto 3x+5y \).

  \item \( \Rr^2 \rightarrow \Rr : (x,y)\mapsto \sqrt{3x^2+5y^2} \).

  \item \( \Rr^2 \rightarrow \Rr : (x,y)\mapsto \sin(3x+5y) \).

  \item \( \Rr^2 \rightarrow \Rr^2 : (x,y)\mapsto (-x,y) \).

  \item \( \Rr^2 \rightarrow \Rr :(x,y)\mapsto xy \).

  \item \( \Rr^2 \rightarrow \Rr : (x,y)\mapsto \frac{x^2 y}{x^2+y^2} \mbox{ si } x^2+y^2 \ne0 \mbox{ et} 0 \mbox{ sinon} \).

  \item \( \mathcal{C} \rightarrow \mathcal{C}_d : f\mapsto \{x\mapsto e^{-x}\int_0^1f(t)\,dt\} \).

  \item \( \mathcal{P} \rightarrow \mathcal{P}_n : A \mapsto \) quotient de \( A \) par \( B \) à l'ordre \( n \) selon les puissances croissantes (\( B \) et \( n \) fixés, avec \( B(0)\ne0 \)).

  \item \( \Rr^2 \rightarrow \Rr^2 : M \mapsto M' \) défini par: \( {\overrightarrow{OM'} = \frac{\overrightarrow{OM}}{\bigl\Vert\overrightarrow{OM}\bigr\Vert} \mbox{ si }
            \overrightarrow{OM}\ne\overrightarrow{0} \mbox{ et } 0 \mbox{ sinon}.} \)

  \item \( \Rr^3 \rightarrow \Rr : M\mapsto \overrightarrow{OM}\cdot \overrightarrow{V} \) où \( \overrightarrow{V} = (4,-1,1/2) \).

  \item \( \Rr \rightarrow \Rr^3 : x\mapsto (2x,x/\pi,x\sqrt{2}) \).

  \item \( \mathcal{C} \rightarrow \Rr : f\mapsto \max_{t\in[0,1]}f(t) \).

  \item \( \mathcal{C} \rightarrow \Rr : f\mapsto \max_{t\in[0,1]}f(t)-\min_{t\in[0,1]}f(t) \).

  \item \( \Rr^2 \rightarrow \Rr^2 : (x,y)\mapsto \) la solution du système d'équations en \( (u,v) \):
        \[
          \left\{
          \begin{array}{rcl}
            3u-v  & = & x  \\
            6u+2v & = & y.
          \end{array}
          \right.
        \]

  \item \( \Rr^2 \rightarrow \Rr^2 : (x,y)\mapsto \) le symétrique de \( (x,y) \) par rapport à la droite d'équation \( x+y-a = 0 \) (discuter selon les valeurs de \( a \)).

  \item \( \Rr^3 \rightarrow \Rr^3 : (x,y,z)\mapsto \) la projection de \( (x,y,z) \) sur le plan \( x+y+z-a = 0 \) parallèlement à \( Oz \) (discuter selon les valeurs de \( a \)).

  \item \( \mathcal{C}_d \rightarrow \mathcal{C} : f\mapsto f' \).

  \item \( \Rr^3 \rightarrow \Rr^2 : (x,y,z)\mapsto (2x-3y+z,x-y+z/3) \).

  \item \( \Rr \rightarrow \mathcal{C}_d : \lambda\mapsto \) la solution de l'équation différentielle \( y'-\frac{y}{x^2+1} = 0 \) valant \( \lambda \) en \( x_0 = 1 \).

  \item \( \mathcal{C}\rightarrow \Rr : f\mapsto \int_0^1\ln(1+\vert f(t)\vert)\,dt \).

  \item \( \Rr \rightarrow \Rr : x\mapsto \) la \( 17 \)-ième décimale de \( x \) (en écriture décimale).

  \item \( \mathcal{C}_d \rightarrow \Rr : f \mapsto f'(1/2)+\int_0^1{f(t)\,dt} \).

  \item \( \Rr \rightarrow \Rr : x\mapsto \ln(3^{x\sqrt{2}}) \).

  \item \( \Rr \times \mathcal{C}(\Rr) \rightarrow \mathcal{C}(\Rr) : (\lambda,f)\mapsto \) la primitive de \( f \) qui vaut \( \lambda \) en \( x_0 = \pi \).

  \item \( \mathcal{C}^1(\Rr) \rightarrow \mathcal{C}(\Rr): f\mapsto \{x\mapsto f'(x)+f(x) \cdot
        \sin x\} \).

\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
