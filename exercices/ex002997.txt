\exercice{2997, quercia, 2010/03/08}

\enonce[Sous-groupes de type fini de \( \Qq \)]
\begin{enumerate}
  \item Soit \( H \) un sous-groupe additif de \( \Qq \) engendré par un nombre fini d'éléments. Montrer que \( H \) est monogène.

  \item Trouver un sous-groupe non trivial de \( \Qq \) qui n'est pas engendré par une famille finie.
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
