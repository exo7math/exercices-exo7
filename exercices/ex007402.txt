\exercice{7402, mourougane, 2021/08/10}

\enonce
\begin{enumerate}
  \item Les groupes \( \Zz/9\Zz \) et \( \Zz/3\Zz\times \Zz/3\Zz \) sont-ils isomorphes ?

  \item Les groupes \( \mathbb{F}_7 \) et \( \Zz/2\Zz\times \Zz/3\Zz \) sont-ils isomorphes ? Justifier.

  \item Les groupes \( (\mathbb{F}_7)^\star \) et \( \Zz/2\Zz\times \Zz/3\Zz \) sont-ils isomorphes ? Justifier.
\end{enumerate}
\finenonce

\noindication

\correction
\begin{enumerate}
  \item Non, car \( \Zz/3\Zz\times \Zz/3\Zz \) ne contient aucun élément d'ordre 9.

  \item Non, car ils n'ont pas le même ordre.

  \item Les groupes \( (\mathbb{F}_7)^\star \) et \( \Zz/2\Zz\times \Zz/3\Zz \) sont-ils isomorphes ? Par le théorème sur le groupe des inversibles d'un corps fini et par le théorème chinois, ils sont tous les deux cycliques d'ordre \( 6 \), donc isomorphes.
\end{enumerate}
\fincorrection

\finexercice
