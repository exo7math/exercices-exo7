\exercice{101, liousse, 2003/10/01}

\enonce[Examen octobre 1999]
On définit une fonction \( f \) de \( \Cc-\{i\} \) dans \( \Cc-\{1\} \) en posant
\[
  f(z) = \frac{z+i}{z-i}.
\]
\begin{enumerate}
  \item On suppose \( z \) réel. Quel est le module de \( f(z) \)?

  \item Trouver les nombres complexes \( z \) tels que \( f(z) = z \).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
