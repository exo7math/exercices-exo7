\exercice{92, cousquer, 2003/10/01}

\enonce
Résoudre dans \( \Rr \) les équations suivantes:
\begin{enumerate}
  \item \( \cos^2(x)-\sin^2(x) = \sin(3x) \).

  \item \( \cos^4(x)-\sin^4(x) = 1 \).
\end{enumerate}
\finenonce

\noindication

\correction
\begin{enumerate}
  \item \( \cos^2(x)-\sin^2(x) = \sin(3x) \) ssi \( x = \pi/2+2k\pi \) ou \( x = \pi/10+2k\pi/5 \), avec \( k\in\Zz \).

  \item \( \cos^4(x)-\sin^4(x) = 1 \) ssi \( x = k\pi \), avec \( k\in\Zz \).
\end{enumerate}
\fincorrection

\finexercice
