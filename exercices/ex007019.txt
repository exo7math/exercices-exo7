\exercice{7019, megy, 2016/08/25}

\enonce
% Bernoulli amélioré
\begin{enumerate}
  \item Soit \( a \in \Rr_+ \). Montrer \( \forall n\in\Nn^*,\: (1+a)^n \ge 1+na + \frac{n(n-1)}{2}a^2 \).

  \item Soit \( (u_n)_{n\in\Nn} \) la suite définie par \( u_n = \frac{3n}{3^n} \). Montrer que pour tout \( n\in\Nn^* \), on a \( 0\le u_n \le\frac{3n}{2n^2+1} \).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
