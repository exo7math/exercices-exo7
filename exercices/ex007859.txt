\exercice{7859, mourougane, 2021/08/11}

\enonce[Groupes d'automorphismes]
\begin{enumerate}
  \item Montrer que les automorphismes du groupe \( (\Zz/n\Zz,+) \) sont obtenus par multiplication par un inversible de \( (\Zz/n\Zz,\times) \).

  \item Décrire un isomorphisme de \( \Zz/10\Zz \) sur \( (\Zz/11\Zz)^\star \).

  \item Montrer que si \( G \) et \( H \) sont deux groupes d'ordre premiers entre eux, alors
        \[
          Aut(G\times H) = Aut(G)\times Aut (H).
        \]

  \item En déduire le groupe des automorphismes de \( \Zz/133\Zz \).

  \item Soit \( p \) un nombre premier et \( n \) un entier naturel non nul. Montrer que
        \[
          Aut((\Zz/p\Zz)^n) = GL(n,\mathbb{F}_p).
        \]

  \item Montrer que \( Aut(\Zz/2\Zz\times\Zz/2\Zz)\to Bij((1,0),(1,1),(0,1)) \) est un isomorphisme.
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
