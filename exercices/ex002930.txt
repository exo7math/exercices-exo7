\exercice{2930, quercia, 2010/03/08}

\enonce[Configuration de points]
Déterminer les nombres \( z \in \Cc \) tels que \( \dots \)
\begin{enumerate}
  \item \( z,z^2,z^4 \) sont alignés.

  \item \( 1,z,z^2 \) forment un triangle rectangle.

  \item \( z,\frac 1z,-i \) sont alignés.
\end{enumerate}
\finenonce

\noindication

\correction
\begin{enumerate}
  \item \( z \in \Rr \) ou \( z \in -\frac 12 + i\Rr \).

  \item \( z \in -1 + i\Rr \) ou \( z \in i\Rr \) ou \( \left|z+\frac 12\right| = \frac 12 \).

  \item \( z\in i\Rr \) ou \( |z-i| = \sqrt2 \).
\end{enumerate}
\fincorrection

\finexercice
