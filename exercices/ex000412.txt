\exercice{412, cousquer, 2003/10/01}

\enonce
Dans \( \Rr[X] \) et dans \( \Cc[X] \), décomposer les polynômes suivants en facteurs irréductibles.
\begin{enumerate}
  \item \( X^3-3 \).

  \item \( X^{12}-1 \).
\end{enumerate}
\finenonce

\noindication

\correction
\begin{enumerate}
  \item \( \left\lbrace\begin{array}{rcl}
          X^3-3 & = & (X-\sqrt[3]{3})(X^2+\sqrt[3]{3}\, X+\sqrt[3]{9})                                                                                 \\
                & = & (X-\sqrt[3]{3}) (X+\frac{\sqrt[3]{3}}{2}- i\frac{\sqrt3\sqrt[3]{3}}{2}) (X+\frac{\sqrt[3]{3}}{2}+ i\frac{\sqrt3\sqrt[3]{3}}{2}).
        \end{array}\right. \)

  \item \( \left\lbrace\begin{array}{rcl}
          X^{12}-1 & = & \textstyle(X-1) (X+1) (X^2+1) (X^2-X+1) (X^2+X+1)\times{} \\
                   &   & \qquad\textstyle(X^2-\sqrt3\,X+1) (X^2+\sqrt3\,X+1)       \\
                   & = & \textstyle(X-1) (X+1) (X- i) (X+ i)\times{}               \\
                   &   & \qquad\textstyle\bigl(X-\frac{1+ i\sqrt3}{2}\bigr)
          \bigl(X-\frac{1- i\sqrt3}{2}\bigr)
          \bigl(X-\frac{-1+ i\sqrt3}{2}\bigr)
          \bigl(X-\frac{-1- i\sqrt3}{2}\bigr)\times{}                              \\
                   &   & \qquad\textstyle\bigl(X-\frac{\sqrt3+ i}{2}\bigr)
          \bigl(X-\frac{\sqrt3- i}{2}\bigr)
          \bigl(X-\frac{-\sqrt3+ i}{2}\bigr)
          \bigl(X-\frac{-\sqrt3- i}{2}\bigr).
        \end{array}\right. \)
\end{enumerate}
\fincorrection

\finexercice
