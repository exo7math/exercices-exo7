\exercice{3388, quercia, 2010/03/09}

\enonce[Classes d'équivalence dans \( \mathcal{M}_{n,1}(\Zz) \)]
\begin{enumerate}
  \item Soit \( M \in \mathcal{M}_n(\Zz) \). Montrer que \( M \in GL_n(\Zz) \) si et seulement si \( |\det M| = 1 \).

  \item Soit \( X = \begin{pmatrix} x_1 \\ \vdots \\ x_n
        \end{pmatrix} \in \mathcal{M}_{n,1}(\Zz) \) et \( d \) le pgcd de \( x_1, \dots, x_n \). Montrer qu'il existe \( A \in GL_n(\Zz) \) telle que \( AX = \begin{pmatrix}d \\ 0 \\ \vdots \\ 0
        \end{pmatrix} \) (par récurrence sur \( n \)).

  \item Soient \( X,Y \in \mathcal{M}_{n,1}(\Zz) \). CNS pour qu'il existe \( A \in GL_n(\Zz) \) telle que \( AX = Y \) ?
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
