\exercice{6458, drutu, 2011/10/16}

\enonce
\begin{enumerate}
  \item Soit \( n\ge 2 \) un nombre entier. Montrer que \( n \) est premier si et seulement si tout groupe d'ordre \( n \) a seulement deux sous-groupes.

  \item Montrer qu'un groupe d'ordre \( p^m,\; m\ge 1 \), où \( p\in \Nn \) est un nombre premier, contient un sous-groupe d'ordre \( p \).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
