\exercice{1305, cousquer, 2003/10/01}

\enonce
\begin{enumerate}
  \item Montrer que les transformations géométriques qui conservent globalement un rectangle forment un groupe. Faire l'étude de ce groupe.

  \item Étudier le groupe \( \Zz/4\Zz \).

  \item Montrer qu'il n'existe que deux sortes de groupes à quatre éléments.
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
