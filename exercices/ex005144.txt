\exercice{5144, rouget, 2010/06/30}

\enonce[I]
Calculer les sommes suivantes :
\begin{enumerate}
  \item (**) \( \sum_{1\le i<j\le n}^{}1 \).

  \item (**) \( \sum_{1\le i,j\le n}^{}j \) et \( \sum_{1\le i<j\le n}^{}j \).

  \item (*) \( \sum_{1\le i,j\le n}^{}ij \).

  \item (***) Pour \( n\in\Nn^* \), on pose \( u_n = \frac{1}{n^5}\sum_{k = 1}^n\sum_{h = 1}^n(5h^4-18h^2k^2+5k^4) \). Déterminer\( \lim_{n\rightarrow +\infty}u_n \) (utiliser les résultats de l'exercice \ref{exo:suprou7}, 2)).
\end{enumerate}
\finenonce

\noindication

\correction
\begin{enumerate}
  \item Soit \( n \) un entier supérieur ou égal à \( 2 \). Parmi les \( n^2 \) couples \( (i,j) \) tels que \( 1\le i,j\le n \), il y en a \( n \) tels que \( i = j \) et donc \( n^2-n = n(n-1) \) tels que \( 1\le i,j\le n \) et \( i\neq j \). Comme il y a autant de couples \( (i,j) \) tels que \( i>j \) que de couples \( (i,j) \) tels que \( i<j \), il y a \( \frac{n(n-1)}{2} \) couples \( (i,j) \) tels que \( 1\le i<j\le n \). Finalement,
        \[
          \sum_{1\le i<j\le n}^{}1 = \frac{n(n-1)}{2}.
        \]

  \item Soit \( n\in\Nn^* \).

        \[
          \sum_{1\le i,j\le n}^{}j = \sum_{j = 1}^n\left(\sum_{i = 1}^nj\right) = \sum_{j = 1}^nnj = n\sum_{j = 1}^nj = n.\frac{n(n+1)}{2}
          = \frac{n^2(n+1)}{2}.
        \]
        Soit \( n \) un entier supérieur ou égal à \( 2 \).

        \begin{align*}
          \sum_{1\le i<j\le n}^{}j & =\sum_{j=2}^n\left(\sum_{i=1}^{j-1}j\right)=\sum_{j=2}^n(j-1)j=\sum_{j=2}^nj^2-\sum_{j=2}^nj \\
                                   & =(\frac{n(n+1)(2n+1)}{6}-1)-(\frac{n(n+1)}{2}-1)=\frac{n(n+1)}{2}(\frac{2n+1}{3}-1)          \\
                                   & =\frac{n(n+1)^2}{6}.
        \end{align*}

  \item Soit \( n\in\Nn^* \).

        \[
          \sum_{1\le i,j\le n}^{}ij = (\sum_{1\le i\le n}^{}i)(\sum_{1\le j\le n}^{}j) = \frac{n^2(n+1)^2}{4}.
        \]

  \item Soit \( n\in\Nn^* \).

        \[
          \sum_{1\le h,k\le n}^{}h^2k^2 = \sum_{h = 1}^n(h^2\sum_{k = 1}^nk^2) = (\sum_{k = 1}^nk^2)(\sum_{h = 1}^nh^2) = \left(\frac{n(n+1)(2n+1)}{6}\right)^2.
        \]
        Comme d'autre part, \( \sum_{h = 1}^nh^4 = \sum_{k = 1}^nk^4 = \frac{n(n+1)(6n^3+9n^2+n-1)}{30} \), on a

        \[
          \sum_{1\le h,k\le n}^{}h^4 = \sum_{h = 1}^n(\sum_{k = 1}^nh^4) = \sum_{h = 1}^nnh^4 = n\sum_{h = 1}^nh^4 = \frac{n^2(n+1)(6n^3+9n^2+n-1)}{30},
        \]
        et bien sûr \( \sum_{1\le h,k\le n}^{}k^4 = \frac{n^2(n+1)(6n^3+9n^2+n-1)}{30} \). Par suite,

        \begin{align*}
          u_n & =\frac{1}{n^5}\left(2.5\frac{n^2(n+1)(6n^3+9n^2+n+14)}{30}-18\frac{n^2(n+1)^2(2n+1)^2}{36}\right) \\
              & =\frac{1}{n^5}(2n^6-2n^6+n^5(\frac{15}{3}-\frac{12}{2})+\mbox{termes de degré au plus}\;4)        \\
              & =-1+\mbox{termes tendant vers}\;0
        \end{align*}

        Par suite,

        \[
          \lim_{n\rightarrow +\infty}u_n = -1.
        \]
\end{enumerate}
\fincorrection

\finexercice
