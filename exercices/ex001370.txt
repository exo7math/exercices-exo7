\exercice{1370, ortiz, 1999/04/01}

\enonce[Bac 1978]
Soit l'anneau \( A = \Zz{/}91\Zz \).
\begin{enumerate}
  \item Déterminer les diviseurs de zéro de l'anneau \( A \).

  \item Résoudre dans \( A \) l'équation \( x^2+\overline{2}x-\overline{3} = %
        \overline{0} \).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
