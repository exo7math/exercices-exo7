\exercice{7327, mourougane, 2021/08/10}

\enonce
\begin{enumerate}
  \item Déterminer la liste des inversibles de \( \Zz/15\Zz \). Déterminer l'ordre de chaque élément dans \( \left(\Zz/15\Zz\right)^\times \).

  \item Déterminer la liste des inversibles de \( \Zz/3\Zz\times \Zz/5\Zz \). Déterminer l'ordre de chaque élément dans \( \left(\Zz/3\Zz\times \Zz/5\Zz\right)^\times \).

  \item Donner un isomorphisme de groupes de \( \left(\Zz/15\Zz\right)^\times \) dans \( \left(\Zz/3\Zz\times \Zz/5\Zz\right)^\times \).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
