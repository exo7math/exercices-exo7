\exercice{1873, maillot, 2001/09/01}

\enonce[partiel 1999]
Soit \( (E,\|\cdot\|) \) un espace vectoriel normé, \( x \) un élément de \( E \) et \( A \) un compact de \( E \).
\begin{enumerate}
  \item Montrer que l'application de \( E \) dans \( \Rr \) qui à \( y \) associe \( \|y\| \) est continue.

  \item Montrer que l'application de \( E \) dans \( \Rr \) qui à \( y \) associe \( \|y-x\| \) est continue.

  \item Montrer que la distance de \( x \) à \( A \) est atteinte, c'est-à-dire qu'il existe \( a\in A \) tel que
        \[
          \inf_{y\in A}\|y-x\| = \|a-x\|.
        \]
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
