\exercice{3186, quercia, 2010/03/08}

\enonce[\( X^n + 1/X^n = P_n(X+1/X) \)]
\begin{enumerate}
  \item Montrer que pour tout entier \( n \in \Nn \) il existe un unique polynôme \( P_n \in {\Zz[X]} \) vérifiant :
        \[
          \forall\ z \in \Cc^*,\ z^n + z^{-n} = P_n(z + z^{-1}).
        \]

  \item Déterminer le degré, le coefficient dominant, et les racines de \( P_n \).

  \item Pour \( P \in {\Cc[X]} \), on note \( \tilde P \) le polynôme tel que :
        \[
          \forall\ z \in \Cc^*,\ P(z) + P(z^{-1}) = \tilde P(z + z^{-1}).
        \]
        Étudier l'application \( P \mapsto \tilde P \).
\end{enumerate}
\finenonce

\noindication

\correction
\begin{enumerate}
  \item \( P_0(u) = 2 \), \( P_1(u) = u \), \( P_{n+1}(u) = uP_n(u) - P_{n-1}(u) \).

  \item \( u_k = 2\cos\Bigl(\frac{(2k+1)\pi}{2n}\Bigr),\ k = 0, \dots, n-1 \).

  \item
\end{enumerate}
\fincorrection

\finexercice
