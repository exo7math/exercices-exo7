\exercice{1468, gourio, 2001/09/01}

\enonce
\begin{enumerate}
  \item Montrer que sur \( M_n(\Rr) \) l'application :
        \[
          \left( A,B\right) \rightarrow tr(^tAB)
        \]
        est un produit scalaire.

  \item Soit \( N \) la norme associée, montrer que :
        \[
          \forall A,B\in M_n(\Rr), N(AB)\le N(A)N(B).
        \]

  \item Montrer que :
        \[
          \forall A\in M_n(\Rr),\left| tr(A)\right| \le \sqrt{n}N(A).
        \]
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
