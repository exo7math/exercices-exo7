\exercice{103, cousquer, 2003/10/01}

\enonce
\begin{enumerate}
  \item Montrer que si \( x+y+z = a \), \( yz+zx+xy = b \), \( xyz = c \), alors \( x \), \( y \) et \( z \) sont solutions de l'équation \( Z^3-aZ^2+bZ-c = 0 \). Trouver \( x \), \( y \) et \( z \) si on suppose \( a = b = 0 \) et \( c = -8 \).

  \item Résoudre le système
        \[
          \left\{\begin{array}{rcl}
            x+y+z       & = & 4 \\
            x^2+y^2+z^2 & = & 4 \\
            x^3+y^3+z^3 & = & 1
          \end{array}\right.
        \]
\end{enumerate}
\finenonce

\noindication

\correction
\begin{enumerate}
  \item À permutation près, \( x = -2 \), \( y = -2j \) et \( z = -2j^2 \) (\( j \) désigne la racine cubique de l'unité \( \frac{-1+i\sqrt3}{2} \)).

  \item À permutation près, \( x = 1 \), \( y = \frac{3+i\sqrt3}{2} \) et \( z = \frac{3-i\sqrt3}{2} \).
\end{enumerate}
\fincorrection

\finexercice
