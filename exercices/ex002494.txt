\exercice{2494, sarkis, 2009/04/01}

\enonce
\begin{enumerate}
  \item Montrez que \( d(x,y) = |x-y| \) est bien une distance sur l'ensemble des réels.

  \item Pour tout couple d'éléments \( X = (x_1,\dots,x_n) \) et \( Y = (y_1,\dots,y_n) \) de \( \Rr^n \), on définit \( d(X,Y) = \sup_{i = 1,\dots,n}|x_i-y_i| \). Montrez que \( d \) est bien une distance sur \( \Rr^n \).

  \item Faire de même avec \( d(X,Y) = |x_1-y_1|+|x_2-y_2|+...+|x_n-y_n| \).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
