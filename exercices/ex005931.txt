\exercice{5931, tumpach, 2010/11/11}

\enonce
\begin{enumerate}
  \item Calculer \( \int_{-\infty}^{+\infty} e^{-x^2}\,dx \).\\
        \emph{Indication :} On pourra d'abord calculer \( \int_{\Rr^2} e^{-(x^2 + y^2)}\, dx dy \) en passant en coordonnées polaires.

  \item \emph{Calcul de l'aire de la sphère unité de \( \Rr^n \).} Soit \( \mathcal{S}_{n-1} = \{(x_1, \dots, x_n) \in \Rr^n ,\, \sum_{i = 1}^n x_i^2 = 1 \} \) la sphère unité de \( \Rr^n \). On note \( \mathcal{A}_{n-1} \) son aire. Calculer
        \[
          \int_{\Rr^n} e^{-\sum_{i = 1}^n x_i^2} \,dx_1\dots dx_n
        \]
        en fonction de \( \mathcal{A}_{n-1} \). En déduire l'expression de \( \mathcal{A}_{n-1} \) en fonction de la fonction \( \Gamma \):
        \[
          \Gamma(s) := \int_0^{+\infty} x^{s-1} e^{-x}\,dx.
        \]

  \item \emph{Calcul du volume de la boule unité de \( \Rr^n \).} Soit \( \mathcal{B}_n = \{(x_1, \dots, x_n)
        \in \Rr^n ,\, \sum_{i = 1}^n x_i^2 \le 1 \} \) la boule fermée de rayon 1 dans \( \Rr^n \). On note \( \mathcal{V}_n \) son volume. Montrer que \( \mathcal{V}_n = \frac{\mathcal{A}_{n-1}}{n} \). En déduire que :
        \[
          \mathcal{V}_n =
          \frac{{\pi}^{\frac{n}{2}}}{\Gamma\left(\frac{n}{2} + 1 \right)}.
        \]

  \item \emph{Application :} Que vaut l'aire de la sphère de rayon \( R \) dans \( \Rr^2 \)? \( \Rr^3 \)? Que vaut le volume de la boule de rayon \( R \) dans \( \Rr \)? \( \Rr^2 \)? \( \Rr^3 \)?
\end{enumerate}
\finenonce

\noindication

\correction
\begin{enumerate}
  \item Soit \( I = \int_{-\infty}^{+\infty} e^{-x^2}\,dx \). On a :
        \[
          I^2 = \int_{\Rr^2} e^{-(x^2 + y^2)}\, dx dy.
        \]
        L'application \( \Phi : \Rr_{+}^{*} \times ]0, 2\pi[
        \rightarrow \Rr^2 \setminus \{ (x, 0), x \ge 0\} \) définie par :
        \[
          \Phi(r, \theta) = \left(r\cos \theta, r\sin \theta \right)
        \]
        est un \( \mathcal{C}^1 \)-difféomorphisme. De plus
        \[
          \int_{\Rr^2} e^{-(x^2 + y^2)}\, dx dy =
          \int_{\Rr^2\setminus\{ (x, 0), x \ge 0\}} e^{-(x^2 + y^2)}\, dx dy
        \]
        car l'ensemble \( \{ (x, 0), x \ge 0\} \) est négligeable. On en déduit que :
        \[
          I^2 = \int_{r = 0}^{+\infty}\int_{\theta = 0}^{2\pi} e^{-r^2} r \,dr d\theta = 2\pi \left[\frac{-e^{-r^2}}{2} \right]_0^{+\infty} =
          \pi.
        \]
        Ainsi \( I = \int_{-\infty}^{+\infty} e^{-x^2}\,dx = \sqrt{\pi} \).

  \item
        \emph{Calcul de l'aire de la sphère unité de \( \Rr^n \).} Soit \( \mathcal{S}_{n-1} = \{(x_1, \dots, x_n)
        \in \Rr^n ,\, \sum_{i = 1}^n x_i^2 = 1 \} \) la sphère unité de \( \Rr^n \). On note \( \mathcal{A}_{n-1} \) son aire. D'après la question précédente, on a :
        \[
          \int_{\Rr^n} e^{-\sum_{i = 1}^n x_i^2} \,dx_1\dots dx_n = {\pi}^{\frac{n}{2}}.
        \]
        D'autre part, puisque l'aire de la sphère de rayon \( r \) dans \( \Rr^n \) vaut \( r^{n-1}\mathcal{A}_{n-1} \), il vient :
        \[
          \int_{\Rr^n} e^{-\sum_{i = 1}^n x_i^2} \,dx_1\dots dx_n = \mathcal{A}_{n-1} \int_0^{+\infty} e^{-r^2}
          r^{n-1}\,dr.
        \]
        En posant le changement de variable \( x = r^2 \), on obtient :
        \[
          \int_{\Rr^n} e^{-\sum_{i = 1}^n x_i^2} \,dx_1\dots dx_n = \frac{1}{2}\mathcal{A}_{n-1}\int_0^{+\infty} e^{-x}
          x^{\frac{n}{2}-1} \,dx,
        \]
        d'où :
        \[
          \mathcal{A}_{n-1} = \frac{2{\pi}^{\frac{n}{2}}}{\Gamma\left(
            \frac{n}{2}\right)}.
        \]

  \item \emph{Calcul du volume de la boule unité de \( \Rr^n \).} Soit \( \mathcal{B}_n = \{(x_1, \dots, x_n)
        \in \Rr^n ,\, \sum_{i = 1}^n x_i^2 \le 1 \} \) la boule fermée de rayon 1 dans \( \Rr^n \). On note \( \mathcal{V}_n \) son volume. On a :
        \[
          \mathcal{V}_n = \int_0^1 r^{n-1}
          \mathcal{A}_{n-1} \,dr = \mathcal{A}_{n-1} \left[\frac{r^n}{n}
            \right]_0^1 = \frac{\mathcal{A}_{n-1}}{n}.
        \]
        On en déduit que :
        \[
          \mathcal{V}_n = \frac{2{\pi}^{\frac{n}{2}}}{n\Gamma\left(
            \frac{n}{2}\right)}.
        \]
        Ce qui se réduit à :
        \[
          \mathcal{V}_n =
          \frac{{\pi}^{\frac{n}{2}}}{\Gamma\left(\frac{n}{2} + 1 \right)}
        \]
        en utilisant l'identité : \( \Gamma(s + 1) = s\Gamma(s) \).

  \item
        \emph{Application :} L'aire de la sphère de rayon \( R \) dans \( \Rr^2 \) vaut
        \[
          \mathcal{A}_1 R = \frac{2\pi}{\Gamma(1)} R = 2\pi R,
        \]
        qui est bien le périmètre du cercle de rayon \( R \) dans le plan.

        Sachant \( \Gamma(\frac 12) = \sqrt \pi \), l'aire de la sphère de rayon \( R \) dans \( \Rr^3 \) vaut
        \[
          \mathcal{A}_2 R^2 = \frac{2 {\pi}^{\frac{3}{2}}}{\Gamma\left(\frac{3}{2}
            \right)}
          R^2 = \frac{2\pi \sqrt{\pi}}{\frac{1}{2}\Gamma\left(\frac{1}{2}\right)} R^2 = 4\pi R^2
        \]
        qui est bien l'aire de la sphère \( S^2 \).

        Le volume de la boule de rayon \( R \) dans \( \Rr \) vaut
        \[
          \mathcal{V}_1 R = \frac{2{\pi}^{\frac{1}{2}}}{\Gamma\left(
            \frac{1}{2}\right)} R = \frac{2 \sqrt{\pi}}{\sqrt{\pi}} R = 2R,
        \]
        qui est bien la longueur du segment \( [-R, R] \).

        Le volume de la boule de rayon \( R \) dans \( \Rr^2 \) vaut
        \[
          \mathcal{V}_2 R^2 = \frac{2\pi}{2\Gamma(1)} R^2 = \pi R^2,
        \]
        qui est bien l'aire du disque de rayon \( R \).

        Le volume de la boule de rayon \( R \) dans \( \Rr^3 \) vaut
        \[
          \mathcal{V}_3 R^3 = \frac{\mathcal{A}_2}{3} R^3 = \frac{4}{3}
          \pi R^3.
        \]
\end{enumerate}
\fincorrection

\finexercice
