\exercice{2319, matexo1, 2002/02/01}

\enonce
Calculer les primitives des fonctions suivantes\,:
\begin{eqnarray*}
  t &\mapsto &t^2 \exp (t^3), \\
  t &\mapsto &\frac{\sin^3 t}{{1 + \cos^2 t}}, \\
  t &\mapsto &\frac{1}{{1-t^2+2 \sqrt{1-t^2}}}, \\
  t &\mapsto &\frac{\sinh^2 t}{{\cosh t}}, \\
  t &\mapsto &\frac{\cos t}{{\cos 2t}}, \\
  t &\mapsto &\frac{1}{{1+\tanh^2 t}}.
\end{eqnarray*}
\finenonce

\noindication

\nocorrection

\finexercice
