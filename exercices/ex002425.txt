\exercice{2425, matexo1, 2002/02/01}

\enonce
\begin{center}
  {\sc VRAI ou FAUX}
\end{center}

\begin{enumerate}
  \item L'ensemble \( \{0\} \) est un espace vectoriel réel.

  \item L'ensemble \( \{0,1\} \) est un espace vectoriel réel.

  \item Tout sous-espace vectoriel autre que \( \{0\} \) possède un sous-espace strict.

  \item L'intersection de deux sous-espaces vectoriels (d'un même espace plus grand) est un espace vectoriel.

  \item La réunion de deux sous-espaces vectoriels est un espace vectoriel.

  \item La somme de deux sous-espaces vectoriels est un espace vectoriel.

  \item Le produit cartésien \( E\times F \) de deux espaces vectoriels est un espace vectoriel.
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
