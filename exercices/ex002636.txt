\exercice{2636, debievre, 2009/05/19}

\enonce
\begin{enumerate}
  \item Y a-t-il une fonction \( g\colon \Rr^2\to\Rr \) telle que
        \[
          \mathrm{d} g = x^2y^2 \mathrm{d} x + x^3 y\mathrm{d} y?
        \]

  \item Trouver les fonctions \( b\colon\Rr^2\to\Rr \) telles qu'il existe \( g\colon \Rr^2\to\Rr \) satisfaisant la condition
        \[
          \mathrm{d} g = x^2y^2 \mathrm{d} x + b(x,y)\mathrm{d} y.
        \]
        Étant donnée alors la fonction \( b \), déterminer toutes les fonctions \( g \) correspondantes.
\end{enumerate}
\finenonce

\indication
Soient \( h \), \( u \), \( v \) des fonctions des deux variables \( x \) et \( y \). Rappeler que
\begin{align*}
  \mathrm dh                            & =\frac{\partial h}{\partial x}\mathrm dx +
  \frac{\partial h}{\partial y} \mathrm dy ,
  \\
  \mathrm d(u\mathrm dx + v \mathrm dy) & =
  \left(\frac{\partial v}{\partial x} -\frac{\partial u}{\partial y}\right)
  \mathrm dx \mathrm dy ,
  \\
  \mathrm dx \mathrm dy                 & =-\mathrm dy\mathrm dx .
\end{align*}
\finindication

\correction
\begin{enumerate}
  \item La forme différentielle \( x^2y^2 \mathrm{d} x + x^3 y\mathrm{d} y \) de degré 1 n'est pas fermée car la forme différentielle de degré 2
        \[
          d(x^2y^2 \mathrm{d} x + x^3 y\mathrm{d} y) = 2x^2y \mathrm{d} y\mathrm{d} x + 3x^2 y\mathrm{d} x\mathrm{d} y = x^2 y \mathrm{d} x\mathrm{d} y
        \]
        est non nulle. Par conséquent, une fonction \( g\colon \Rr^2\to\Rr \) du type cherché ne peut pas exister.

  \item Une fonction \( b \) du type cherché doit satisfaire à l'équation différentielle partielle
        \[
          2x^2y -\frac{\partial b}{\partial x} = 0
        \]
        d'où \( b(x,y) = \frac 23 x^3 y +k(y) \) où \( k \) est une fonction de la variable \( y \). Une fonction \( g \) correspondante doit alors satisfaire aux équations différentielles partielles
        \[
          \frac{\partial g}{\partial x} = x^2 y^2,
          \
          \frac{\partial g}{\partial y} = \frac 23 x^3 y +k(y).
        \]
        Il s'ensuit que \( g \) est de la forme \( g(x,y) = \frac 13 x^3 y^2 +K(y) \) où \( K \) est une fonction de la varriable \( y \).
\end{enumerate}
\fincorrection

\finexercice
