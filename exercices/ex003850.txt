\exercice{3850, quercia, 2010/03/11}

\enonce[Prolongement d'inégalités]
\begin{enumerate}
  \item Soient \( {f,g} : \Rr \to \Rr \) continues telles que : \( \forall\ x \in \Qq,\ f(x) < g(x) \).
        \begin{enumerate}
          \item Montrer que \( f \le g \).

          \item Montrer qu'on n'a pas nécessairement : \( \forall\ x \in \Rr,\ f(x) < g(x) \).
        \end{enumerate}

  \item Soit \( f : \Rr \to \Rr \) continue dont la restriction à \( \Qq \) est strictement croissante. Montrer que \( f \) est strictement croissante.

\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
