\exercice{7553, mourougane, 2021/08/10}

\enonce[La fonction zeta de Riemann]
On rappelle que pour tout entier naturel \( n \), et tout nombre complexe \( z \), \( n^z \) désigne \( p_z(n) = \exp(z\log n) \).
\begin{enumerate}
  \item Déterminer le module de \( n^z \).

  \item Montrer que la série \( \sum_{n\ge 1} \frac{1}{n^z} \) converge normalement sur l'ouvert \( \{z\in\Cc / \Re(z)>1\} \).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
