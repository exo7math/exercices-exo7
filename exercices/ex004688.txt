\exercice{4688, quercia, 2010/03/16}

\enonce[Approximation d'un irrationnel]
Soit \( x \in \Rr^* \) et \( (r_n) \) une suite de rationnels convergeant vers \( x \). On écrit \( r_n = \frac{p_n}{q_n} \) avec \( p_n\in\Zz \), \( q_n\in\Nn^* \).

\begin{enumerate}
  \item Montrer que si l'une des suites \( (p_n) \), \( (q_n) \) est bornée, alors l'autre l'est aussi, et \( x \in \Qq \).

  \item En déduire que si \( x \in \Rr\setminus \Qq \), alors \( |p_n| \xrightarrow[n\to\infty]{} +\infty \) et \( q_n \xrightarrow[n\to\infty]{} +\infty \).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
