\exercice{6750, queffelec, 2011/10/16}

\enonce
Soit \( P \) et \( Q \) deux polynômes tels que \( \deg Q>\deg P \).
\begin{enumerate}
  \item Exprimer \( \sum\hbox{Res}(\frac{P}{Q}) \) à l'aide des coefficients de \( P \) et \( Q \).

  \item Soit \( P(z) = z^n+\sum_{k=0}^{n-1} a_k z^k \) un polynôme dont toutes les racines sont dans \( D(0,R) \). Montrer que \( f(x) = \displaystyle\frac{1}{2i\pi}\int_{\Gamma_R}\frac{e^{xz}}{P(z)}\ dz \) est la solution de l'équation différentielle d'ordre \( n \), \( y^{(n)}+ a_{n-1}y^{(n-1)}+\cdots+a_0y = 0 \) de condition initiale \( y^{(j)}(0) = 0 \) si \( j<n-1 \) et \( y^{(n-1)}(0) = 1 \).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
