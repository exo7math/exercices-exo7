\exercice{1531, legall, 1998/09/01}

\enonce
\begin{enumerate}
  \item Soit \( A = (a_{ij}) \in \textbf{M}_n(\Rr) \). Montrer que \( S = {}^t\! A\cdot A \) est une matrice symétrique dont toutes les valeurs propres \( \lambda_1, \dots, \lambda_n \) sont positives. Démontrer l'égalité : \( \sum_{i = 1}^n \lambda_i = \sum_{1\le i,j\le n}
        a_{ij}^2 \).

  \item Soit \( S \in \textbf{M}_n(\Rr) \) une matrice symétrique. Existe-t-il une matrice \( A \in \textbf{M}_n(\Rr) \) telle que \( S = {}^t\! A\cdot A \) ? Donner une condition nécessaire et suffisante sur \( S \) pour que \( A \) soit inversible. Application à \( S = \left(\begin{array}{cc}2 & 1 \\
             1     & 2\end{array}\right) \).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
