\exercice{1894, legall, 2003/10/01}

\enonce
\begin{enumerate}
  \item Montrer que l'application \( (f,g)\mapsto \langle f,g\rangle = \displaystyle{\int_0^1f(t)g(t)dt} \) est un produit scalaire euclidien sur \( C[0,1] \), l'espace vectoriel des fonctions continues sur \( [0,1] \) à valeurs réelles.

  \item On note \( C = \{ f\in C[0,1] ; \displaystyle{\int_0^1f(t)dt = 1} \). Montrer que \( \displaystyle{ \inf_{f\in C} \int_0^1f^2(t)dt = 1} \) et que cette borne inférieure est atteinte.
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
