\exercice{1961, liousse, 2003/10/01}

\enonce[Médianes]
On considère dans \( P \) trois points \( A \), \( B \) et \( C \).

\begin{enumerate}
  \item Déterminer dans le repère \( (A,\vec{AB},\vec{AC}) \) des équations pour les médianes du triangle ABC.

  \item En déduire que les médianes d'un triangle sont concourantes.
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
