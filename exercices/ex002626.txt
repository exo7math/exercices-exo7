\exercice{2626, debievre, 2009/05/19}

\enonce
Soit \( f\colon\Rr^3 \rightarrow \Rr \) une fonction de classe \( C^1 \) et soit \( g\colon \Rr^3 \rightarrow \Rr \) la fonction définie par
\[
  g(x,y,z) = f(x-y,y-z,z-x).
\]
Montrer que
\begin{equation}
  \frac{\partial g}{\partial x} + \frac{\partial g}{\partial y} + \frac{\partial g}{\partial z} = 0.
  \label{ex3}
\end{equation}
\finenonce

\indication
Calculer mène à la vérité.
\finindication

\correction
\begin{align*}
  \frac{\partial g}{\partial x}
   & =
  \frac{\partial f}{\partial x}-\frac{\partial f}{\partial z}
  \\
  \frac{\partial g}{\partial y}
   & =
  \frac{\partial f}{\partial y}-\frac{\partial f}{\partial x}
  \\
  \frac{\partial g}{\partial z}
   & =
  \frac{\partial f}{\partial z}-\frac{\partial f}{\partial y}
\end{align*}
d'où \eqref{ex3}.
\fincorrection

\finexercice
