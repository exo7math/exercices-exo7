\exercice{6519, drutu, 2011/10/16}

\enonce
\begin{enumerate}
  \item Montrer que \( \Qq (\sqrt{d}),\; d = 7,\; 11 \), est un corps.

  \item Montrer que \( a+b\sqrt{7}\to a+b\sqrt{11} \) n'est pas un isomorphisme de corps entre \( \Qq (\sqrt{7}) \) et \( \Qq (\sqrt{11}) \).

  \item Montrer que les corps \( \Qq (\sqrt{7}) \) et \( \Qq (\sqrt{11}) \) ne sont pas isomorphes.
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
