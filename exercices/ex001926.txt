\exercice{1926, drutu, 2003/10/01}

\enonce
\begin{enumerate}
  \item {\bf{Théorème de Guldin}}
        Soit \( D_0 \) un domaine tracé dans le demi-plan \( \{ (x,0,z)\in \Rr^3 \mid x\ge 0 \} \). Si l'on fait tourner \( D_0 \) autour de l'axe \( Oz \), on obtient un domaine \( D \) de \( \Rr^3 \). En utilisant les coordonnées cylindriques, montrer que
        \[
          Vol\, (D) = 2\pi Aire\, (D_0)\cdot x_G\, ,
        \]
        où \( (x_G,z_G) \) sont les coordonnées du centre d'inertie du domaine \( D_0 \).

  \item Calculer les volumes des domaines suivants :
        \begin{enumerate}
          \item le tore obtenu en faisant tourner autour de \( Oz \) le domaine \( D_0 = \{ (x,0,z)\mid \frac{(x-c)^2}{a^2} +\frac{z^2}{b^2}\le 1 \} \), où \( a<c \) ;

          \item \( D = \left\{ (x,y,z)\in \Rr^3 \mid x^2 +y^2 +z^2 \le 4R^2\, ,\, x^2 +y^2 \le R^2 \right\} \), où \( R>0 \).
        \end{enumerate}
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
