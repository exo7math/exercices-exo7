\exercice{5966, tumpach, 2010/11/11}

\enonce
Soit \( \Omega = \Rr^n \) muni de la mesure de Lebesgue. Pour tout \( 1\le p <+\infty \), on note \( L^p(\Rr^n) \) l'espace des fonctions \( f :\Rr^n \rightarrow \Cc \) telles que \( \|f\|_p := \left(\int_{\Rr^n}
|f|^p(x)\,dx\right)^{\frac{1}{p}}<+\infty \) modulo l'équivalence \( f\sim g \Leftrightarrow f-g = 0  \mu-p.p \). L'espace des fonctions essentiellement bornées sera noté \( L^{\infty}(\Rr^n) \).
\begin{enumerate}
  \item
        \begin{itemize}
          \item Pour quelle valeur de \( \alpha \) la fonction \( x\mapsto \frac{1}{\left(1 + |x|^2 \right)^{\alpha}} \) appartient-elle à \( L^p(\Rr^n) \) ?

          \item Pour quelle valeur de \( \beta \) la fonction \( x\mapsto
                \frac{1}{|x|^{\beta}} e^{-\frac{|x|^2}{2}} \) appartient-elle à \( L^p(\Rr^n) \) ?

          \item Soit \( 1\le q < p \le +\infty \). En utilisant \( (a) \) et \( (b) \), trouver une fonction \( f \) qui appartienne à \( L^q(\Rr^n) \) mais pas à \( L^p(\Rr^n) \) et une fonction \( g \) qui appartienne à \( L^p(\Rr^n) \) mais pas à \( L^q(\Rr^n) \).
        \end{itemize}

  \item
        \begin{itemize}
          \item Soit \( 1\le q< p<+\infty \). Montrer que l'espace \( L^p(\Rr^n) \cap L^q(\Rr^n) \) est un espace de Banach pour la norme \( \|\cdot \|_{p,q} = \| \cdot \|_p + \| \cdot \|_q \).

          \item Soit \( r \) tel que \( q< r<p \). Montrer que
                \[
                  \|f\|_r \le \|f\|_p^{\alpha}\|f\|_q^{1-\alpha}
                \]
                où \( \frac{1}{r} = \frac{\alpha}{p} + \frac{1-\alpha}{q} \), \( \alpha\in[0, 1] \). On pourra écrire \( r = r\alpha + r(1-\alpha) \) et utiliser l'inégalité de Hölder pour un couple de réels conjugués bien choisi.

          \item En déduire que si \( f_n \) converge vers \( f \) dans \( L^p(\Rr^n) \cap L^q(\Rr^n) \) alors \( f_n \) converge vers \( f \) dans \( L^r(\Rr^n) \), i.e \( L^p(\Rr^n) \cap L^q(\Rr^n) \) est un sous-espace de Banach de \( L^r(\Rr^n) \).
        \end{itemize}

  \item Soit \( f\in L^p([0, +\infty[)\cap L^q([0, +\infty[) \) avec \( 1\le q<2<p \). Montrer que la fonction \( h \) définie par \( h(r) = \frac{1}{\sqrt{r}}f(r) \) appartient à \( L^1([0, +\infty[) \) et trouver des constantes \( C_{p,q} \) et \( \gamma \) telles que \( \|h\|_1 \le C_{p, q}
        \|f\|_q^{\gamma}\|f\|_p^{(1-\gamma)} \).
\end{enumerate}
\finenonce

\noindication

\correction
Soit \( \Omega = \Rr^n \) muni de la mesure de Lebesgue. Pour tout \( 1\le p <+\infty \), on note \( L^p(\Rr^n) \) l'espace des fonctions \( f :\Rr^n \rightarrow \Cc \) telles que \( \|f\|_p := \left(\int_{\Rr^n}
|f|^p(x)\,dx\right)^{\frac{1}{p}}<+\infty \) modulo l'équivalence \( f\sim g \Leftrightarrow f-g = 0  \mu-p.p \). L'espace des fonctions essentiellement bornées sera noté \( L^{\infty}(\Rr^n) \).
\begin{enumerate}
  \item
        \begin{itemize}
          \item La fonction \( x\mapsto
                \frac{1}{\left(1 + |x|^2 \right)^{\alpha}} \) appartient à \( L^p(\Rr^n) \) pour \( 2\alpha p > n \).

          \item La fonction \( x\mapsto \frac{1}{|x|^{\beta}} e^{-\frac{|x|^2}{2}} \) appartient à \( L^p(\Rr^n) \) pour \( p\beta< n \).

          \item Soit \( 1\le q < p \le +\infty \). La fonction
                \[
                  f(x) = (1+|x|^2)^{-\frac{n}{p+q}}
                \]
                vérifient \( f \in L^p(\Rr^n) \) et \( f\notin L^q(\Rr^n) \). La fonction
                \[
                  g(x) = |x|^{-\frac{n}{2}\left(\frac{1}{p}+\frac{1}{q}\right)}
                  e^{-\frac{|x|^2}{2}}
                \]
                vérifient \( g\in L^q(\Rr^n) \) et \( g\notin L^p(\Rr^n) \).

        \end{itemize}

  \item
        \begin{itemize}
          \item Soit \( 1\le q< p<+\infty \) et \( f_n \) une suite de Cauchy pour la norme \( \|\cdot\|_{p, q} = \|
                \cdot \|_p + \| \cdot \|_q \). Comme \( \|\cdot\|_p \le
                \|\cdot\|_{p, q} \), \( f_n \) est une suite de Cauchy dans \( L^p(\Rr^n) \), donc elle converge vers une fonction \( f\in L^p(\Rr^n) \) pour la norme \( \|\cdot\|_p \). De même, \( \|\cdot\|_q\le \|\cdot\|_{p, q} \), donc \( f_n \) converge vers une fonction \( g\in L^q(\Rr^n) \) pour la norme \( \|\cdot\|_q \). De plus, il existe une sous-suite de \( f_{n_k} \) qui converge vers \( f \) presque-partout et il existe une sous-suite de \( f_{n_k} \) qui converge vers \( g \) presque-partout. Ainsi \( f = g  \mu \)-p.p. et \( f_n \) converge vers \( f = g \) dans \( L^p(\Rr^n)\cap L^q(\Rr^n) \).

          \item Soit \( r \) tel que \( q< r<p \). Montrons que
                \[
                  \|f\|_r \le \|f\|_p^{\alpha}\|f\|_q^{1-\alpha}
                \]
                où \( \frac{1}{r} = \frac{\alpha}{p} + \frac{1-\alpha}{q} \), \( \alpha\in[0, 1] \). Puisque \( 1 = \frac{\alpha r}{p} +
                \frac{(1-\alpha)r}{q} \), les réels \( p' = \frac{p}{\alpha r} \) et \( q' = \frac{q}{(1 - \alpha)r} \) sont conjugués. D'après l'inégalité de Hölder,
                \begin{eqnarray*}
                  \int_{\Rr^n} |f|^r(x)\,dx& =& \int_{\Rr^n}
                  |f|^{r\alpha}(x)\cdot |f|^{(1-\alpha) r}(x)\,dx\\
                  & \le & \left(\int_{\Rr^n} |f|^{\alpha r p'}(x)\,dx
                  \right)^{\frac{1}{p'}} \left( \int_{\Rr^n}
                  |f|^{(1-\alpha)r q'}(x)\,dx\right)^{\frac{1}{q'}}\\
                  & \le & \left(\int_{\Rr^n} |f|^p(x)\,dx
                  \right)^{\frac{\alpha r}{p}} \left( \int_{\Rr^n}
                  |f|^q(x)\,dx\right)^{\frac{(1-\alpha)r}{q}}\\
                  & \le & \|f\|_p^{\alpha r} \|f\|_q^{(1-\alpha)r},
                \end{eqnarray*}
                ce qui équivaut à \( \|f\|_r \le
                \|f\|_p^{\alpha}\|f\|_q^{(1-\alpha)} \). On peut également écrire \( r = \beta q + (1-\beta)p \) avec \( \beta \in ]0, 1[ \) et appliquer Hölder avec les réels conjugués \( \frac{1}{\beta} \) et \( \frac{1}{1-\beta} \):
                \begin{eqnarray*}
                  \int_{\Rr^n} |f|^r(x)\,dx& =& \int_{\Rr^n}
                  |f|^{\beta q}(x)\cdot|f|^{(1-\beta)p}(x)\,dx \\
                  &\le & \left(\int_{\Rr^n} |f|^q(x)\,dx \right)^{\beta}
                  \left( \int_{\Rr^n} |f|^p(x)\,dx\right)^{(1-\beta)},
                \end{eqnarray*}
                ce qui implique
                \[
                  \|f\|_r \le
                  \|f\|_q^{\frac{q\beta}{r}}\|f\|_p^{\frac{p(1-\beta)}{r}}
                \]
                qui est l'inégalité cherchée car \( \alpha = \frac{p\beta}{r} \) vérifie bien \( \frac{1}{r} = \frac{\alpha}{p} +
                \frac{1-\alpha}{q} \).

          \item Si \( f_n \) converge vers \( f \) dans \( L^p(\Rr^n) \cap L^q(\Rr^n) \) alors \( f_n \) converge vers \( f \) dans \( L^p(\Rr^n) \) et dans \( L^q(\Rr^n) \), donc dans \( L^r(\Rr^n) \) d'après l'inégalité précédente. En conclusion, \( L^p(\Rr^n) \cap L^q(\Rr^n) \) est fermé dans \( L^r(\Rr^n) \) donc un sous-espace de Banach de \( L^r(\Rr^n) \).
        \end{itemize}

  \item Soit \( f\in L^p([0, +\infty[)\cap L^q([0, +\infty[) \) et \( h \) la fonction définie par \( h(r) = \frac{1}{\sqrt{r}}f(r) \). On notera \( p' \) le conjugué de \( p \) et \( q' \) le conjugué de \( q \). Montrons que \( h \) appartient à \( L^1([0, +\infty[) \). On a :
        \begin{eqnarray*}
          \int_0^{+\infty} \frac{1}{\sqrt{r}} |f(r)|\,dr = \int_0^R
          \frac{1}{\sqrt{r}} |f(r)|\,dr + \int_R^{+\infty}
          \frac{1}{\sqrt{r}} |f(r)|\,dr\\
          \le \left( \int_0^R r^{-\frac{p'}{2}}\,dr\right)^{\frac{1}{p'}} \left(\int_0^R |f(r)|^p \,dr\right)^{\frac{1}{p}} + \left(\int_R^{+\infty}
          r^{-\frac{q'}{2}}\,dr \right)^{\frac{1}{q'}} \left(
          \int_R^{+\infty}
          |f(r)|^q\,dr\right)^{\frac{1}{q}}\\
          \le \left(\frac{1}{1-\frac{p'}{2}}\right)^{\frac{1}{p'}}
          R^{\left(\frac{1}{p'}-\frac{1}{2}\right)}\|f\|_p +
          \left(\frac{1}{\frac{q'}{2} - 1}\right)^{\frac{1}{q'}}
          R^{\left(\frac{1}{q'}-\frac{1}{2}\right)} \|f\|_q.
        \end{eqnarray*}
        En optimisant par rapport à \( R \), on obtient :
        \begin{eqnarray*}
          \int_0^{+\infty} \frac{1}{\sqrt{r}} |f(r)|\,dr &\le & C_{p,q}
          \|f\|_p^{1-\gamma}\|f\|_q^{\gamma},
        \end{eqnarray*}
        où, en posant \( \alpha = \frac{1}{2}- \frac{1}{p} \) et \( \beta = \frac{1}{q} - \frac{1}{2} \), on a \( \gamma = \frac{\alpha}{\alpha+\beta} \), et \( C_{p,q} = \frac{\alpha +
          \beta}{\alpha^{\gamma} \beta^{1-\gamma}}\left(1-\frac{p'}{2}
        \right)^{-\frac{1-\gamma}{p'}}\left(\frac{q'}{2} - 1
        \right)^{-\frac{\gamma}{q'}} \).
\end{enumerate}
\fincorrection

\finexercice
