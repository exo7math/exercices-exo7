\exercice{5973, tumpach, 2010/11/11}

\enonce[Coordonnées sphériques dans \( \Rr^n \)]
Soit \( \Omega' \) l'ouvert de \( \Rr^n \) défini par
\[
  \Omega' = \{(r, \theta_1, \dots, \theta_{n-1})\in\Rr^n |  0< r, 0 <\theta_1, \dots, \theta_{n-2} < \pi, 0 < \theta_{n-1} < 2\pi\}.
\]
Soit l'application \( S \) de \( \Omega' \) dans \( \Rr^n \) définie par
\[
  \begin{array}{lcl}
    x_1     & = & r \cos \theta_1,                                      \\
    x_2     & = & r \sin \theta_1 \cos \theta_2,                        \\
    \vdots  &   &                                                       \\
    x_{n-2} & = & r \sin \theta_1 \sin \theta_2 \dots \cos \theta_{n-2} \\
    x_{n-1} & = & r \sin \theta_1 \sin\theta_2 \dots \sin \theta_{n-2}
    \cos \theta_{n-1}                                                   \\
    x_n     & = & r \sin \theta_1 \sin \theta_2 \dots \sin\theta_{n-2}
    \sin \theta_{n-1},
  \end{array}
\]
où \( (x_1, \dots, x_n) \) sont les coordonnées cartésiennes de \( x\in\Rr^n \).
\begin{enumerate}
  \item Soit \( \Omega = \Rr^n\setminus\{ x\in\Rr^n | x_n = 0 \text{et} x_{n-1} \ge 0 \} \). Montrer que \( \Omega \) est une partie ouverte de \( \Rr^n \) dont le complémentaire est de mesure nulle, et que \( S \) est un difféomorphisme de \( \Omega' \) sur \( \Omega \).

  \item Soit \( f \) une fonction borélienne positive sur \( \Rr^n \). Montrer que
        \[
          \begin{array}{lcl}
            \int_{\Rr^n} f(x)\,dx & = & \int_{\Omega'} (f\circ S)(r,
            \theta_1, \dots, \theta_{n-1})\,r^{n-1}
            \sin^{n-2}\theta_1\,\sin^{n-3}\theta_2 \dots \sin
            \theta_{n-2}\,dr\,d\theta_1 \dots d\theta_{n-1}          \\ & & \\ & = &
            \int_{\Omega'} (f\circ S)(r, \theta_1, \dots,
            \theta_{n-1})\,r^{n-1}\,dr\,d\sigma,
          \end{array}
        \]
        où \( d\sigma \) est la mesure uniforme sur la sphère unité de \( \Rr^n \).

  \item En utilisant les coordonnées sphériques, calculer le volume \( \mathcal{V}_4 \) de la boule unité de \( \Rr^4 \) et l'aire \( \mathcal{A}_3 \) de la sphère unité \( \mathcal{S}^3 \) de \( \Rr^4 \).
\end{enumerate}
\finenonce

\noindication

\correction
\begin{enumerate}
  \item Posons \( \Omega = \Rr^n\setminus\{ x\in\Rr^n | x_n = 0 \text{et} x_{n-1} \ge 0 \} \). Comme \( 0 < \theta_{n-1} < 2\pi \), l'image de \( \Omega' \) par \( S \) est incluse dans \( \Omega \). Réciproquement, soit \( x \) un élément de \( \Omega \). Posons \( r = |x| \), alors pour tout \( i \in\{1, \dots, n-2\} \), on peut définir par récurrence \( \theta_i\in(0, \pi) \) grâce à son cosinus :
        \[
          \cos\theta_i = \frac{x_i}{r
            \sin\theta_1\dots\sin\theta_{i-1}}.
        \]
        Quant à \( \theta_{n-1} \), il est déterminé par son sinus et son cosinus. Comme \( x_n \neq 0 \) ou \( x_{n-1} < 0 \), nécessairement \( \theta_{n-1} \neq 0(\text{modulo} 2\pi) \).

        L'application \( S \) est continûment différentiable, car chacune de ses composantes l'est. La matrice jacobienne a ses vecteurs colonnes orthogonaux, et de norme respectivement \( 1 \), \( r \), \( r\sin\theta_1 \), \( \dots \), \( r\sin\theta_1\dots\sin\theta_{n-2} \). Son déterminant vaut alors \( r^{n-1}\left(\sin \theta_1\right)^{n-2}\dots
        \sin\theta_{n-2} \). Comme ce déterminant ne s'annule jamais, \( S \) est un difféomorphisme de \( \Omega' \) sur \( \Omega \).

  \item C'est la formule du changement de variable.

  \item On a :
        \[
          \begin{array}{lcl} \mathcal{V}_4 & = & \int_{r = 0}^1\int_{\theta_1 = 0}^{\pi}\int_{\theta_2 = 0}^{\pi} \int_{\theta_3 = 0}^{2\pi}
             r^3\sin^2\theta_1\sin\theta_2\,dr d\theta_1 d\theta_2 d\theta_3                                                                                              \\
                                 & = & 2\pi\left[\frac{r^4}{4}\right]_0^1\left(\int_0^{\pi}\sin^2\theta_1 d\theta_1\right)\left(\int_0^{\pi}\sin\theta_2 d\theta_2\right) \\
                                 & = & \frac{\pi}{2}\left(\int_0^{\pi}\frac{1 -\cos 2\theta_1}{2}
             d\theta_1\right)\left[-\cos\theta_2\right]_0^{\pi}                                                                                                           \\
                                 & = & \frac{\pi^2}{2}.                                                                                                                   \\
             \mathcal{A}_3       & = & \int_{\theta_1 = 0}^{\pi}\int_{\theta_2 = 0}^{\pi} \int_{\theta_3 = 0}^{2\pi}
             \sin^2\theta_1\sin\theta_2\, d\theta_1 d\theta_2 d\theta_3                                                                                                   \\
                                 & = & 2\pi^2.
          \end{array}
        \]
\end{enumerate}
\fincorrection

\finexercice
