\exercice{2938, quercia, 2010/03/08}

\enonce[Sphère de \( \Rr^3 \)]
Soient \( u,v \in \Cc \) tels que \( u+v \ne 0 \). On pose \( x = \frac{1+uv}{u+v} \), \( y = i\frac{1-uv}{u+v} \), \( z = \frac{u-v}{u+v} \).
\begin{enumerate}
  \item CNS sur \( u \) et \( v \) pour que \( x,y,z \) soient réels ?

  \item On suppose cette condition réalisée. Montrer que le point \( M(x,y,z) \) dans l'espace appartient à la sphère de centre \( O \) et de rayon 1.

  \item A-t-on ainsi tous les points de cette sphère ?
\end{enumerate}
\finenonce

\noindication

\correction
\begin{enumerate}
  \item \( z\in\Rr \Leftrightarrow \exists\ \alpha\in\Rr \) tq \( u = \alpha v \).\par \( x,y\in\Rr \Leftrightarrow \alpha = \frac1{|v|^2} \Leftrightarrow u = \frac 1{\bar v} \).

  \item

  \item Il manque seulement les deux pôles.
\end{enumerate}
\fincorrection

\finexercice
