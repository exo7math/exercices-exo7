\exercice{1459, legall, 1998/09/01}

\enonce
\begin{enumerate}
  \item Soient \( (E, \langle , \rangle ) \) un espace euclidien et \( \Vert . \Vert \) la norme associée ; \( n \in \Nn^* \), et \( v_1, \dots, v_n \in E \). Montrer l'inégalité : \( \displaystyle\Vert \sum_{i = 1}^n v_i\Vert^2 \le\ n\sum_{i = 1}^n \Vert v_i\Vert^2 \).

  \item Soient \( n \in \Nn^* \), \( x_1, \dots, x_n \in \Rr_+^* \) tels que \( \displaystyle \sum_{i = 1}^n x_i = 1 \). Montrer que \( \displaystyle \sum_{i = 1}^n
          \frac{1}{x_i} \ge n^2 \). Étudier le cas d'égalité.
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
