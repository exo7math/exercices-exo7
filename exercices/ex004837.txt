\exercice{4837, quercia, 2010/03/16}

\enonce[Boule unité non compacte]
Soit \( E = {\cal C}([0,2\pi]) \) muni de la norme \( \|.\|_2 \). Pour \( n \in \Nn \), on pose \( f_n(x) = \cos(nx) \).

\begin{enumerate}
  \item Calculer \( \|f_n-f_p\|_2 \) pour \( n,p \in \Nn \).

  \item En déduire que \( \overline B(0,1) \) n'est pas compacte.

\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
