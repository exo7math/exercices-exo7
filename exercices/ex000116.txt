\exercice{116, ridde, 1999/11/01}

\enonce
Écrire la négation des phrases suivantes :
\begin{enumerate}
  \item \( (\forall x) (\exists n) \mid (x \le n) \).

  \item \( (\exists M) \mid (\forall n) (\left|u_n\right| \le M) \).

  \item \( (\forall x) (\forall y) (xy = yx) \).

  \item \( (\forall x) (\exists y) \mid (yxy^{-1} = x) \).

  \item \( (\forall \epsilon>0) (\exists N \in \Nn) \mid (\forall n \ge N) (\left|u_n\right| < \epsilon) \).

  \item \( (\forall x \in \Rr) (\forall \epsilon >0) (\exists \alpha >0) \mid (\forall f \in \mathcal{F}) (\forall y \in \Rr) (\left|x-y\right|<\alpha \Rightarrow \left|f (x)-f (y)\right|< \epsilon) \).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
