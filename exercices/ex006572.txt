\exercice{6572, drutu, 2011/10/16}

\enonce
\begin{enumerate}
  \item Écrire une base de \( \Qq [i,\, \sqrt{2}] \) sur \( \Qq \).

  \item Dans cette base, écrire la matrice de l'application \( m_\alpha^E \), où \( \alpha = i+\sqrt{2} \). Déduire le polynôme minimal de \( \alpha \) dans \( \Qq[X] \) à partir de cette matrice.

  \item Retrouver le polynôme minimal de \( \alpha \) dans \( \Qq[X] \) par calcul direct.
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
