\exercice{1537, ortiz, 1999/04/01}

\enonce
\begin{enumerate}
  \item Soit \( E \) un espace vectoriel euclidien, \( f\in\mathcal{L}(E) \) un endomorphisme symétrique positif. Montrer que si \( x\in E \) alors \( (f(x)|x)\ge 0 \).

  \item Soit \( M = \left( m_{i,j}\right)_{i,j}\in \mathcal{M}_n(\Rr) \) symétrique positive. Montrer que pour tout \( i = 1,..,n,\;m_{ii}\ge 0 \) et \( \text{tr}(M)\ge 0 \).

  \item Soient \( A,\;B\in \mathcal{M}_n(\Rr) \) symétriques positives.
        \begin{enumerate}
          \item Montrer qu'il existe \( D\in \mathcal{M}_n(\Rr) \) diagonale et \( M\in \mathcal{M}_n(\Rr) \) symétrique positive telle que \( \text{tr}(AB) = \text{tr}(DM) \).

          \item En déduire que \( \text{tr}(AB) \le \text{tr}(A) \text{tr}(B) \).
        \end{enumerate}
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
