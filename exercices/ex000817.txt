\exercice{817, ridde, 1999/11/01}

\enonce
Soit \( f \in C^2 ([a, b], \Rr) \).
\begin{enumerate}
  \item Montrer que \( \int_a^bf (t)dt = \frac{b-a}2 (f (a) + f (b)) +
        \frac 12 \int_a^b f'' (x) (a-x) (b-x)dx \).

  \item En déduire un encadrement de \( \int_a^bf (t)dt \) si \( \forall x\in [a, b]
        \,\, m \le f'' (x) \le M \).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
