\exercice{1740, drutu, 2003/10/01}

\enonce
Soit \( x = (x_1, \cdots x_n) \in \Rr^n \). On pose
\[
  \|x\|_1 = \sum_{i = 1}^n |x_i| \quad ; \quad \|x\|_2 = \left(\sum_{i = 1}^n |x_i|^2\right)^{1/2}
\]
et
\[
  \|x\|_{\infty} = \sup \{ |x_i| : 1 \le i \le n\} .
\]
\begin{enumerate}
  \item Démontrer que \( \|\cdot\|_1 \) est une norme sur \( \Rr^n \).

  \item Démontrer que
        \[
          \|x\|_{\infty} \le \|x\|_2 \le \|x\|_1 \le n \|x\|_{\infty}
        \]
        et
        \[
          \|x\|_2 \le \sqrt{n}\|x\|_{\infty},
        \]
        pour tout \( x \in \Rr^n \). Discuter le cas \( n = 1 \).

  \item Représenter dans \( \Rr^2 \) la boule unité fermée
        \[
          B_{\|\cdot\|} = \{ x = (x_1,x_2) \in \Rr^2 \; ;\; \|x\| \le 1\}
        \]
        pour chacune des normes \( \|\cdot\|_1, \|\cdot\|_2 \) et \( \|\cdot\|_{\infty} \).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
