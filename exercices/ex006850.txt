\exercice{6850, gijs, 2011/10/16}

\enonce
Soit \( t \) un réel, \( \vert t\vert \le \pi \).
\begin{enumerate}
  \item On considère la série de fonctions holomorphes
        \[
          \sum_{k = -\infty}^{+\infty}\frac{(-1)^ke^{ikt}}{k^2-z^2}.
        \]
        Montrer que sa somme \( S(z) \) est une fonction méromorphe dans \( \Cc \).

  \item Soit \( \gamma_n \), \( n\ge 0 \), le chemin parcouru dans le sens positif dont l'image \( \gamma_n^* \) dans \( \Cc \) est le carré de sommets \( \left(n+\frac{1}{2}\right)(1+i) \), \( \left(n+\frac{1}{2}\right)(-1+i) \), \( \left(n+\frac{1}{2}\right)(-1-i) \), \( \left(n+\frac{1}{2}\right)(1-i) \).

        \begin{enumerate}
          \item Montrer que, quel que soit \( z \) vérifiant \( z = \left(n+\frac{1}{2}\right)+iy \), \( -\left(n+\frac{1}{2}\right)\le y\le n+\frac{1}{2} \), on a
                \[
                  \left\vert\frac{e^{itz}}{\sin{\pi z}}\right\vert\le 2
                \]
                et que, quel que soit \( z \) vérifiant \( z = x+i\left(n+\frac{1}{2}\right) \), \( -\left(n+\frac{1}{2}\right)\le x\le n+\frac{1}{2} \), on a
                \[
                  \left\vert\frac{e^{itz}}{\sin{\pi z}}\right\vert\le \frac{2}{1-e^{-\pi}}.
                \]
                En déduire que \( \left\vert\frac{e^{itz}}{\sin{\pi z}}\right\vert \) est bornée sur \( \gamma_n^* \).

          \item Soit \( a \) appartenant à \( \Cc\setminus \Zz \). On pose
                \[
                  f(z) = \frac{e^{itz}}{(z^2-a^2)\sin{\pi z}}.
                \]
                Montrer que l'on a
                \[
                  \lim_{n\to +\infty}\int_{\gamma_n}f(z)dz = 0.
                \]
        \end{enumerate}

  \item Calculer le résidu de \( f \) en chacun de ses pôles. En déduire la valeur de
        \[
          S(a) = \sum_{k = -\infty}^{+\infty}\frac{(-1)^ke^{ikt}}{k^2-a^2}.
        \]

  \item Calculer de même
        \[
          \sum_{k = -\infty}^{+\infty}\frac{(-1)^ke^{ikt}}{(k+a)^2}.
        \]
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
