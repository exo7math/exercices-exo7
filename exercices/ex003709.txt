\exercice{3709, quercia, 2010/03/11}

\enonce[Matrice à trou]
\begin{enumerate}
  \item Compléter la matrice \( A = \frac 17 \begin{pmatrix} 6  & 3 & . \\
                -2 & 6 & . \\
                3  & . & .
        \end{pmatrix} \) en une matrice orthogonale positive.

  \item Reconnaître l'application de matrice \( A \) dans la base canonique de \( \Rr^3 \).
\end{enumerate}
\finenonce

\noindication

\correction
\begin{enumerate}
  \item \( \frac 17 \begin{pmatrix} 6 & 3 & -2 \\ -2 &6 &3 \\ 3 &-2 &6
        \end{pmatrix} \).

  \item Rotation autour de \( (1,1,1) \) d'angle \( -\arccos(\frac{11}{14}) \).
\end{enumerate}
\fincorrection

\finexercice
