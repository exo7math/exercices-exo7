\exercice{3451, quercia, 2010/03/10}

\enonce[Calcul de déterminants]
Calculer les déterminants suivants :

\begin{enumerate}
  \item \( \begin{vmatrix}
          x & a & b & x \\
          a & x & x & b \\
          b & x & x & a \\
          x & b & a & x
        \end{vmatrix} \).

  \item \( \begin{vmatrix}
          a-b-c & 2a    & 2a    \\
          2b    & b-c-a & 2b    \\
          2c    & 2c    & c-a-b
        \end{vmatrix} \).

  \item \( \begin{vmatrix}
          a+b     & b+c     & c+a     \\
          a^2+b^2 & b^2+c^2 & c^2+a^2 \\
          a^3+b^3 & b^3+c^3 & c^3+a^3
        \end{vmatrix} \).

  \item \( \begin{vmatrix}
          a+b & ab & a^2+b^2 \\
          b+c & bc & b^2+c^2 \\
          c+a & ca & c^2+a^2
        \end{vmatrix} \).

  \item \( \begin{vmatrix}
          a^2 & b^2 & ab  \\
          b^2 & ab  & a^2 \\
          ab  & a^2 & b^2
        \end{vmatrix} \).

  \item \( \begin{vmatrix}
          a   & b      &        & (0) \\
          c   & \ddots & \ddots &     \\
              & \ddots & \ddots & b   \\
          (0) &        & c      & a
        \end{vmatrix} \).
        \label{tridiag}

  \item \( \begin{vmatrix}
          a      & b      & \dots  & b      \\
          b      & \ddots & (0)    & \vdots \\
          \vdots & (0)    & \ddots & b      \\
          b      & \dots  & b      & a
        \end{vmatrix} \).

        % Déterminants de Pascal

  \item \( \begin{vmatrix}
          C_n^0     & C_n^1     & \dots & C_n^p     \\
          C_{n+1}^0 & C_{n+1}^1 & \dots & C_{n+1}^p \\
          \vdots    & \vdots    &       & \vdots    \\
          C_{n+p}^0 & C_{n+p}^1 & \dots & C_{n+p}^p
        \end{vmatrix} \).

        % Déterminant ai+bi, bi

  \item \( \begin{vmatrix}
          a_1 + b_1 & b_1       & \dots  & \dots  & b_1       \\
          b_2       & a_2 + b_2 & b_2    & \dots  & b_2       \\
          \vdots    &           & \ddots &        & \vdots    \\
          \vdots    &           &        & \ddots & b_{n-1}   \\
          b_n       & \dots     & \dots  & b_n    & a_n + b_n
        \end{vmatrix} \).

        % Déterminant |ai-bj|

  \item \( \begin{vmatrix}
          a_1-b_1 & \dots & a_1-b_n \\
          \vdots  & \dots & \vdots  \\
          a_n-b_1 & \dots & a_n-b_n
        \end{vmatrix} \), \( (n \ge 3) \).

        % Déterminant circulant

  \item \( \begin{vmatrix}
          1      & 2      & \dots & n      \\
          2      & 3      & \dots & 1      \\
          \vdots & \vdots &       & \vdots \\
          n      & 1      & \dots & n-1
        \end{vmatrix} \).

        % Déterminant |i-j|

  \item \( \begin{vmatrix}
          0      & 1     & 2      & \dots  & n-1    \\
          1      & 0     & 1      &        & \vdots \\
          2      & 1     & 0      & \ddots & 2      \\
          \vdots &       & \ddots & \ddots & 1      \\
          n-1    & \dots & 2      & 1      & 0
        \end{vmatrix} \).

        Pour \ref{tridiag} : Chercher une relation de récurrence linéaire d'ordre 2. On notera \( \alpha \) et \( \beta \) les racines dans \( \Cc \) de l'équation caractéristique, et on exprimera le déterminant en fonction de \( \alpha \) et \( \beta \).
\end{enumerate}
\finenonce

\noindication

\correction
Notation : \( \varepsilon_n = \begin{cases}1  & \text{si}n \equiv 0 \text{ ou }1 (\mathrm{mod}\, 4) \\
             -1 & \text{sinon.}\end{cases} \)

\begin{enumerate}
  \item \( (b-a)^2(a+b+2x)(a+b-2x) \).

  \item \( (a+b+c)^3 \)

  \item \( 2abc(a-b)(b-c)(c-a) \)

  \item \( (a-b)(b-c)(c-a)(ab+ac+bc) \).

  \item \( -(a^3-b^3)^2 \).

  \item \( \frac{\alpha^{n+1} - \beta^{n+1}}{\alpha - \beta} \) où \( \alpha \ne \beta \) sont les racines de \( X^2 - aX + bc = 0 \).
        \par \( (n+1)\left(\frac a2\right)^n \) si \( \alpha = \beta \).

  \item \( a^{n-3}(a-b)\bigl(a^2 + ab -2(n-2)b^2\bigr) \).

  \item 1.

  \item \( a_1 a_2 \dots a_n\left(1+\frac{b_1}{a_1}+ \dots + \frac{b_n}{a_n}\right) \).

  \item 0

  \item \( \varepsilon_n \frac{n^{n-1}(n+1)}2 \).

  \item \( (-1)^{n-1}(n-1)2^{n-2} \).
\end{enumerate}
\fincorrection

\finexercice
