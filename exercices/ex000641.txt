\exercice{641, bodin, 1998/09/01}

\enonce
\begin{enumerate}
  \item Soit \( f \) une fonction continue sur \( ]a,b[ \) telle que \( f(]a,b[)\subset [a,b] \). Montrer, par considération de \( \phi(x) = f(x)-x \), qu'il existe \( c \) dans \( [a,b] \) tel que \( f(c) = c \).

  \item Soit \( f \) une fonction continue sur \( [0,1] \) telle que \( f(0) = f(1) \). Montrer qu'il existe \( c \) dans \( [0,\frac{1}{2}] \) tel que \( f(c) = f(c+\frac{1}{2}) \).

  \item Un mobile parcourt, à vitesse continue, une distance \( d \) en une unité de temps. Montrer qu'il existe un intervalle d'une demi-unité de temps pendant lequel il parcourt une distance \( \frac{d}{2} \).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
