\exercice{1881, maillot, 2001/09/01}

\enonce
\begin{enumerate}
  \item Montrer que \( \forall p\ge 1 \), l'application \( \left(\begin{array}{cccl}
          N_p: & \Rr^n & \longrightarrow & \Rr                                  \\
               & x     & \longmapsto     & (\sum_{k = 1}^n |x_k|^p)^\frac{1}{p}
        \end{array}\right) \) est une norme (on utilisera la convexité de \( x^p \)).

  \item Pour \( x\in\Rr^n \) fixé, montrer que \( \lim_{p\rightarrow +\infty}
        N_p(x) = \max_{1\le i\le n} |x_i| \), et que cela définit une norme, appelée {\bf norme infinie}, et notée \( N_\infty \).

  \item Établir les inégalités suivantes :
        \[
          \forall x \in \Rr^n,\; N_\infty(x)\le N_1(x)\le\sqrt{n}N_2(x)\le nN_\infty(x).
        \]
        Que peut-on en déduire ?

  \item Dessiner les boules unités des normes 1,2, et \( \infty \) dans \( \Rr^2 \).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
