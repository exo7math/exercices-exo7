\exercice{5980, tumpach, 2010/11/11}

\enonce
\textbf{Théorème}
Soit \( \{\varphi_n\}_{n\in\Nn} \) une suite de fonctions de \( \Rr^n \) dans \( \Rr \) telles que :
\begin{itemize}
  \item[(i)] \( \int_{\Rr^n} \varphi_n = 1 \) \item[(ii)] il existe une constante \( K > 0 \) telle que \(  \sup_{n\in\Nn} \int_{\Rr^n} |\varphi_n|(x) \,dx \le K \) \item[(iii)] Pour tout \( \varepsilon>0 \), on a \( \lim_{n\rightarrow +\infty}\int_{\|x\|>\varepsilon}
        |\varphi_n(x)|\,dx = 0 \).
\end{itemize}
Alors pour tout \( f\in L^p(\Rr^n) \), \( 1\le p <+\infty \), \( \lim_{n\rightarrow+\infty}\|\varphi_n*f - f\|_p = 0 \).

\bigskip

Le but de cet exercice est de démontrer ce théorème.

Soit \( \{\varphi_n\}_{n\in\Nn} \) une suite de fonctions vérifiant les hypothèses (i), (ii) et (iii) du théorème, et soit \( 1\le p <+\infty \).
\begin{enumerate}
  \item En notant \( q \) l'exposant conjugué de \( p \) (\( \frac{1}{p} +
        \frac{1}{q} = 1 \)), et en utilisant l'inégalité de Hölder pour la mesure \( d\nu(x) = |\varphi_n|(x)\,dx \), montrer que
        \[
          |\varphi_n * f - f|^p(x) \le \left(\int_{\Rr^n}
          |\varphi_n|(x)\,dx \right)^{\frac{p}{q}} \left(\int_{\Rr^n}
          |f(x - y) - f(x)|^p |\varphi_n|(y)\,dy\right).
        \]

  \item En déduire que
        \[
          \|\varphi_n * f - f\|_p^p \le K^{\frac{p}{q}}
          \int_{\Rr^n} \|\tau_yf - f\|_p^p |\varphi_n(y)|\,dy.
        \]

  \item Soit \( \delta>0 \), montrer que
        \[
          \|\varphi_n * f - f\|_p^p \le K^{\frac{p}{q}}\left(\sup_{|y|\le\delta}\|\tau_yf - f\|_p^p + 2^p\|f\|_p^p\int_{|y|>\delta} |\varphi_n(y)|\,dy \right).
        \]

  \item En déduire le théorème cherché.
\end{enumerate}
\finenonce

\noindication

\correction
Soit \( \{\varphi_n\}_{n\in\Nn} \) une suite de fonctions vérifiant les hypothèses (i), (ii) et (iii) du théorème, et soit \( 1\le p <+\infty \).
\begin{enumerate}
  \item En notant \( q \) l'exposant conjugué de \( p \) (\( \frac{1}{p} +
        \frac{1}{q} = 1 \)), on a
        \[
          \begin{array}{lcl}
            |\varphi_n * f - f|^p(x) & = & |\int_{\Rr^n} f(x - y)
            \varphi_n(y)\,dy - f(x) \int_{\Rr^n}\varphi_n(y)\,dy|^p \\
                                     &   &                          \\ & \le & \left(\int_{\Rr^n} |f(x - y) - f(x)| |\varphi_n(y)|\,dy \right)^p.
          \end{array}
        \]
        En utilisant l'inégalité de Hölder pour la mesure \( d\nu(x) = |\varphi_n|(x)\,dx \), on a
        \[
          \begin{array}{lcl}
            |\varphi_n * f - f|^p(x) & \le & \left(\int_{\Rr^n}
            1^q\,d\nu(y) \right)^{\frac{p}{q}}\left(
            \int_{\Rr^n}|f(x - y) - f(x)|^p\,d\nu(y)\right)^{\frac{p}{p}}                          \\
                                     & \le & \left(\int_{\Rr^n} |\varphi_n|(y)\,dy
            \right)^{\frac{p}{q}} \left(\int_{\Rr^n} |f(x - y) - f(x)|^p |\varphi_n|(y)\,dy\right) \\ & \le & K^{\frac{p}{q}}\left(\int_{\Rr^n} |f(x - y) - f(x)|^p |\varphi_n|(y)\,dy\right).
          \end{array}
        \]

  \item On en déduit que
        \[
          \begin{array}{lcl}
            \|\varphi_n * f - f\|_p^p & \le & K^{\frac{p}{q}}\int_{x\in\Rr^n}\left(\int_{y\in\Rr^n}|f(x - y) - f(x)|^p |\varphi_n|(y)\,dy\right)\,dx
          \end{array}
        \]
        D'après le théorème de Tonelli :
        \[
          \begin{array}{lcl}
            \|\varphi_n * f - f\|_p^p & \le & K^{\frac{p}{q}}\int_{y\in\Rr^n}\left(\int_{x\in\Rr^n}|f(x - y) - f(x)|^p \,dx\right)|\varphi_n|(y)\,dy \\ & \le & K^{\frac{p}{q}}
            \int_{\Rr^n} \|\tau_yf - f\|_p^p |\varphi_n(y)|\,dy.
          \end{array}
        \]

  \item Soit \( \delta>0 \), on a
        \[
          \begin{array}{lcl}
             &     & \|\varphi_n * f - f\|_p^p                                                                                         \\ & \le & K^{\frac{p}{q}}\left(
            \int_{|y|\le \delta}\|\tau_yf - f\|_p^p |\varphi_n(y)|\,dy +
            \int_{|y|>\delta} \|\tau_yf - f\|_p^p |\varphi_n(y)|\,dy
            \right)                                                                                                                    \\ & \le & K^{\frac{p}{q}}\left(\sup_{|y|\le\delta}\|\tau_yf - f\|_p^p
            \int_{|y|\le \delta} |\varphi_n(y)|\,dy + \int_{|y|>\delta}
            \left(\|\tau_yf\|_p + \|f\|_p\right)^p |\varphi_n(y)|\,dy
            \right)                                                                                                                    \\& \le & K^{\frac{p}{q}}\left(K\,\sup_{|y|\le\delta}\|\tau_yf - f\|_p^p + \left(2\|f\|_p\right)^p \int_{|y|>\delta}
            |\varphi_n(y)|\,dy \right)                                                                                                 \\
             & \le & K^{\frac{p}{q}}\left(K\,\sup_{|y|\le\delta}\|\tau_yf - f\|_p^p + 2^p\|f\|_p^p\int_{|y|>\delta} |\varphi_n(y)|\,dy
            \right).
          \end{array}
        \]

  \item Soit \( \varepsilon>0 \). Par continuité des translations dans \( L^p(\Rr^n) \) (cf l'exercice précédent), il existe un \( \delta>0 \) tel que
        \[
          |y|\le\delta\quad \Rightarrow \|\tau_yf - f\|_p^p <
          \frac{K^{-\left(\frac{p}{q}+1\right)}}{2}  \varepsilon.
        \]
        D'après l'hypothèse (iii), il existe un \( N\in\Nn \) tel que pour \( n>N \), on a
        \[
          \int_{|y|>\delta} |\varphi_n(y)|\,dy <
          \frac{K^{-\frac{p}{q}}}{2^{p+1}\|f\|_p^p}  \varepsilon.
        \]
        Ainsi pour tout \( n>N \),
        \[
          \|\varphi_n * f - f\|_p^p < \varepsilon,
        \]
        i.e. \( \lim_{n\rightarrow+\infty}\|\varphi_n * f - f\|_p = 0 \).
\end{enumerate}
\fincorrection

\finexercice
