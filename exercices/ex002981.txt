\exercice{2981, quercia, 2010/03/08}

\enonce[Morphismes de \( \Qq \) additif]
Déterminer tous les morphismes de
\begin{enumerate}
  \item \( (\Qq,+) \) dans \( (\Qq,+) \).

  \item \( (\Qq,+) \) dans \( (\Zz,+) \).

  \item \( (\Qq,+) \) dans \( (\Qq^*,\times) \).
\end{enumerate}
\finenonce

\noindication

\correction
\begin{enumerate}
  \item \( x \mapsto ax \), \( a\in\Qq \).

  \item \( x \mapsto 0 \).

  \item \( x \mapsto 1 \).
\end{enumerate}
\fincorrection

\finexercice
