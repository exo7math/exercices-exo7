\exercice{4857, quercia, 2010/03/16}

\enonce[\( f \) et \( g \) colinéaires]
Soient \( {f,g} : {I\subset \Rr} \to E \) deux fonctions vectorielles de classe \( \mathcal{C}^1 \).

\begin{enumerate}
  \item On suppose : \( \forall\ t \in I,\ f(t) \) et \( g(t) \) sont colinéaires. Est-ce que \( f'(t) \) et \( g'(t) \) sont colinéaires ?

  \item On suppose : \( \forall\ t \in I,\ f'(t) \) et \( g'(t) \) sont colinéaires. Existe-t-il \( \vec c \in E \) tel que \( f-\vec c \) et \( g \) soient colinéaires ?

\end{enumerate}
\finenonce

\noindication

\correction
\begin{enumerate}
  \item Non : \( f(t) = (t,t^2) \), \( g(t) = (1,t) \).

  \item Non : \( f(t) = \left(\frac{t^2}2, \frac{t^3}3\right) \), \( g(t) = \left(t,\frac{t^2}2\right) \).
\end{enumerate}
\fincorrection

\finexercice
