\exercice{6636, queffelec, 2011/10/16}

\enonce
\begin{enumerate}
  \item Trouver le rayon de convergence de la série \( \sum a_nz^n \) où \( a_{2n+1} = a^{2n+1} \) et \( a_{2n} = b^{2n} \) avec \( 0<a,b<1 \).

  \item Pour la série \( \sum a_nz^n \), où \( a_{2n} = a^nb^n \) et \( a_{2n+1} = a^nb^{n+1} \) avec \( a,b>0 \), comparer l'inverse du rayon, \( R^{-1} \), avec \( \limsup \frac{a_{n+1}}{a_n} \).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
