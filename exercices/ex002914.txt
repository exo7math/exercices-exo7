\exercice{2914, quercia, 2010/03/08}

\enonce[Nombre d'opérations]
\begin{enumerate}
  \item Combien existe-t-il d'opérations internes sur un ensemble à \( n \) éléments ?

  \item Combien sont commutatives ?

  \item Combien ont un élément neutre ?

  \item Combien sont commutatives et ont un élément neutre ?
\end{enumerate}
\finenonce

\noindication

\correction
\begin{enumerate}
  \item \( n^{n^2} \).

  \item \( n^{n(n+1)/2} \).

  \item \( n\times n^{(n-1)^2} \).

  \item \( n\times n^{n(n-1)/2} \).
\end{enumerate}
\fincorrection

\finexercice
