\exercice{4993, quercia, 2010/03/17}

\enonce[Courbes en polaires]
Construire les courbes en polaires suivantes :
\begin{enumerate}
  \item \( \rho = \dfrac{\cos\frac{\theta}{2}}{1+\sin\theta} \)

  \item \( \rho = \dfrac{\cos 2\theta}{\cos\theta} \)

  \item \( \rho = \dfrac{\sin\theta}{2\cos\theta-1} \)

  \item \( \rho = \dfrac1{\cos\theta+\sin2\theta} \)

  \item \( \rho = \cos\theta+\dfrac{1}{\cos\theta} \)

  \item \( \rho = \dfrac{\cos2\theta}{2\cos\theta-1} \)

  \item \( \rho = \cos\dfrac{\theta}{3} \)

  \item \( \rho = 1+\sin3\theta \)

  \item \( \rho = \dfrac{1}{\sqrt{\theta}} \)

  \item \( \rho = \ln\theta \)

  \item \( \rho = \frac{\sin\theta}{\theta} \)
\end{enumerate}
\finenonce
