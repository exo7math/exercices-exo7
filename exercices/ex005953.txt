\exercice{5953, tumpach, 2010/11/11}

\enonce
Soit \ \( f\in
\mathcal{L}^1(\Rr ) \). Sa \emph{transformée de Fourier}
est la fonction \ \( \hat{f}:\Rr \rightarrow \Cc \) \ \ définie par
\[
  \hat{f}(y):=\int_\Rr\text{e}^{-ixy}f(x)dx,
\]
montrer que
\begin{enumerate}
  \item \( \hat{f} \) \ est continue,

  \item \( \hat{f} \) \ est bornée et \ \( \sup |\hat{f}|\le
        \left\| f\right\|_{L^1} \) \ (=\( \int_\Rr\left| f(x)\right| dx \)),

  \item Si \( x \to xf(x) \) est intégrable, alors \( \hat{f} \) \ est dérivable et on a
        \[
          \frac{d}{dy}\ \hat{f}\ = \widehat{-ixf(x)}.
        \]
\end{enumerate}
\finenonce

\noindication

\correction
\begin{enumerate}
  \item Notons \( g(x,y) = e^{-ixy} f(x) \). Alors,
        \begin{enumerate}
          \item[{i.)}] pour tout \( y\in \Rr \), la fonction \( x\mapsto g(x,y) \) est mesurable; \item[{ii.)}] pour presque tout \( x\in
                \Rr \) (pour tout les \( x\in \Rr \) tels que \( f(x) \) est finie) la fonction \( y\mapsto g(x,y) \) est continue pour tout \( y\in
                \Rr \); \item[{iii.)}] \( |g(x,y)| = |e^{-ixy} f(x)|\le |f(x)| \) et \( |f|\in \mathcal{L}^1(\Rr,\Rr_+) \).
        \end{enumerate}
        On doit montrer que pour tout suite \( \{y_n\}_{n\in\Nn} \) convergeant vers \( y \), on a \( \lim_{n\rightarrow +\infty}
        \hat{f}(y_n) = \hat{f}(y) \). Posons \( g_n(x) = g(x, y_n) \). D'après le théorème de convergence dominée,
        \[
          \lim_{n\rightarrow +\infty} \hat{f}(y_n)  :=  \lim_{n\rightarrow +\infty} \int_\Rr g_n(x)\,dx = \int_\Rr g(x,y) dx = : \hat{f}(y).
        \]
        Ainsi \( \hat{f} \) est continue.

  \item Pour tout \( y\in \Rr \), \( |\hat{f}(y)|\le
        \displaystyle\int_\Rr|e^{-ixy} f(x)|dx\le
        \displaystyle\int_\Rr| f(x)|dx = \|f\|_{L_1} \) et donc
        \[
          \sup|\hat{f}|\le\|f\|_{L_1}.
        \]

  \item Soit \( g(x,y) = e^{-ixy} f(x) \). Alors, on a
        \begin{enumerate}
          \item[{i.)}] pour tout \( y\in \Rr \), la fonction \( x\mapsto g(x,y) \) est intégrable;

          \item[{ii.)}] pour presque tout \( x\in \Rr \) la fonction \( y\mapsto g(x,y) \) est dérivable pour tout \( y\in \Rr \);

          \item[{iii.)}] \( |\frac{\partial g(x,y)}{\partial y}| = |-ix e^{-ixy}
                f(x)|\le |x f(x)| \) avec \( x\mapsto xf(x) \) intégrable.
        \end{enumerate}
        Ainsi, d'après l'exercice \ref{derivation} (le théorème de dérivation sous le signe \( \int \)), on a
        \[
          \frac{d}{dy}\hat{f} = \int_\Rr e^{-ixy}(-ix f(x))dx = \widehat{-iyf(y)}.
        \]
\end{enumerate}
\fincorrection

\finexercice
