\exercice{1107, barraud, 2003/09/01}

\enonce
On considère l'espace \( \mathcal{M}_n(\Kk) \) des matrices carrées \( n\times n \) à coefficients dans le corps \( \Kk \). On rappelle que la \emph{trace} \( \mathrm{tr}(A) \) d'une matrice \( A\in\mathcal{M}_n(\Kk) \) est la somme de ses coefficients diagonaux.

Pour une matrice \( M \) donnée, on note \( \alpha_M \) l'application définie par
\[
  \forall X\in\mathcal{M}_n(\Kk),\quad \alpha_M(X) = \mathrm{tr}(MX).
\]
\begin{enumerate}
  \item Vérifier que \( \forall M\in\mathcal{M}_n(\Kk),\quad \alpha_M\in(\mathcal{M}_n(\Kk))^{*} \).

        \bigskip

        On note \( \phi \) l'application suivante :
        \[
          \phi :
          \begin{array}{ccc}
            \mathcal{M}_n(\Kk) & \rightarrow & (\mathcal{M}_n(\Kk))^{*} \\[3mm] M & \mapsto & \alpha_M
          \end{array}
        \]

  \item Étudier l'injectivité et la surjectivité de \( \phi \).

  \item En déduire que pour toute forme linéaire \( \alpha\in(\mathcal{M}_n(\Kk))^{*} \), il existe une matrice \( A\in\mathcal{M}_n(\Kk) \) telle que :
        \[
          \forall X\in\mathcal{M}_n(\Kk),\quad \alpha(X) = \mathrm{tr}(AX).
        \]

  \item Déterminer toutes les formes linéaires \( \alpha\in(\mathcal{M}_n(\Kk))^{*} \) telles que
        \[
          \forall (X,Y)\in(\mathcal{M}_n(\Kk))^2,\quad \alpha(XY) = \alpha(YX).
        \]
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
