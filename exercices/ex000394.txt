\exercice{394, ortiz, 1999/04/01}

\enonce
\begin{enumerate}
  \item Montrer que les polynômes \( X-1 \) et \( X-2 \) sont premiers entre eux et en déduire \( d = \text{pgcd} ((X-1)^2,(X-2)^3) \) et des polynômes \( U \) et \( V \) tels que
        \[
          {U(X-1)^2+V(X-2)^3 = d.}
        \]

  \item Déterminer le polynôme \( P \), de degré minimal, tel que le reste de la division euclidienne de \( P \) par \( (X-1)^2 \) est \( 2X \) et le reste de la division euclidienne de \( P \) par \( (X-2)^3 \) est \( 3X \).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
