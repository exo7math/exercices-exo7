\exercice{6672, queffelec, 2011/10/16}

\enonce
Soit \( U \) un ouvert de \( \Cc \) contenant \( \bar D \) et \( f\in H(U) \). On note \( \gamma \) le paramétrage de \( \partial D \) par \( t\in [0,2\pi]\to e^{it} \).
\begin{enumerate}
  \item Calculer \( I_1 = \displaystyle\int_{\gamma} (2+z+\frac{1}{z})\frac{f(z)}{z}\ dz \) et \( I_2 = \displaystyle\int_{\gamma} (2-z-\frac{1}{z})\frac{f(z)}{z}\ dz \).

  \item En déduire la valeur de \( \displaystyle\int_0^{2\pi}
        f(e^{i\theta})\cos^2(\frac{\theta}{2})\ d\theta \) et \( \displaystyle\int_0^{2\pi}
        f(e^{i\theta})\sin^2(\frac{\theta}{2})\ d\theta \).

  \item Pour \( |a|\not=1 \), évaluer \( I(a) = \displaystyle\frac{1}{2i\pi}\int_{\gamma}
        \frac{\overline{f(z)}}{z-a}\ dz \).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
