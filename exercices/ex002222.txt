\exercice{2222, matos, 2008/04/23}

\enonce[Taille des éléments dans l'élimination de Gauss]
Notons \( \tilde{A}_k \) la matrice carrée d'ordre \( (n-k+1) \) formée des éléments \( a_{ij}^k , k\le i,j \le n \) de la matrice \( A_k = (a_{ij}^k) \) obtenue comme résultat de la \( (k-1) \)--ème étape de l'élimination de Gauss. On suppose \( A = A_1 \) symétrique définie positive.
\begin{enumerate}
  \item Notant \( (. , .) \) le produit scalaire euclidien et \( v'\in\Rr^{n-k} \) le vecteur formé par les \( (n-k) \) dernières composantes d'un vecteur \( v = (v_i)_{i = k}^n \in \Rr^{n-k+1} \) quelconque, établir l'identité
        \[
          (\tilde{A}_k v,v) = (\tilde{A}_{k+1} v', v') + \frac{1}{a_{kk}^k} \left|a_{kk}^k v_k +\sum_{i = k+1}^n a_{ik}^k v_i\right|^2 .
        \]

  \item Montrer que chaque matrice \( \tilde{A}_k \) est symétrique définie positive.

  \item Établir les inégalités suivantes:
        \[
          0< a_{ii}^{k+1} \le a_{ii}^k ,\ \ \ k+1\le i \le n
        \]
        \[
          \max_{k+1\le i \le n} a_{ii}^{k+1} = \max_{k+1\le i,j \le n}\left|a_{ij}^{k+1}\right| \le \max_{k \le i,j \le n}\left|a_{ij}^k\right| = \max_{k\le i\le n}a_{ii}^k
        \]
\end{enumerate}
\finenonce

\noindication

\correction
\begin{enumerate}
  \item À la k-ème étape de l'élimination de Gauss, l'élément \( a_{ij}^{k+1} \) est donné par
        \[
          a_{ij}^{k+1} = a_{ij}^k-\frac{a_{kj}^ka_{ik}^k}{a_{kk}^k} \quad k+1\le i,j\le n
        \]
        et on remarque immédiatement par récurrence que toutes les matrices \( \tilde{A}_k \) sont symétriques. On a

        \( (\tilde{A}_{k+1} v',v') = \sum_{i = k+1}^n v_i (\sum_{j = k+1}^n a_{ij}^kv_j) - \frac{1}{a_{kk}^k}(\sum_{i = k+1}^n a_{ik}^k v_i)^2 \)

        \( (\tilde{A}_k v,v) = \sum_{i = k+1}^nv_i(\sum_{j = k+1}^n a_{ij}^kv_j) +\sum_{i = k+1}^n (a_{ik}^k +a_{ki}^k )v_iv_k +a_{kk}^kv_k^2 \)

        Par symétrie \( a_{ik}^k = a_{ki}^k \) et donc

        \( (\tilde{A}_kv,v) = (\tilde{A}_{k+1}v',v') +\frac{1}{a_{kk}^k} [(\sum_{i = k+1}^n a_{ik}^k v_i)^2 + 2v_k\sum_{i = k+1}^n a_{ik}^kv_i a_{kk}^k +(a_{kk}^k)^2v_k^2] =  \)

        \[
          (\tilde{A}_{k+1}v',v') +\frac{1}{a_{kk}^k} [a_{kk}^k v_k +\sum_{i = k+1}^n a_{ik}^k v_i]^2
        \]

  \item Faisons un raisonnement par récurrence
        \begin{itemize}
          \item \( \tilde{A}_1 \) est symétrique définie positive;

          \item Par hypothèse supposons que \( \tilde{A}_k \) est définie positive;

          \item Supposons par l'absurde que \( \tilde{A}_{k+1} \) ne soit pas définie positive: alors \( \exists v'\neq 0:\quad (\tilde{A}_{k+1}v',v')\le 0 \). On définit le vecteur \( v\in\Rr^{n-k+1} \) par:
                \begin{itemize}
                  \item \( v_i = v'_i,\quad k+1\le i\le n \)

                  \item \( v_k \) est solution de \( a_{kk}^k v_k +\sum_{i = k+1}^n a_{ik}^k v_i = 0 \)
                \end{itemize}
                Alors \( (\tilde{A}_kv,v) = 0 \) et \( v\neq 0 \); donc \( \tilde{A}_k \) n'est pas définie positive, ce qui contredit l'hypothèse de récurrence.
        \end{itemize}

  \item Première inégalité: en utilisant la relation d'élimination on obtient: \( a_{ii}^{k+1} = a_{ii}^k -\frac{\left|a_{ki}^k\right|^2}{a_{kk}^k} \)
        \begin{itemize}
          \item une matrice définie positive a tous ses éléments diagonaux strictement positifs, donc \( a_{ii}^{k+1}>0 \)

          \item \( \frac{\left|a_{ki}^k\right|^2}{a_{kk}^k} \ge 0, \quad k+1\le i\le n \)
        \end{itemize}
        donc \( a_{ii}^{k+1}\le a_{ii}^k, k+1 \le i \)

        Deuxième inégalité: supposons qu'il existe un élément \( a_{ij}^k, i<j \) tel que \( \left|a_{ij}^k\right| \ge \max_{k\le l \le n} a_{ll}^k \). On considère le vecteur \( v\neq 0 \) défini par
        \[
          v_i = 1, v_j = -\mbox{sign} (a_{ij}^k), v_l = 0\quad l\neq i,j
        \]
        Alors
        \[
          (\tilde{A}_k v,v) = ( a_{ii}^k -\left|a_{ij}^k\right|)-(\left|a_{ij}^k\right| -a_{jj}^k) \le 0
        \]
        ce qui est impossible. Donc
        \[
          \max_{1\le i,j\le n}\left|a_{ij}^k\right| = \max_{1\le i \le n}\left|a_{ii}^k\right|
        \]
\end{enumerate}
\fincorrection

\finexercice
