\exercice{5946, tumpach, 2010/11/11}

\enonce
Soit \( f\in
\mathcal{L}^1(\Rr) \). Que vaut la limite
\[
  \lim_{n\to\infty} \int_\Rr f(x)\cos^n(\pi x) d\lambda (x) \ ?
\]
\finenonce

\noindication

\correction
Soit \( f\in
\mathcal{L}^1(\Rr) \). Comme \( \cos(\pi x)<1 \) si \( x\notin
\Zz \), \( \cos^n(\pi x)\rightarrow 0 \) lorsque \( n\rightarrow\infty \) presque partout (pour tout \( x\in
\Rr\setminus\Zz \)). Notons \( f_n(x) = f(x) \cos^n(\pi x) \). Alors, pour tout \( n\in \Nn \) on a \( |f_n(x)|\le |f(x)| \) et comme \( |f|\in \mathcal{L}^1(\Rr) \), par le théorème de convergence dominée de Lebesgue,
\[
  \lim\limits_{n\rightarrow\infty}\int_\Rr f(x)\cos^n(\pi x)d\lambda(x) =
  \lim\limits_{n\rightarrow\infty}\int_\Rr f_n(x)d\lambda(x) = \int_\Rr
  \lim\limits_{n\rightarrow\infty}f_n(x)d\lambda(x) = 0.
\]
\fincorrection

\finexercice
