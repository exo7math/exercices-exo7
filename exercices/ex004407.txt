\exercice{4407, quercia, 2010/03/12}

\enonce[Morphismes \( (\Rr,+) \to (\Cc,*) \)]
Soit \( f : \Rr \to {\Cc^*} \) telle que : \( \forall\ x,y\in\Rr,\ f(x+y) = f(x)f(y) \).

\begin{enumerate}
  \item Si \( f \) est dérivable, montrer qu'il existe \( \lambda \in \Cc \) tel que : \( \forall\ x \in \Rr,\ f(x) = e^{\lambda x} \).

  \item Obtenir le même résultat si \( f \) est seulement supposée continue (prendre une primitive, \( F \), de \( f \) et montrer qu'elle est de classe \( \Cc^2 \)).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
