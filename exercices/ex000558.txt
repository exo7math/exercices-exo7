\exercice{558, cousquer, 2003/10/01}

\enonce
\begin{enumerate}
  \item Étudier dans \( \Cc \) une suite \( (u_n) \) telle que \( \forall n \in\Nn^\ast,\;u_{n+1} = u_n^2 \). Discuter suivant \( u_0 \).

  \item On considère dans \( \Cc \) une suite \( (v_n) \) telle que \( \forall n,\; v_{n+1} = \frac{1}{2}\bigl(v_n+\frac{A}{v_n}\bigr) \) où \( A \) est un nombre complexe non nul donné. Étudier l'existence et la convergence de cette suite suivant les valeurs de \( v_0 \). On pourra noter \( a \) une des racines carrées de \( A \) et poser \( w_n = \frac{v_n-a}{v_n+a} \).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
