\exercice{3272, quercia, 2010/03/08}

\enonce[Ensemble image d'une fonction rationelle]
\label{Imf}
Soit \( F \in {\Cc(X)} \). Étudier \( F( \Cc\setminus \{\text{pôles}\} ) \).
\finenonce

\noindication

\correction
\( F = \frac PQ \). Si \( P = \lambda Q \) : \( \Im F = \{\lambda\} \).\par Si \( P = \lambda Q + \mu \) : \( \Im F = \Cc \setminus \{\lambda\} \).\par Sinon, \( \Im F = \Cc \).
\fincorrection

\finexercice
