\exercice{6503, drutu, 2011/10/16}

\enonce
\begin{enumerate}
  \item Montrer qu'il existe un morphisme d'anneaux et un seul de \( \Zz /4\Zz \) à \( \Zz /2\Zz \), de \( \Zz /12\Zz \) à \( \Zz /3\Zz \), de \( \Zz /12\Zz \) à \( \Zz /4\Zz \).

  \item Montrer qu'il n'existe pas de morphisme d'anneaux de \( \Zz /3\Zz \) à \( \Zz /4\Zz \).

  \item \( m \) et \( n \) étant des entiers positifs, trouver des conditions pour qu'il existe un morphisme d'anneaux de \( \Zz /m\Zz \) à \( \Zz /n\Zz \).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
