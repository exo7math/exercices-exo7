\exercice{2557, tahani, 2009/04/01}

\enonce
\begin{enumerate}
  \item Pour chacune des équations suivantes où \( y = y(x) \) est réelle de variable réelle, décrire les solutions en précisant leur intervalle maximal de définition et dessiner les trajectoires:
        \[
          \mbox{ (i) } y' = e-y \mbox{ (ii) } y'-y = e^x \mbox{ (iii) }
          xy'-2y = 0.
        \]

  \item Quelles sont les courbes isoclines de l'équation \( y' = y^2-x \); en déduire l'allure des trajectoires.
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
