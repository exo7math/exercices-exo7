\exercice{3018, quercia, 2010/03/08}

\enonce[Fonctions trigonométriques]
On pose\\ \( A = \left\{ f : \Rr \to \Rr
\text{ de la forme } f(x) = a_0 + \sum_{k = 1}^n a_k\cos(kx),\ n \in
\Nn,\ a_i \in \Rr \right\} \).
\begin{enumerate}
  \item Montrer que \( A \) est un sous-anneau de \( \Rr^\Rr \).

  \item Soit \( f\in A \). Montrer que si \( f = 0 \), alors les coefficients \( a_k \) sont tous nuls \( \left({\text{calculer } \int_{t = 0}^{2\pi} f(t)\cos(nt)\,d t}\right) \).

  \item En déduire que \( A \) est intègre.
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
