\exercice{3210, quercia, 2010/03/08}

\enonce[Degré minimal dans la formule de Bézout]
Soient \( P,Q \in { K[X]} \) non nuls et \( D = \text{pgcd}(P,Q) \).
\begin{enumerate}
  \item Démontrer qu'il existe \( U,V \in { K[X]} \) uniques tels que : \( \begin{cases} UP + VQ = D                     \\
          \deg U < \deg Q - \deg D \hfill \\
          \deg V < \deg P - \deg D .\end{cases} \)

  \item Montrer que la méthode des divisions euclidiennes fournit \( U \) et \( V \).

\end{enumerate}
\finenonce
