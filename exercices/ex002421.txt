\exercice{2421, drutu, 2007/10/01}

\enonce
Soit \( X \) un espace muni d'une métrique \( \mathrm{dist} : X\times X \to
\Rr_+ \).
\begin{enumerate}
  \item Montrer que si \( f:\Rr_+ \to \Rr_+ \) est une fonction croissante telle que \( f(0) = 0 \) et \( f(x+y)\le f(x) +f(y) \) alors \( \mathrm{dist}_f (x,y) = f(\mathrm{dist} (x,y)) \) est une métrique sur \( X \).

  \item Montrer que
        \[
          \mathrm{dist}'(x,y) = \frac{\mathrm{dist}(x,y)}{1+\mathrm{dist} (x,y)}\, ,\; \forall x,y,
        \]
        est une métrique sur \( X \).

  \item Montrer que les métriques \( \mathrm{dist} \) et \( \mathrm{dist}' \) sont topologiquement équivalentes.
\end{enumerate}
\finenonce

\noindication

\correction
\begin{enumerate}
  \item On vérifie facilement les trois propriétés de métrique.

  \item Soit \( f:\Rr_+ \to \Rr_+ \, ,\, f(x) = \frac{x}{x+1} = 1-\frac{1}{x+1} \). On a que \( f(0) = 0 \) et \( f'(x) = \frac{1}{(x+1)^2} \), donc la fonction \( f \) est croissante sur \( \Rr_+ \). L'inégalité \( f(x+y)\le f(x)+f(y) \) pour \( x,y\in \Rr_+ \) est équivalente à
        \[
          \frac{1}{x+y+1}\ge \frac{1}{x+1}+\frac{1}{y+1}-1
          \Leftrightarrow \frac{1}{x+y+1}+1 \ge \frac{1}{x+1}+\frac{1}{y+1}\Leftrightarrow 1+x+y\le (1+x)(1+y)\, .
        \]
        La dernière égalité est évidemment vérifiée pour \( x\ge 0\, ,\, y\ge 0\ \),.

  \item D'après le cours, la métrique \( \mathrm{dist} \) et la métrique \( \mathrm{dist}_2 = \min (\mathrm{dist} ,1) \) sont topologiquement équivalentes. Ainsi il suffit de montrer que \( \mathrm{dist}_1 \) et \( \mathrm{dist}_2 \) sont topologiquement équivalentes.

        Puisque \( 1+\mathrm{dist} \ge 1 \), on a que \( \mathrm{dist}_1\le \mathrm{dist} \). Aussi \( \mathrm{dist}_1\le 1 \), d'où \( \mathrm{dist}_1\le \mathrm{dist}_2 \).

        La fonction \( f \) étant croissante, pour tout \( x,y \) on a que \( \mathrm{dist}_1(x,y) = f(\mathrm{dist} (x,y)) \ge f(\mathrm{dist}_2(x,y)) \). D'autre part, \( \mathrm{dist}_2(x,y)\le 1 \) implique \( f(\mathrm{dist}_2(x,y)) = \frac{\mathrm{dist}_2(x,y)}{1+\mathrm{dist}_2(x,y)}\ge
        \frac{\mathrm{dist}_2(x,y)}{2} \).

        On a obtenu que pour tout \( x,y \),
        \[
          \frac{\mathrm{dist}_2(x,y)}{2}\le \mathrm{dist}_1(x,y)\le \mathrm{dist}_2(x,y)\, .
        \]
        Ainsi, les métriques \( \mathrm{dist}_1 \) et \( \mathrm{dist}_2 \) sont équivalentes.
\end{enumerate}
\fincorrection

\finexercice
