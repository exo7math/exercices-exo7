\exercice{45, cousquer, 2003/10/01}

\enonce
Trouver les racines quatrièmes de \( 81 \) et de \( -81 \).
\finenonce

\noindication

\correction
\begin{enumerate}
  \item \( 3 \), \( 3i \), \( -3 \) et \( -3i \).

  \item \( \frac{3\sqrt2}{2}(1+i) \), \( \frac{3\sqrt2}{2}(-1+i) \), \( \frac{3\sqrt2}{2}(-1-i) \) et \( \frac{3\sqrt2}{2}(1-i) \).
\end{enumerate}
\fincorrection

\finexercice
