\exercice{1736, gourio, 2001/09/01}

\enonce
Montrer que :
\[
  \forall n\in \Nn^{*},\forall (x_1,...,x_n)\in \left( \Rr^{+*}\right)^n, 1+\left( \prod\limits_{k = 1}^nx_k\right)^{\frac{1}{n}}\le
  \left( 1+\prod\limits_{k = 1}^nx_k\right)^{\frac{1}{n}}.
\]
\finenonce

\noindication

\nocorrection

\finexercice
