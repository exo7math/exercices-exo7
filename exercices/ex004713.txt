\exercice{4713, quercia, 2010/03/16}

\enonce[Système d'ordre 1]
Soient \( 0 < a < b \) et \( (x_n),(y_n) \) les suites définies par : \( \begin{cases}x_0 = a \\ y_0 = b
\end{cases} \) \quad et \quad \( \begin{cases}x_{n+1} = \frac{x_n+y_n}2 \\
  y_{n+1} = \sqrt{ x_{n+1}y_n }.
\end{cases} \)

\begin{enumerate}
  \item Montrer que ces suites convergent vers la même limite.

  \item On pose \( a = b\cos\varphi \). Exprimer cette limite en fonction de \( b \) et \( \varphi \).

\end{enumerate}
\finenonce

\noindication

\correction
\begin{enumerate}
  \item

  \item \( \ell = b\frac{\sin\varphi}{\varphi} \).
\end{enumerate}
\fincorrection

\finexercice
