\exercice{6705, queffelec, 2011/10/16}

\enonce
Existe-t-il des fonctions \( f \) holomorphes sur \( D(0,1) \) telles que pour tout entier \( n>0 \) ?

\begin{enumerate}
  \item \( f\left( \frac{1}{2n}\right) = f\left( \frac{1}{2n+1}\right) = \frac{1}{n} \) ?

  \item \( f\left( \frac{1}{n}\right) = f\left( -\frac{1}{n}\right) = \frac{1}{n^2} \) ?

  \item \( f\left( \frac{1}{n}\right) = f\left( -\frac{1}{n}\right) = -\frac{1}{n^3} \) ?
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
