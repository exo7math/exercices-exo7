\exercice{6468, drutu, 2011/10/16}

\enonce
\begin{enumerate}
  \item Montrer que tout sous-groupe d'un groupe cyclique est cyclique.

  \item Soit \( d\in \Nn^* \) un diviseur de \( n\in \Nn^* \). Montrer qu'il existe un unique sous-groupe d'ordre \( d \) dans \( \Zz /n\Zz \).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
