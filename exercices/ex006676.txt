\exercice{6676, queffelec, 2011/10/16}

\enonce
Calculer les intégrales suivantes.
\begin{enumerate}
  \item \( \displaystyle\int_\gamma \frac{\cos z}{z}\ dz \) où \( \gamma (t) = e^{it} \) (\( t\in [0,2\pi] \))

  \item \( \displaystyle\int_\gamma \frac{\sin z}{z}\ dz \) où \( \gamma (t) = e^{it} \) (\( t\in [0,2\pi] \))

  \item \( \displaystyle\int_\gamma \frac{\cos {z^2}}{z}\ dz \) où \( \gamma (t) = e^{it} \) (\( t\in [0,2\pi] \))

  \item \( \displaystyle\int_\gamma \frac{e^{\pi z}}{z^3+z}\ dz \) où \( \gamma (t) = 2e^{it} \) (\( t\in [0,2\pi] \))

  \item \( \displaystyle\int_\gamma \frac{dz}{(z-a)^n} \) (\( n\in\Zz \)), où \( \gamma \) est un chemin fermé ne passant pas par \( a \)

  \item \( \displaystyle\int_{\gamma_r}\frac{3z^2-12z+11}{z^3-6z^2+11z-6}\ dz \), où \( \gamma_r(t) = re^{it} \) (\( t\in [0,2\pi] \))

  \item \( \displaystyle\int_{\gamma }\left(\frac{z}{z-1}\right)^n\ dz \), \( n\in\Nn^* \), où \( \gamma (t) = 1+e^{it} \) (\( t\in [0,2\pi] \))
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
