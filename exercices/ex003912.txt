\exercice{3912, quercia, 2010/03/11}

\enonce[Calcul de somme]
Calculer \( \cotan x - 2\cotan2x \). Simplifier \( \sum_{k = 0}^n \frac 1{2^k}\tan\frac\alpha{2^k} \).
\finenonce

\noindication

\correction
\( \cotan x - 2\cotan2x = \tan x \), \( \sum = \frac 1{2^n}\cotan\frac\alpha{2^n} - 2\cotan2\alpha \).
\fincorrection

\finexercice
