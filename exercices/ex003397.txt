\exercice{3397, quercia, 2010/03/10}

\enonce[Calcul de \( A^k \)]
Calculer \( A^k \) pour \( k \in \Nn \) :
\begin{enumerate}
  \item \( A = \begin{pmatrix}1   &        & (2) \\
                   & \ddots       \\
               (2) &        & 1
        \end{pmatrix} \).
	(Des \( 2 \) partout, sauf des \( 1 \) sur la diagonale.)

  \item \( A = \begin{pmatrix} 1 & 2 & 3 & 4 \\
                0 & 1 & 2 & 3 \\
                0 & 0 & 1 & 2 \\
                0 & 0 & 0 & 1
        \end{pmatrix} \).

  \item \( A = \begin{pmatrix} x^2 & xy  & xz  \\
                xy  & y^2 & yz  \\
                xz  & yz  & z^2
        \end{pmatrix} \).

\end{enumerate}
\finenonce

\noindication

\correction
\begin{enumerate}
  \item \( \begin{pmatrix} a &        & b \\
                  & \ddots     \\
                b &        & a
        \end{pmatrix} \) avec \( \begin{cases} a = \frac1n\bigl( (2n-1)^k + (n-1)(-1)^k \bigr) \\
          b = \frac1n\bigl( (2n-1)^k - (-1)^k \bigr).
        \end{cases} \)

  \item \( \begin{pmatrix} 1 & 2k & 2k^2+k & \frac 13(4k^3 + 6k^2 + 2k) \\
                0 & 1  & 2k     & 2k^2+k                     \\
                0 & 0  & 1      & 2k                         \\
                0 & 0  & 0      & 1
        \end{pmatrix} \).

  \item \( \left[{ \begin{pmatrix} x \\ y \\ z
              \end{pmatrix} \begin{pmatrix} x & y & z
              \end{pmatrix} }\right]^k = (x^2+y^2+z^2)^{k-1}A \).
\end{enumerate}
\fincorrection

\finexercice
