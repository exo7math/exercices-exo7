\exercice{1116, legall, 1998/09/01}

\enonce
\begin{enumerate}
  \item Soient \( A \in M_p(\Rr) \) et \( B \in M_q(\Rr) \). Calculer (en fonction de \( \hbox{det}(A) \) et \( \hbox{det}(B) \)) le déterminant de la matrice \( M = \begin{pmatrix}A & 0 \\ 0 & B
        \end{pmatrix} \in M_{p+q}(\Rr) \). (On pourra pour cela décomposer \( M \) comme produit de deux matrices de déterminant évident et utiliser la multiplicativité du déterminant.)

  \item Soient \( A \in M_p(\Rr) \), \( B \in M_q(\Rr) \) et \( C
        \in M_{p,q}(\Rr) \). Calculer le déterminant de la matrice \( M = \begin{pmatrix}A & C \\ 0 & B
        \end{pmatrix} \in M_{p+q}(\Rr) \). (On pourra généraliser la méthode de 1.)
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
