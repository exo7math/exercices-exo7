\exercice{3461, quercia, 2010/03/10}

\enonce[Calcul de rang]
Chercher les rangs des matrices suivantes :

\begin{enumerate}
  \item \( \begin{pmatrix} 1 & 2  & -4 & -2 & -1 \\
                0 & -2 & 4  & 2  & 0  \\
                1 & 1  & -2 & -1 & 1
        \end{pmatrix} \).

  \item \( \begin{pmatrix} 0  & -1 & 2  & -2 \\
                -7 & -7 & 2  & -8 \\
                0  & 4  & -6 & 6  \\
                2  & -2 & 0  & -2
        \end{pmatrix} \).

  \item \( \begin{pmatrix} 1  & 7 & 2 & 5 \\
                -2 & 1 & 1 & 5 \\
                -1 & 2 & 1 & 4 \\
                1  & 4 & 1 & 2
        \end{pmatrix} \).

  \item \( \begin{pmatrix} 1  & 4 & -1 & 2  & 4 \\
                2  & 0 & -3 & -1 & 7 \\
                -2 & 3 & 2  & 1  & 4
        \end{pmatrix} \).
\end{enumerate}
\finenonce

\noindication

\correction
\begin{enumerate}
  \item 3.

  \item 4.

  \item 2.

  \item 3.
\end{enumerate}
\fincorrection

\finexercice
