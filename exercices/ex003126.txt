\exercice{3126, quercia, 2010/03/08}

\enonce[Congruences simultanées]
\begin{enumerate}
  \item Soient \( a,b,a',b' \in \Zz \) avec \( b \wedge b' = 1 \). Montrer que le système : \( \begin{cases}x \equiv a (\mathrm{mod}\, b) \\
          x \equiv a' (\mathrm{mod}\, {b'})\end{cases} \) possède des solutions et qu'elles sont congrues entre elles modulo \( bb' \).

  \item Généraliser.
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
