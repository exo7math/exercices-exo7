\exercice{4974, quercia, 2010/03/17}

\enonce[Ensi P 90]
Déterminer l'équation de la sphère contenant les cercles d'équations \( \begin{cases} x^2+y^2 = 9 \\ z = 0
\end{cases} \) et \( \begin{cases} x^2+y^2 = 25 \\ z = 2.
\end{cases} \)
\finenonce

\noindication

\correction
\( x^2+y^2+z^2-10z = 9 \).
\fincorrection

\finexercice
