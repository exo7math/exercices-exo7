\exercice{2637, debievre, 2009/05/19}

\enonce
Soit \( g\colon\Rr_{>0}\times \Rr_{>0}\to \Rr \) une fonction de classe \( C^1 \) telle que \( g(1,1) = 3 \) et dont la différentielle vaut
\begin{equation}\label{eq}
  \mathrm{d} g= (2xy+y^2)\mathrm{d} x + (x^2+2xy) \mathrm{d} y.
\end{equation}
Soit
\[
  h\colon \Rr_{>0}\times\Rr_{>0} \longrightarrow \Rr_{>0}\times\Rr_{>0}
\]
l'application de classe \( C^1 \) définie par
\[
  h(x,y) = (u(x,y), v(x,y)) = (x^2y, xy^2)\in \Rr_{>0}\times \Rr_{>0}.
\]
\begin{enumerate}
  \item Calculer \( \mathrm{d} u+\mathrm{d} v \).

  \item Déterminer \( g \) à partir du calcul précédent et (\ref{eq}), et sans autre calcul.

  \item Montrer que \( h \) est une bijection. (On pourra calculer explicitement \( h^{-1} \).)

  \item Déterminer explicitement \( \mathrm{d} (g\circ h^{-1}) \).

  \item Calculer les matrices jacobiennes \( J_h(x,y) \) et \( J_{h^{-1}}(u,v) \) et vérifier par un calcul direct que
        \[
          J_h(x,y)J_{h^{-1}}(h(x,y)) = I_2,
        \]
        où \( I_2 \) est la matrice identité d'ordre 2.
\end{enumerate}
\finenonce

\indication
On va déterminer une primitive d'une forme différentielle de degré 1 par un changement de variables tel que, dans les nouvelles variables, la primitive soit presque évidente.
\finindication

\correction
\begin{enumerate}
  \item Un calcul immédiat donne \( \mathrm d u+\mathrm d v = \mathrm d g \).

  \item Par conséquent, \( g = u+v+c \) où la constante \( c \) est déterminée par la condition
        \[
          3 = g(1,1) = u(1,1)+v(1,1)+c = 1+1+c
        \]
        d'où \( c = 1 \).

  \item Un calcul direct montre que l'application réciproque
        \[
          k \colon \Rr_{>0}\times\Rr_{>0} \longrightarrow \Rr_{>0}\times\Rr_{>0}
        \]
        de \( h \) est donnée par la formule
        \[
          k(u,v) = (x(u,v),y(u,v)) =
          \left(\left(\frac{u^2}{v}\right)^{1/3},\left(\frac{v^2}{u}\right)^{1/3}\right).
        \]

  \item \( \mathrm d (g \circ k) = \mathrm d (u \circ k)+ \mathrm d (v \circ k) = \mathrm d u +\mathrm d v \) car \( u(k(u,v)) = u \) et \( v(k(u,v)) = v \).

  \item Un calcul immédiat donne
        \[
          J_h = \left [\begin{matrix} 2xy & x^2\\ y^2 &2xy \end{matrix}\right ], \quad J_k = \left [\begin{matrix}
              \frac 23 (uv)^{-1/3} & -\frac{u^{2/3}}{3v^{4/3}}
              \\ -\frac{v^{2/3}}{3u^{4/3}} &\frac 23 (uv)^{-1/3}
            \end{matrix}
            \right ]
        \]
        d'où \( J_h(x,y)J_k(h(x,y)) = I_2 \).
\end{enumerate}
\fincorrection

\finexercice
