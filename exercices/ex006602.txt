\exercice{6602, hueb, 2011/10/16}

\enonce
Calculer les transformées de Laplace des fonctions suivantes, où \( a,b,A,T \) sont des nombres réels positifs.
\begin{enumerate}
  \item \( f(t) = \begin{cases}
          \frac Aa t, & 0\le t \le a \\
          A,          & t \ge a\end{cases} \)

  \item \( f(t) = \begin{cases} nA, & \text{pour}\ (n-1) T \le t < nT,\
              \text{où}\ n\in \Nn, n\ge 1, \ \text{c.-à-d.}
              \\
              A,  & 0\le t <T                         \\
              2A, & T\le t <2T, \ \text{etc.}\end{cases} \)

  \item \( f(t) = \begin{cases}
          \frac{4A}T t,       & 0\le t < \frac T4           \\
          -\frac{4A}T t + 2A, & \frac T4 \le t < \frac 34 T \\
          \frac{4A}T t - 4A,  & \frac 34 T \le t < T        \\
          0,                  & t \ge T\end{cases} \)

  \item \( f(t) = \begin{cases}
          0,  & 0\le t <a       \\
          A,  & a\le t < a+b    \\
          -A, & a+b\le t < a+2b \\
          0,  & t \ge a+2b\end{cases} \)
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
