\exercice{1431, legall, 1998/09/01}

\enonce
\begin{enumerate}
  \item Soit \( G \) un groupe, \( H \) un sous-groupe de \( G \). Montrer que les propriétés suivantes sont équivalentes :
        \vskip1mm \hskip2mm {\em i)} \( \forall g \in G : gHg^{-1}\subset H \).
        \vskip1mm \hskip2mm {\em ii)} \( \forall g \in G : gHg^{-1} = H \).
        \vskip1mm \hskip2mm {\em iii)} \( \forall g \in G : gH = Hg \).

  \item En déduire que tout sous-groupe d'indice \( 2 \) est distingué.
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
