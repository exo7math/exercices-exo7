\exercice{3711, quercia, 2010/03/11}

\enonce[Expressions analytiques]
Reconnaître les endomorphismes de \( \Rr^3 \) définis par les expressions analytiques dans la base canonique :

\begin{enumerate}
  \item \( \begin{cases} 3x' = 2x+2y+z \\ 3y' = -2x+y+2z \\ 3z' = x-2y+2z
        \end{cases} \)

  \item \( \begin{cases} 9x' = 8x+y-4z \\ 9y' = -4x+4y-7z \\ 9z' = x+8y+4z
        \end{cases} \)

  \item \( \begin{cases} 3x' = -2x+2y-z \\ 3y' = 2x+y-2z \\ 3z' = -x-2y-2z
        \end{cases} \)

  \item \( \begin{cases} 4x' = -2x - y\sqrt6 + z\sqrt6 \\
          4y' = x\sqrt6 + y + 3z        \\
          4z' = -x\sqrt6 + 3y + z
        \end{cases} \)

  \item \( \begin{cases}x' = \frac x{\sqrt3} + \frac y{\sqrt2} - \frac z{\sqrt6} \\
          y' = \frac x{\sqrt3} + \frac{2z}{\sqrt6}                 \\
          z' = \frac x{\sqrt3} - \frac y{\sqrt2} - \frac z{\sqrt6}
        \end{cases} \)

  \item \( \begin{cases} 3x' = x+2y+2z \\ 3y' = 2x+y-2z \\ 3z' = 2x-2y+z
        \end{cases} \)

  \item \( \begin{cases} 7x' = -2x+6y-3z \\ 7y' = 6x+3y+2z \\ 7z' = -3x+2y+6z
        \end{cases} \)

  \item \( \begin{cases} 3x' = 2x-2y+z \\ 3y' = -2x-y+2z \\ 3z' = x+2y+2z
        \end{cases} \)

  \item \( \begin{cases} 3x' = 2x+y+2z \\ 3y' = 2x-2y-z \\ 3z' = -x-2y+2z
        \end{cases} \)

  \item \( \begin{cases} 4x' = -x+3y-z\sqrt6 \\
          4y' = 3x-y-z\sqrt6  \\
          4z' = x\sqrt6+y\sqrt6+2z
        \end{cases} \)

  \item \( \begin{cases} 15x' = 5x-10z \\ 15y' = -8x+5y+6z \\ 15z' = 6x-10y+8z
        \end{cases} \)
        \label{rotproj}
\end{enumerate}
\finenonce

\correction
\begin{enumerate}
  \item rotation autour de \( (1,0,1) \) d'angle \( -\arccos(1/3) \).

  \item rotation autour de \( (-3,1,1) \) d'angle \( -\arccos(7/18) \).

  \item demi-tour autour de \( (-1,-2,1) \).

  \item rotation autour de \( (0,1,1) \) d'angle \( 2\pi/3 \).

  \item rotation autour de \( (-2-\sqrt3,1+\sqrt2,\sqrt2 - \sqrt3) \) d'angle \( \arccos( \frac{\sqrt6 - \sqrt2 + 1}{2\sqrt6} ) \).

  \item symétrie par rapport à \( x = y+z \).

  \item symétrie par rapport à \( 3x = 2y-z \).

  \item symétrie par rapport à \( x+2y-z = 0 \).

  \item symétrie-rotation autour de \( (1,-3,1) \) d'angle \( -\arccos(5/6) \).

  \item symétrie-rotation autour de \( (1,-1,0) \) d'angle \( \pi/3 \).

  \item projection sur \( 2x+2y+z = 0 \) puis rotation d'angle \( \arccos(3/5) \).
\end{enumerate}
\fincorrection

\finexercice
