\exercice{1826, drutu, 2003/10/01}

\enonce
004 | Trouvez les points critiques des fonctions suivantes et déterminez si ce sont des minima locaux, des maxima locaux ou des points selles.
\begin{enumerate}
  \item \( f(x,y) = x^3+6x^2+3y^2-12xy+9x \) ;

  \item \( f(x,y) = \sin x+y^2-2y+1 \) ;

  \item \( f(x,y,z) = \cos 2x\cdot \sin y+z^2 \) ;

  \item \( f(x,y,z) = (x+y+z)^2 \).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
