\exercice{3041, quercia, 2010/03/08}

\enonce[Parties adjacentes]
Soient \( A,B \subset \Rr \) vérifiant :
\[
  \begin{cases}\forall\ a \in A,\ \forall\ b \in B,\ a \le b \\
    \forall\ \epsilon > 0,\ \exists\ a \in A,\ \exists\ b \in B \text{ tq } b-a \le \epsilon\end{cases}
\]
(on dit que \( A \) et \( B \) sont {\it adjacentes\/}). Montrer que \( \sup(A) = \inf(B) \).
\finenonce

\noindication

\nocorrection

\finexercice
