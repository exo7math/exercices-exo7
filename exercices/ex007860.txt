\exercice{7860, mourougane, 2021/08/11}

\enonce[Exemple de produits semi-directs]
\begin{enumerate}
  \item Montrer que, après avoir fixé un générateur de \( (\Zz/11\Zz)^\star \), la donnée d'un morphisme de groupes de \( \Zz/5\Zz \) dans \( Aut(\Zz/11\Zz) \) revient à la donnée d'un morphisme de groupes de \( \Zz/5\Zz \) dans \( \Zz/10\Zz \).

  \item En déduire une structure de produit semi-direct sur \( \Zz/11\Zz\rtimes\Zz/5\Zz \).

  \item Montrer que toutes les structures de produit semi-direct donnent des groupes isomorphes.
        \textit{On pourra montrer que si \( \phi, \psi\in Hom(\Zz/5\Zz,Aut(\Zz/11\Zz)) \) alors il existe \( \gamma\in Aut(\Zz/11\Zz) \) tel que \( \psi (h) = \gamma\circ\phi (h)\circ \gamma^{-1} \).}

  \item Montrer que tous les morphismes de \( \Zz/p\Zz\to Aut (\Zz/q\Zz) \) sont de la forme \( t\mapsto \{x\mapsto k^t x\} \) où \( k \) est un élément de \( (\Zz/q\Zz)^\star \) d'ordre \( p \).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
