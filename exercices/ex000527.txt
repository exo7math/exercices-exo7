\exercice{527, gourio, 2001/09/01}

\enonce
\begin{enumerate}
  \item Donner un exemple de suite bornée divergente, puis de suite divergente telle que
        \[
          \forall k\in \Nn,\lim_{n\rightarrow \infty }x_{n+k}-x_n = 0.
        \]

  \item Donner un exemple de suite divergente qui a une seule valeur d'adhérence (i.e. telle qu'il existe une seule extraction \( \phi \) telle que \( x_{\phi(n)\text{ }} \) converge).

  \item Donner un exemple de suite \( (x_n)_{n\in \Nn} \) divergente telle que \( \forall k\ge 2,(x_{nk})_{n\in \Nn} \) converge.
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
