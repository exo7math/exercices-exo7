\exercice{3497, quercia, 2010/03/10}

\enonce[Diagonalisation en dimension \( 3 \)]
Diagonaliser les matrices suivantes :
\begin{enumerate}
  \item \( A = \begin{pmatrix}-1 & 2 & -3 \\ 2 & 2 &-6 \\ -2 & 2 &-6
        \end{pmatrix} \)

  \item \( A = \begin{pmatrix} 0 & \phantom-1 & 2 \\ 1 & 1 & 1 \\ 1 & 0 &-1
        \end{pmatrix} \)

  \item \( A = \begin{pmatrix} 1 & 2 & 3 \\ 2 & 3 & 1 \\ 3 & 1 & 2
        \end{pmatrix} \)

  \item \( A = \begin{pmatrix} 1 & 2 & 3 \\ 2 &-1 &-4 \\ 3 & 1 & 2
        \end{pmatrix} \)

  \item \( A = \begin{pmatrix} 1 & 2 & 3 \\ 2 & 1 & 3 \\ 4 & 2 & 0
        \end{pmatrix} \)

  \item \( A = \begin{pmatrix} 1 & 1 & 0 \\ -1 & 1 & 2 \\ 0 & 0 & 2
        \end{pmatrix} \)

  \item \( A = \begin{pmatrix} 2 & -1 & -1 \\ -1 & 2 &-1 \\ -1 &-1 & 2
        \end{pmatrix} \)

  \item \( A = \begin{pmatrix} 1 & -1 & 2 \\ 3 &-3 & 6 \\ 2 &-2 & 4
        \end{pmatrix} \)

  \item \( A = \begin{pmatrix} 7 & -12 & -2 \\ 3 &-4 & 0 \\ -2 & 0 &-2
        \end{pmatrix} \)
  \item\( A = \begin{pmatrix}-2 & 8 & 6 \\ -4 & 10 & 6 \\ 4 &-8 &-4
        \end{pmatrix} \)
\end{enumerate}
\finenonce

\noindication
