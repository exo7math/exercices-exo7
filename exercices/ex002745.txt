\exercice{2745, tumpach, 2009/10/25}

\enonce
Soit la matrice
\[
  B = \left( \begin{array}{cccc}1 & 2 & 3 & 1\\-1 & 2 & -1 &-3\\ -3 & 5 & 2 &-3\end{array}\right).
\]
\begin{enumerate}
  \item Déterminer une base du noyau de \( B \).

  \item Déterminer une base de l'image de \( B \).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
