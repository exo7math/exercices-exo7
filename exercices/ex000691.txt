\exercice{691, bodin, 1998/09/01}

\enonce
\begin{enumerate}
  \item Soit la fonction \( f:\lbrack -1,+\infty\lbrack\to\Rr \), définie par \( f(x) = \frac{1}{\sqrt{x^2+2x+2}} \). Montrer que \( f \) admet une réciproque que l'on explicitera.

  \item Trouver un intervalle de \( \Rr \) sur lequel la fonction \( g(x) = \tan(x^3) \) admette une fonction réciproque (on précisera alors le domaine de définition de cette réciproque et son image).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
