\exercice{3638, quercia, 2010/03/10}

\enonce[Système unisolvent]
Soient \( f_1,\dots,f_n : \Rr \to \Rr \), \( n \) fonctions linéairement indépendantes. On pose \( E = \text{vect}(f_1,\dots,f_n) \) et pour \( x \in \Rr \), \( \delta_x : E \to \Rr, f \mapsto f(x) \).

\begin{enumerate}
  \item Montrer que la famille \( (\delta_x)_{x\in\Rr} \) engendre \( E^* \).

  \item En déduire qu'il existe \( x_1,\dots,x_n \in \Rr \) tels que \( \det M \ne 0 \) où \( M \) est la matrice de terme général \( f_i(x_j) \).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
