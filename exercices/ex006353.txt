\enonce
Trouver des conditions nécessaires et suffisantes pour que la matrice \( \begin{pmatrix} a & b \\
                c & d
\end{pmatrix} \) soit

  \item diagonalisable,
        \quad semi-simple,
        \quad nilpotente.
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
