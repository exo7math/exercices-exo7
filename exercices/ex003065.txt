\exercice{3065, quercia, 2010/03/08}

\enonce[Partie entière]
\begin{enumerate}
  \item Soient \( a\in\Zz \) et \( b\in\Nn^* \). Montrer que :
        \( \left\lfloor \frac{a}{b} \right\rfloor + \left\lfloor \frac{a+1}{b} \right\rfloor + \dots + \left\lfloor \frac{a+b-1}{b} \right\rfloor = a \).

  \item Soient \( a\in\Rr \) et \( b\in\Nn^* \). Montrer que :
        \( \left\lfloor \frac{a}{b} \right\rfloor + \left\lfloor \frac{a+1}{b} \right\rfloor + \dots + \left\lfloor \frac{a+b-1}{b} \right\rfloor = \lfloor a \rfloor \).
\end{enumerate}
\finenonce

\noindication

\correction
\begin{enumerate}
  \item \( a = bq+r \Rightarrow \sum = \underbrace{q + q + \dots + q}_{b-r}
        + \underbrace{(q+1) + \dots + (q+1)}_r = bq+r = a \).

  \item
\end{enumerate}
\fincorrection

\finexercice
