\exercice{174, cousquer, 2003/10/01}

\enonce
\begin{enumerate}
  \item Soit \( n\in\Nn \); montrer que pour tout entier \( k\ge 1 \) on a
        \[
          n^k+kn^{k-1}\le (n+1)^k.
        \]

  \item Soit \( b \) un réel positif ou nul. Montrer par récurrence que pour tout \( n\ge 1 \) on a
        \[
          (1+b)^n\le 1+\frac{nb}{1!}+\frac{(nb)^2}{2!}+\dots+\frac{(nb)^n}{n!}.
        \]
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
