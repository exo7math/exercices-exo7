\exercice{5964, tumpach, 2010/11/11}

\enonce
Soit \( \Omega \) un sous-ensemble de \( \Rr^n \) dont la mesure de Lebesgue est \emph{finie} : \( \mu(\Omega)< +\infty \). Pour tout \( 1\le p <+\infty \), on note \( L^p(\Omega) \) l'espace des fonctions \( f :\Omega \rightarrow \Cc \) telles que \( \|f\|_p :=
\left(\int_{\Omega} |f|^p(x)\,dx\right)^{\frac{1}{p}}<+\infty \) modulo l'équivalence \( f\sim g \Leftrightarrow f-g = 0  \mu-p.p \). L'espace des fonctions essentiellement bornées sera noté \( L^{\infty}(\Omega) \).
\begin{enumerate}
  \item Montrer que si \( q\le p \), alors \( L^p(\Omega) \subset L^q(\Omega) \). En particulier, pour \( 1<q<2<p \), on a :
        \[
          L^{\infty}(\Omega) \subset L^p(\Omega) \subset L^2(\Omega)
          \subset L^q(\Omega) \subset L^1(\Omega).
        \]

  \item Soit \( \mathcal{B}^n(0, 1) \) la boule unité centrée en \( 0 \) de \( \Rr^n \). En considérant les fonctions
        \[
          f_{\alpha}(x) = |x|^{-\alpha}
        \]
        montrer que pour \( q<p \), l'inclusion \( L^p(\mathcal{B}^n(0,1))
        \subset L^q(\mathcal{B}^n(0,1)) \) est stricte.
\end{enumerate}
\finenonce

\noindication

\correction
Soit \( \Omega \) un sous-ensemble de \( \Rr^n \) dont la mesure de Lebesgue est \emph{finie} : \( \mu(\Omega)< +\infty \). Pour tout \( 1\le p <+\infty \), notons \( L^p(\Omega) \) l'espace des fonctions \( f :\Omega \rightarrow \Cc \) telles que \( \|f\|_p :=
\left(\int_{\Omega} |f|^p(x)\,dx\right)^{\frac{1}{p}}<+\infty \) modulo l'équivalence \( f\sim g \Leftrightarrow f-g = 0  \mu-p.p \). L'espace des fonctions essentiellement bornées sera noté \( L^{\infty}(\Omega) \).
\begin{enumerate}
  \item Si \( f\in L^{\infty}(\Omega) \), alors
        \[
          \| f \|_p^p = \int_{\Omega} |f|^p(x)\,dx \le \|f\|_{\infty}^p
          \mu(\Omega)<+\infty,
        \]
        ainsi \( L^{\infty}(\Omega) \subset L^p(\Omega) \) pour tout \( p \) et \( \|f\|_p \le
        \|f\|_{\infty}\left(\mu(\Omega)\right)^{\frac{1}{p}} \). Montrons que si \( q\le p \), alors \( L^p(\Omega) \subset L^q(\Omega) \). Soit \( f\in L^p(\Omega) \), on a par exemple :
        \begin{eqnarray*}
          \| f \|_q^q = \int_{\Omega}|f|^q(x)\,dx &=& \int_{\{|f|\ge1\}}|f|^q(x)\,dx +
          \int_{\{|f|<1\}} |f|^q(x)\,dx \\
          & \le & \int_{\{|f|\ge1\}}|f|^p(x)\,dx + \int_{\{|f|<1\}}
          1\,dx\\
          & \le & \|f\|_p^p + \mu(\Omega)<+\infty.
        \end{eqnarray*}
        Ou encore, en utilisant l'inégalité de Hölder pour les réels conjugués \( r = \frac{p}{q} > 1 \) et \( r' = \frac{p}{p-q} \):
        \begin{eqnarray*}
          \| f \|_q^q = \int_{\Omega}|f|^q(x)\,dx &=& \left( \int_{\Omega}|f|^{q\cdot
              \frac{p}{q}}(x)\,dx\right)^{\frac{q}{p}}
          \left(\int_{\Omega} 1^{\frac{p}{p-q}}(x)\,dx\right)^{\frac{p-q}{p}} \\
          & = & \|f\|_p^q \mu(\Omega)^{\frac{p-q}{p}},
        \end{eqnarray*}
        ce qui implique :
        \[
          \|f\|_q \le \|f\|_p \mu(\Omega)^{\frac{p-q}{qp}}.
        \]
        En conclusion, pour \( 1<q<2<p \):
        \[
          L^{\infty}(\Omega) \subset L^p(\Omega) \subset L^2(\Omega)
          \subset L^q(\Omega) \subset L^1(\Omega).
        \]

  \item Montrons que pour \( q<p \), l'inclusion \( L^p(\mathcal{B}^n(0,1)) \subset L^q(\mathcal{B}^n(0,1)) \) est stricte. La fonction \( f_{\alpha} \) appartient à \( L^{\infty}(\mathcal{B}^n(0,1)) \) si et seulement \( \alpha \le 0 \), et à \( L^p(\mathcal{B}^n(0,1)) \) avec \( p<+\infty \) si et seulement si
        \[
          p\alpha - n +1 < 1 \Leftrightarrow \alpha < \frac{n}{p}
        \]
        Soit \( 1\le q<p \), alors \( f_{\frac{1}{2}\left(\frac{n}{p}+\frac{n}{q}\right)} \) appartient à \( L^q(\mathcal{B}^n(0,1))\setminus L^p(\mathcal{B}^n(0,1)) \). En particulier, \( f_{\frac{1}{2}\left(\frac{n}{p}+\frac{n}{q}\right)} \) appartient à \( L^q(\mathcal{B}^n(0,1))\setminus L^{\infty}(\mathcal{B}^n(0,1)) \).
\end{enumerate}
\fincorrection

\finexercice
