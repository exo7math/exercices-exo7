\exercice{1053, legall, 1998/09/01}

\enonce
Déterminer deux éléments \( A \) et \( B \) de \( M_2(\Rr) \) tels que : \( AB = 0 \) et \( BA\not= 0 \).
\finenonce

\noindication

\correction
\[
  A =
  \begin{pmatrix}
    0 & 1 \\
    0 & 0
  \end{pmatrix}, \qquad B =
  \begin{pmatrix}
    1 & 0 \\
    0 & 0
  \end{pmatrix}.
\]
\fincorrection

\finexercice
