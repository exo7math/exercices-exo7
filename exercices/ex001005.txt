\exercice{1005, legall, 1998/09/01}

\enonce
Soit \( E = \Rr^3 \). On définit le système
\[
  S = \left\{ {\mathbf e}_1 = (1,1,1), {\mathbf e}_2 = (1,1,2), {\mathbf e}_3 = (1,2,3)\right\}
\]
\begin{enumerate}
  \item Montrer que \( S \) est une base de \( E \).

  \item Calculer les coordonnées de \( {\mathbf v} = (5,7,12) \) dans cette base.
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
