\exercice{3190, quercia, 2010/03/08}

\enonce[ENS Ulm MP 2002]
Soit \( S\subset \Nn \) fini et \( P = \sum_{s\in S} a_sX^s\in\Cc[X] \).
\begin{enumerate}
  \item On suppose que les \( a_s \) sont réels. Montrer que \( P \) a moins de racines strictement positives distinctes que la suite \( (a_s) \) n'a de changement de signe.

  \item On suppose que \( P \) vérifie : \( \forall\ s\in S,\ P(s) = 0 \). Montrer que \( P \) est nul.
\end{enumerate}
\finenonce

\noindication

\correction
\begin{enumerate}
  \item Récurrence sur \( \mathrm{Card}\,(S) \) en mettant le terme de plus bas degré en facteur et en dérivant le quotient.

  \item Appliquer la question précédente aux suites \( (\Re(a_s)) \) et \( (\Im(a_s)) \).
\end{enumerate}
\fincorrection

\finexercice
