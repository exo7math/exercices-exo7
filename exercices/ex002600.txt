\exercice{2600, delaunay, 2009/05/19}

\enonce
\begin{enumerate}
  \item Donner un exemple de matrice dans \( M_2(\Rr) \), diagonalisable sur \( \Cc \) mais non diagonalisable sur \( \Rr \) (justifier).

  \item Donner un exemple de matrice dans \( M_2(\Rr) \) non diagonalisable, ni sur \( \Cc \), ni sur \( \Rr \) (justifier).
\end{enumerate}
\finenonce

\noindication

\correction
\begin{enumerate}
  \item {\it Donnons un exemple de matrice dans \( M_2(\Rr) \), diagonalisable sur \( \Cc \) mais non diagonalisable sur \( \Rr \)}. (2 points)

        Soit \( A \) la matrice \( A = \begin{pmatrix}0&-1 \\ 1&0\end{pmatrix} \). Son polynôme caractéristique est égal à
        \[
          P_A(X) = \begin{vmatrix}-X&-1 \\ 1&-X\end{vmatrix} = X^2+1.
        \]
        Le polynôme caractéristique de \( A \) admet deux racines complexes conjuguées distinctes \( i \) et \( -i \) elle est donc diagonalisable sur \( \Cc \) mais elle ne l'est pas sur \( \Rr \).

  \item {\it Donnons un exemple de matrice dans \( M_2(\Rr) \) non diagonalisable, ni sur \( \Cc \), ni sur \( \Rr \).} (2 points)

        Soit \( A \) la matrice \( A = \begin{pmatrix}1&1 \\ 0&1\end{pmatrix} \). Son polynôme caractéristique est égal à
        \[
          P_A(X) = \begin{vmatrix}1-X&1 \\ 0&1-X\end{vmatrix} = (1-X)^2.
        \]
        Le polynôme caractéristique de \( A \) admet une racine double \( 1 \), la matrice \( A \) admet l'unique valeur propre \( 1 \), or, elle n'est pas égale à l'identité, par conséquent, elle n'est diagonalisable, ni sur \( \Cc \), ni sur \( \Rr \).

\end{enumerate}
\fincorrection

\finexercice
