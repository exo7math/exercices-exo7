\exercice{3398, quercia, 2010/03/10}

\enonce[Inversion de matrices]
Inverser les matrices suivantes :

\begin{enumerate}
  \item \( \begin{pmatrix}0   &        & (1) \\
                   & \ddots       \\
               (1) &        & 0
        \end{pmatrix} \).

  \item \( \begin{pmatrix}a   &        & (b) \\
                   & \ddots       \\
               (b) &        & a
        \end{pmatrix} \).

  \item \( \begin{pmatrix} 1        & 1      &        & (0) \\
                         & \ddots & \ddots       \\
                         &        & \ddots & 1   \\
                \span(0) &        & 1
        \end{pmatrix} \).

  \item \( \begin{pmatrix} 1        & \bar\alpha & \bar\alpha^2 \\
                \alpha   & 1          & \bar\alpha   \\
                \alpha^2 & \alpha     & 1
        \end{pmatrix} \), \( \alpha \in \Cc \).

  \item \( \begin{pmatrix} (0) &  & a_n \\ &\cdots \\ a_1 &&(0)
        \end{pmatrix} \).

  \item \( \begin{pmatrix} 1+\frac 1{\lambda_1} &        & (1)                  \\
                                     & \ddots                        \\
                (1)                  &        & 1+\frac 1{\lambda_n}
        \end{pmatrix} \).

\end{enumerate}
\finenonce

\noindication

\correction
\begin{enumerate}
  \item \( \frac{A + (2-n)I}{n-1} \).

  \item \( \frac 1{(a-b)(a+(n-1)b)}
        \begin{pmatrix} a+(n-2)b &        & (-b)     \\
                         & \ddots            \\
                (-b)     &        & a+(n-2)b
        \end{pmatrix} \).

  \item \( \begin{pmatrix} 1   & -1     & 1      & \dots  & \pm1   \\
                    & \ddots & \ddots & \ddots & \vdots \\
                    &        & \ddots & \ddots & 1      \\
                    &        &        & \ddots & -1     \\
                (0) &        &        &        & 1
        \end{pmatrix} \).

  \item \( \frac 1{1-\alpha\bar\alpha}
        \begin{pmatrix} 1       & -\bar\alpha        & 0           \\
                -\alpha & 1+\alpha\bar\alpha & -\bar\alpha \\
                0       & -\alpha            & 1
        \end{pmatrix} \).

  \item \( \begin{pmatrix} (0) &  & 1/a_n \\ &\cdots \\ 1/a_1 &&(0)
        \end{pmatrix} \).

  \item \( \text{diag}(\lambda_i) - \frac 1{1 + \lambda_1 + \dots + \lambda_n}(\lambda_i\lambda_j) \).
\end{enumerate}
\fincorrection

\finexercice
