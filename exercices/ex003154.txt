\exercice{3154, quercia, 2010/03/08}

\enonce[Théorème chinois]
Soient \( n,p \in \Nn^* \) tels que \( n \wedge p = 1 \). Pour \( x\in\Zz \) on note \( \overline{x}^{\,n} \), \( \overline{x}^{\,p} \) et \( \overline{x}^{\,np} \) les classes d'équivalence de \( x \) modulo \( n \), \( p \) et \( np \).
\begin{enumerate}
  \item Montrer que l'application \( \phi : {\Zz/\bigl(np\Zz\bigr)} \to {\bigl(\Zz/n\Zz\bigr) \times \bigl(\Zz/p\Zz\bigr)}, {\overline x^{\,np}} \mapsto {(\overline x^{\,n}, \overline x^{\,p})} \) est un morphisme d'anneaux.

  \item En déduire que \( \varphi(np) = \varphi(n)\varphi(p) \) (\( \varphi \) = fonction d'Euler).

  \item Vérifier que l'hypothèse \( n \wedge p = 1 \) est nécessaire.
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
