\exercice{1424, ortiz, 1999/04/01}

\enonce
\begin{enumerate}
  \item Déterminer le centre du groupe \( \mathcal{S}_n \).

  \item
        \begin{enumerate}
          \item Montrer qu'un groupe \( G_1\times G_2 \) contient un sous-groupe distingué isomorphe à \( G_1 \).

          \item Montrer que les groupes \( \mathcal{S}_n \) et \( \Zz/2\Zz\times\mathcal{A}_n \) ne sont pas isomorphes si \( n\ge 3 \).
        \end{enumerate}
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
