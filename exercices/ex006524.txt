\exercice{6524, drutu, 2011/10/16}

\enonce
\begin{enumerate}
  \item Pour les anneaux de polynômes sur un corps \( F \) montrer que \( F[x]/(x^2-1)\simeq F[x]/(x^2-4),\; \; F[x]/(x^2+1)\simeq F[x]/(x^2+2x+2) \), \( F[x,y]/(x+y)\simeq F[x],\; F[x]/(x+1)\simeq F \).

  \item Montrer que \( \Qq[x]/(x^2-2)\simeq \Qq[\sqrt{2}],\; \Zz[x]/(x^2+1)\simeq \Zz[i] \). Déduire du dernier isomorphisme que \( (x^2+1) \) est un idéal premier non-maximal.
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
