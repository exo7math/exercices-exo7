\exercice{3496, quercia, 2010/03/10}

\enonce[Diagonalisation en dimension \( 2 \)]
Diagonaliser les matrices suivantes :
\begin{enumerate}
  \item\( A = \begin{pmatrix} 1 & 5 \\ 2 & 4
        \end{pmatrix} \)
  \item \( A = \begin{pmatrix} 2 & 5 \\ 4 & 3
        \end{pmatrix} \)

  \item \( A = \begin{pmatrix} 5 & 3 \\ -8 &-6
        \end{pmatrix} \)

  \item \( A = \begin{pmatrix} 4 & 4 \\ 1 & 4
        \end{pmatrix} \)
\end{enumerate}
\finenonce

\noindication

\correction
\begin{enumerate}
  \item \( P = \begin{pmatrix} 1 & -5 \\ 1 & 2
        \end{pmatrix} \),\, \( D = \begin{pmatrix} 6 & 0 \\ 0&-1
        \end{pmatrix} \)

  \item \( P = \begin{pmatrix} 5 & 1 \\ -4 & 1
        \end{pmatrix} \),\, \( D = \begin{pmatrix}-2 & 0 \\ 0& 7
        \end{pmatrix} \)

  \item \( P = \begin{pmatrix} 3 & 1 \\ -8 &-1
        \end{pmatrix} \),\, \( \begin{pmatrix}-3 & 0 \\ 0& 2
        \end{pmatrix} \)

  \item \( P = \begin{pmatrix} 2 & -2 \\ 1 & 1
        \end{pmatrix} \),\, \( D = \begin{pmatrix} 6 & 0 \\ 0 & 2
        \end{pmatrix} \)

\end{enumerate}
\fincorrection

\finexercice
