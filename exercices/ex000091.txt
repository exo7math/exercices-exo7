\exercice{91, cousquer, 2003/10/01}

\enonce
Résoudre dans \( \Rr \) les inéquations suivantes :
\begin{align*}
  \cos(5x) + \cos(3x)     & \le \cos(x) \\
  2\cos^2(x) -9\cos(x) +4 & > 0
\end{align*}
\finenonce

\noindication

\correction
\begin{enumerate}
  \item Pour la première inéquation, on factorise à l'aide des formules de trigonométrie :
        \[
          \cos(5x) + \cos(3x) \le \cos(x) \iff \cos(x)\big(2\cos(4x) - 1\big) \le 0.
        \]
        L'étude du signe de ce produit sur \( [-\pi, \pi] \) donne pour ensemble de solutions (modulo \( 2\pi \)) :
        \[
          S_1 = \left[-\frac{7\pi}{12}, -\frac{\pi}{2}\right] \cup \left[-\frac{5\pi}{12}, -\frac{\pi}{12}\right] \cup \left[\frac{\pi}{12}, \frac{5\pi}{12}\right] \cup \left[\frac{\pi}{2}, \frac{7\pi}{12}\right] \cup \left[\frac{11\pi}{12}, \pi\right] \cup \left[-\pi, -\frac{11\pi}{12}\right].
        \]

  \item Pour la seconde inéquation, on factorise le trinôme en \( \cos(x) \) :
        \[
          2\cos^2(x) -9\cos(x) +4 > 0 \iff \big(2\cos(x) - 1\big)\big(\cos(x) - 4\big) > 0.
        \]
        Comme \( \cos(x) - 4 < 0 \) pour tout \( x\in\Rr \), l'inéquation équivaut à \( 2\cos(x) - 1 < 0 \), soit \( \cos(x) < \frac{1}{2} \). L'ensemble des solutions est donc :
        \[
          S_2 = \bigcup_{k\in\Zz} \left]\frac{\pi}{3}+2k\pi, \frac{5\pi}{3}+2k\pi\right[.
        \]
\end{enumerate}
\fincorrection

\finexercice
