\exercice{2221, matos, 2008/04/23}

\enonce[Conditionnement du problème de l'inversion d'une matrice]
Soit \( A \) une matrice inversible donnée.
\begin{enumerate}
  \item si \( (A+\delta A) \) est une matrice inversible, démontrer
        \[
          \frac{\|(A+\delta A)^{-1}-A^{-1}\|}{\|(A+\delta A)^{-1}\|} \le \mbox{cond} (A) \frac{\|\delta A\|}{\|A\|}
        \]

  \item Démontrer que
        \[
          \frac{\|(A+\delta A)^{-1}-A^{-1}\|}{\|A^{-1}\|} \le \mbox{cond} (A) \frac{\|\delta A\|}{\|A\|} (1+{ \cal O} (\|\delta A\|))
        \]
\end{enumerate}
\finenonce

\noindication

\correction
\( B = A+\delta A = A(I+A^{-1}\delta A) \) est une matrice inversible si \( \|A^{-1}\delta A\|<1 \)

\( B^{-1}- A^{-1} = A^{-1} (A-B)B^{-1} \Rightarrow \|B^{-1}- A^{-1}\|\le \|A^{-1}\|\|A-B\|\|B^{-1}\|\Rightarrow \)
\[
  \frac{\|B^{-1}- A^{-1}\|}{\|B^{-1}\|}\le \|A^{-1}\|\|A-B\| =
  \|A^{-1}\|\|\delta A\| = \mbox{cond}(A)\frac{\|\delta A\|}{\|A\|}
\]
\fincorrection

\finexercice
