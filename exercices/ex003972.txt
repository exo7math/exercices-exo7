\exercice{3972, quercia, 2010/03/11}

\enonce[Dérivées de \( e^{-1/x^2} \)]
Soit \( f : \Rr \to \Rr \)
\[
  \begin{cases}x \ne 0 & f(x) = \exp\left(-\frac 1{x^2}\right) \\
                     & f(0) = 0.
  \end{cases}
\]
Montrer que \( f \) est de classe \( \mathcal{C}^\infty \) en \( 0 \) et : \( \forall\ k \in \Nn,\ f^{(k)}(0) = 0 \).
\finenonce

\noindication

\nocorrection

\finexercice
