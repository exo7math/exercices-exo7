\exercice{6475, drutu, 2011/10/16}

\enonce
\begin{enumerate}
  \item Soient les applications \( T:\Rr \cup\{ \infty \} \to \Rr \cup\{ \infty \} ,\; T(x) = x+4,\; T(\infty ) = \infty \), et \( g:\Rr \cup\{ \infty \} \to \Rr \cup\{ \infty \} \),
        \[
          g(x) = \left\{ \begin{array}{ccc}
            \frac{x}{1-2x} & \mbox{ si } & x\in \Rr\setminus \{ \frac{1}{2}\} \; ; \\
            \infty         & \mbox{ si } & x = \frac{1}{2} \; ;                    \\
            -\frac{1}{2}   & \mbox{ si } & x = \infty \;
          \end{array}\right.
        \]
        Montrer que \( T^k([-2,2])\subset ]-\infty ,\, -2]\cup [2,\, \infty [,\; \forall k\in \Zz^* \) et que \( g^m(]-\infty ,\, -1]\cup [1,\, \infty [)\subset [-1,\, 1],\; \forall m\in \Zz^* \).

  \item Soit \( G \) le groupe des applications bijectives \( h:\Rr\cup \{ \infty \} \to \Rr\cup \{ \infty \} \), muni de l'opération de composition. Montrer que \( G \) a une action naturelle sur \( \Rr \cup\{ \infty \} \).

  \item Montrer que le sous-groupe \( \Gamma \) de \( G \) engendré par \( T \) et \( g \) est un groupe libre. {\it Indication }: Regarder les orbites des nombres dans l'intervalle \( ]1,2[ \).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
