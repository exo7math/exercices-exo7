\exercice{832, cousquer, 2003/10/01}

\enonce
Calculer les primitives suivantes.
\begin{enumerate}

  \item \( \displaystyle\int e^{\sin^2x} \sin2x\,dx \).

  \item \( \displaystyle\int\cos^5t\,dt \) ; \( \displaystyle\int\cosh^3t\,dt \); \( \displaystyle\int\cos^4t\,dt \) ; \( \displaystyle\int\sinh^4t\,dt \).

  \item \( \displaystyle\int x^3 e^x\,dx \).

  \item \( \displaystyle\int\ln x \,dx \) ; \( \displaystyle\int x\ln x\,dx \); \( \displaystyle\int\arcsin x \,dx \).

  \item \( \displaystyle\int\cosh t \sin t \, dt \).

  \item \( \displaystyle\int\frac{dx}{\sin x} \).

  \item \( \displaystyle\int\sqrt{a^2-x^2}\,dx \).

  \item \( \displaystyle\int\frac{e^{2x}}{\sqrt{e^x+1}}\,dx \).

  \item \( \displaystyle\int e^{ax}\cos bx \,dx \); \( \displaystyle\int e^{ax}\sin bx \,dx \).

  \item \( \displaystyle\int\sqrt{\frac{x}{(1-x)^3}}\,dx \)\quad pour\quad\( 0<x<1 \).

  \item \( \displaystyle\int\frac{x^2}{\sqrt{1-x^2}}\,dx \).

  \item \( \displaystyle\int\frac{dx}{\cos x+2\sin x +3} \).

  \item \( \displaystyle\int\frac{\sqrt x\,dx}{\sqrt{a^3-x^3}} \) \quad avec
        \quad \( 0<x<a \).

  \item \( \displaystyle\int\frac{\cosh x}{\cosh x+\sinh x}\,dx \).
\end{enumerate}
\finenonce

\noindication

\correction
\begin{enumerate}

  \item Changement de variable \( u = \sin^2x \) (ou d'abord \( u = \sin x \)); \( e^{\sin^2x}+C \).

  \item Deux méthodes : changement de variable \( u = \sin t \) (ou \( u = \sinh t \)), ou linéarisation.\newline \( \frac{1}{15}(15\sin t -10\sin^3 t+3\sin^5 t)+C \) ou \( \frac{1}{80}\sin5t+\frac{5}{48}\sin3t+\frac{5}{8}\sin t +C \);\newline \( \sinh t + \frac{1}{3}\sinh^3 t +C \) ou \( \frac{1}{12}\sinh3t +\frac{3}{4}\sinh t+C \);\newline \( \frac{1}{32}(\sin4t+8\sin2t+12t) +C \); \( \frac{1}{32}(\sinh4t-8\sinh2t+12t)+C \).

  \item Intégrations par parties : \( (x^3-3x^2+6x-6)e^x+C \).

  \item Intégration par parties : \( x\ln x-x+C \); \( \frac{x^2}{2}\ln x -\frac{x^2}{4}+C \); \( x\arcsin x+\sqrt{1-x^2}+C \).

  \item Intégrations par parties : \( \frac{1}{2}(\sinh t\sin t -\cosh t\cos t)+C \).

  \item Changement de variable \( t = \tan\frac{x}{2} \); \( \ln\bigl\vert\tan\frac{x}{2}\bigr\vert+C \) sur chaque intervalle\dots

  \item Changement de variable \( x = a\sin u \); \( \frac{a^2}{2}\arcsin\frac{x}{a} + \frac{x}{2}\sqrt{a^2-x^2} + C \).

  \item Changement de variable \( u = e^x \); \( \frac{2}{3}\sqrt{e^x+1}(e^x-2)+C \).

  \item Intégrations par parties : \( \frac{1}{a^2+b^2}e^{ax}(a\cos bx + b\sin bx)+C \);\newline \( \frac{1}{a^2+b^2}e^{ax}(-b\cos bx + a\sin bx)+C \).

  \item Changement de variable \( t = \sqrt{\frac{x}{1-x}} \); \( 2\sqrt{\frac{x}{1-x}}-2\arctan\sqrt{\frac{x}{1-x}}+C \).

  \item Changement de variable \( t = \arcsin x \); \( \frac{1}{2}(\arcsin x - x\sqrt{1-x^2}) + C \).

  \item Changements de variable \( u = \tan\frac{x}{2} \), \( t = 1+u \); \( \arctan(\tan\frac{x}{2}+1)+C \) sur chaque intervalle\dots{} Mais, au fait, ne cherchait-on pas une primitive {\it sur} \( \bf R \)?

  \item Changement de variable \( x^3 = u^2 \); \( \frac{2}{3}\arcsin\sqrt{\frac{x^3}{a^3}}+C \).

  \item Multiplier et diviser par \( \cosh x-\sinh x \), ou passer en \( e^x \); \( \frac{x}{2} +\frac{\sinh2x}{4}-\frac{\cosh2x}{4}+C \) ou \( \frac{x}{2}-\frac{e^{-2x}}{4}+C \).
\end{enumerate}
