\exercice{4683, quercia, 2010/03/16}

\enonce[Lemme de Césaro]
\begin{enumerate}
  \item Soit \( (b_n) \) une suite réelle strictement croissante tendant vers \( +\infty \), et \( (a_n) \) une suite réelle telle que :\par \( \frac{a_n-a_{n-1}}{b_n-b_{n-1}} \xrightarrow[n\to\infty]{} \ell \in \Rr \). Montrer que \( \frac{a_n}{b_n} \xrightarrow[n\to\infty]{} \ell \).

  \item Application : Quelle est la limite de \( \frac{1^k + 2^k + \dots + n^k}{n^{k+1}} \) \( (k \in \Nn) \) ?
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
