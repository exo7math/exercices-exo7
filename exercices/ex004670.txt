\exercice{4670, quercia, 2010/03/16}

\enonce[Suites convergeant vers 0]
\begin{enumerate}
  \item Soit \( (u_n) \) une suite réelle telle que \( \frac{u_n}{1+u_n} \xrightarrow[n\to\infty]{} 0 \). Montrer que \( u_n \xrightarrow[n\to\infty]{} 0 \).

  \item Même question avec \( \begin{cases} \frac{u_n}{1+u_n^2} \xrightarrow[n\to\infty]{} 0 \\
          (u_n) \text{ est bornée.}
        \end{cases} \)

\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
