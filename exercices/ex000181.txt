\exercice{181, gourio, 2001/09/01}

\enonce
Montrer que :
\[
  \forall n\ge 2,u_n = \sum\limits_{k = 1}^n\frac{1}{k}\notin \Nn.
\]
\emph{Indication} : montrer que
\[
  \forall n\ge 2,\exists (p_n,q_n)\in (\Nn^{*})^2,u_n = \frac{2p_n+1}{2q_n}.
\]
\finenonce

\noindication

\nocorrection

\finexercice
