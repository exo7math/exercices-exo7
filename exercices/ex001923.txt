\exercice{1923, drutu, 2003/10/01}

\enonce
En utilisant un changement de variables, calculer l'intégrale de \( f \) sur \( D \) avec
\begin{enumerate}
  \item \( D = \{ (x,y)\in \Rr^2 \mid \pi^2 < x^2+y^2 \le 4\pi^2 \}\; ;\; f(x,y) = \sin \sqrt{x^2+y^2} \) ;

  \item \( D = \left\{ (x,y)\in \Rr^2 \mid \frac{x^2}{a^2}+\frac{y^2}{b^2} \le 1 \right\}\; \mbox{avec }a\, ,\, b>0\; ;\; f(x,y) = x^2+y^2 \) ;

  \item \( D = \{ (x,y,z)\in \Rr^3 \mid x^2+y^2 \le 1\, ,\, 0\le z\le h \}\; \mbox{avec }h>0 \; ;\; f(x,y,z) = z \) ;

  \item \( D = \{ (x,y)\in \Rr^2 \mid 0 < x^2\le y\le 2x^2\, ,\, 1/x\le y \le 2/x \}\; ;\; f(x,y) = x+y \) (changement de variable \( u = y/x^2\, ,\, v = xy \)) ;

  \item \( D = \{ (x,y,z)\in \Rr^3 \mid x\ge 0\, ,\, y\ge 0\, ,\, z\ge 0\, ,\, x^2+y^2+z^2 \le 1 \}\; ;\; f(x,y,z) = xyz \) ;

  \item \( D = \{ (x,y,z)\in \Rr^3 \mid 1\le x^2+y^2+z^2 \le 4 \}\; ;\; f(x,y,z) = (x^2+y^2+z^2)^\alpha \).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
