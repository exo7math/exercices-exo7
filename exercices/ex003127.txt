\exercice{3127, quercia, 2010/03/08}

\enonce[Congruences simultanées]
Résoudre :
\begin{enumerate}
  \item \( \begin{cases}x\equiv 2 \hfill(\mathrm{mod}\, {140}) \\
          x\equiv -3 \hfill(\mathrm{mod}\, {99}).\end{cases} \)

  \item \( \begin{cases}x\equiv 3 \hfill(\mathrm{mod}\, 4) \\
          x\equiv-2 \hfill(\mathrm{mod}\, 3) \\
          x\equiv 7 \hfill(\mathrm{mod}\, 5).\end{cases} \)
\end{enumerate}
\finenonce

\noindication

\end{enumerate}
