\exercice{5968, tumpach, 2010/11/11}

\enonce
Soit \( \{f_n\}_{n\in\Nn} \) la suite de fonctions définies par :
\[
  f_n(x) = \sqrt{n} \mathbf{1}_{[n, n+\frac{1}{n}]}(x).
\]
\begin{enumerate}
  \item Montrer que \( f_n \) converge faiblement vers \( 0 \) dans \( L^2([0, +\infty[) \) mais ne converge pas fortement dans \( L^2([0, +\infty[) \).

  \item Montrer que \( f_n \) converge fortement vers \( 0 \) dans \( L^p([0, +\infty[) \) pour \( p<2 \).
\end{enumerate}
\finenonce

\noindication

\correction
Soit \( \{f_n\}_{n\in\Nn} \) la suite de fonctions définies par :
\[
  f_n(x) = \sqrt{n} \mathbf{1}_{[n, n+\frac{1}{n}]}(x).
\]
\begin{enumerate}
  \item Quelque soit \( g \) continue à support compact,
        \[
          \int_{[0, +\infty[} f_n(x) g(x)\,dx = \sqrt{n}
          \int_n^{n+\frac{1}{n}} g(x)\,dx \rightarrow 0
        \]
        quand \( n\rightarrow +\infty \). Par densité des fonctions continues à support compact, \( f_n \) converge faiblement vers \( 0 \). Comme \( f_n \) converge presque partout vers \( 0 \) on conclut comme précédemment que \( f_n \) ne converge pas fortement vers \( 0 \) dans \( L^2([0, +\infty[) \) car
        \[
          \|f_n\|_2 = 1.
        \]

  \item Pour \( p<2 \), on a :
        \[
          \int_{[0, +\infty[} |f_n(x)|\,dx = \int_n^{n+\frac{1}{n}}
          n^{\frac{p}{2}}\,dx = n^{ \frac{p}{2}-1} \rightarrow 0,
        \]
        donc \( f_n \) converge fortement vers \( 0 \) dans \( L^p([0, +\infty[) \).
\end{enumerate}
\fincorrection

\finexercice
