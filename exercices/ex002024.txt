\exercice{2024, liousse, 2003/10/01}

\enonce
\begin{enumerate}
  \item Déterminer les plans bissecteurs de :

        \( P: x+y+z+3 = 0 \) et \( P' : 2x + y+2z = 1 \)

        \( Q : 5x + 3y -4z = 8 \) et \( Q':4x-5y-3z = 2 \).

  \item Déterminer l'ensemble des points de l'espace équidistants des trois axes de coordonnées.

  \item On considère la droite \( D \) d'équation paramétrique \( \left\{\begin{array}{l}
          x = 3t-1 \\
          y = 1    \\
          z = -t-1\end{array} \right. \)

        Donner une équation des deux plans \( P \) et \( P' \) contenant \( D \) à une distance de 1 de l'origine (point \( O \) de coordonnées \( (0,0,0) \)).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
