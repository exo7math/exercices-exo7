\exercice{6736, queffelec, 2011/10/16}

\enonce
\begin{enumerate}
  \item Soit \( f(z) = \frac{1}{1+z^4} \). Montrer que \( \int_{\gamma_R}f(z)\ dz \) tend vers 0 quand \( R \) tend vers \( +\infty \), où \( \gamma_R (t) = Re^{it} \) (\( t\in [0,\pi] \)).

  \item Déduire de 1. la valeur de
        \[
          \int_{-\infty}^{+\infty}\frac{dx}{1+x^4}
        \]
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
