\exercice{2338, matexo1, 2002/02/01}

\enonce
Dire si les intégrales suivantes sont convergentes (en discutant éventuellement suivant la valeur des paramètres)\,:
\[
  \int_0^1 \frac{dt}{\sqrt t \sqrt[3]{1-t}},\quad
  \int_0^{\pi /2} \tan t\,dt,\quad
  \int_0^1 \frac{dt}{t^\alpha |\ln t|^\beta} ,\quad
  \int_0^1 \cos(\ln t)\,dt,\quad
  \int_0^1 \sin\frac{1}{t}\, dt,
\]
\[
  \int_0^\infty \frac{t^2+t-1}{\sqrt t\,(t^3 -2t^2+3t-6)}\,dt,\quad
  \int_0^\infty t^\alpha \,[1-e^{-1/\sqrt t}]\,dt,\quad
  \int_0^\infty \frac{\ln t - \ln(1-e^{-t})}{t} e^{-\alpha t}\, dt.
\]
\finenonce

\noindication

\nocorrection

\finexercice
