\exercice{911, legall, 1998/09/01}

\enonce
\begin{enumerate}
  \item On définit les fonctions suivantes : \( f_1 : t \mapsto \cos t .\mbox{ch}t, f_2 : t \mapsto \cos t .\mbox{sh}t, f_3 : t \mapsto \sin t .\mbox{ch}t, f_4 : t \mapsto
        \sin t .\mbox{sh} t \). Montrer que le système \( (f_1,f_2,f_3,f_4) \) est libre dans \( \Rr^\Rr \).

  \item Même question pour la famille \( {\mathcal F} = \{f_{\lambda} : t \mapsto e^{\lambda t}, \lambda \in \Rr\} \).
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
