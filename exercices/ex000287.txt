\exercice{287, cousquer, 2003/10/01}

\enonce
\begin{enumerate}
  \item Montrer que l'intersection de deux sous-groupes de \( \Zz \) est un sous-groupe de \( \Zz \). Caractériser le sous-groupe \( a \Zz\cap b \Zz \). Caractériser les sous-groupes suivants :
        \[
          2\Zz\cap 3\Zz\;; \quad 5 \Zz\cap 13\Zz\;;
          \quad 5 \Zz\cap 25\Zz.
        \]

  \item Montrer que toute intersection de sous-groupes de \( \Zz \) est un sous-groupe de \( \Zz \). Caractériser l'intersection d'une famille finie de sous-groupes. Caractériser les sous-groupes suivants :
        \[
          \bigcap_{n = 1}^{17} 2^n \Zz\;;\quad 4 \Zz\cap 6\Zz\cap 8\Zz\cap 19\Zz\cap 35\Zz.
        \]
\end{enumerate}
\finenonce

\noindication

\nocorrection

\finexercice
