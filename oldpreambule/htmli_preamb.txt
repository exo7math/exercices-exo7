        function GereChkbox(conteneur, a_faire) {
            const checkboxes = document.querySelectorAll(`#${conteneur} input[type="checkbox"]`);
            checkboxes.forEach(checkbox => {
                if (a_faire === '0') checkbox.checked = false;
                else if (a_faire === '1') checkbox.checked = true;
                else if (a_faire === '2') checkbox.checked = !checkbox.checked;
            });
        }

        function montre_div(nom_div) {
            // On cherche la div
            const div = document.getElementById(nom_div);
            
            // Si la div n'existe pas, on force le navigateur à afficher un message d'erreur clair !
            if (!div) {
                alert("ERREUR : Impossible de trouver le bloc caché qui s'appelle -> " + nom_div);
                return;
            }

            // Si on la trouve, on l'affiche ou on la cache
            if (window.getComputedStyle(div).display === "none") {
                div.style.display = "block";
            } else {
                div.style.display = "none";
            }
        }
    </script>
